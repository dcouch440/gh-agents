# Ticket 6.8: Plan View (/plan)

> TUI view for interactive PRD creation with the Planner Bot.

## Goal

Implement the `/plan` view - a chat interface where users interact with the Planner Bot to create Product Requirements Documents. This view has a specialized bot (different from `/main`) that guides users through the planning process.

**Checkpoint**: Enter `/plan`, have a multi-turn conversation with the Planner Bot, create milestones, and approve a PRD.

---

## Context

The Plan View is similar to Chat View (`/main`) but connects to a different bot with a different purpose:

| View | Command | Bot | Purpose |
|------|---------|-----|---------|
| Chat View | `/main` | Orchestrator | Task execution and management |
| **Plan View** | `/plan` | **Planner Bot** | PRD creation and project scoping |

The Plan View needs:
- Chat interface (similar to 6.5)
- Planning phase indicator (Discovery → Scoping → Technical → Milestones → Review)
- PRD preview panel showing current document state
- Session persistence (can leave and resume)

```rust
use ratatui::{
    layout::{Constraint, Direction, Layout, Rect},
    style::{Color, Modifier, Style},
    text::{Line, Span},
    widgets::{Block, Borders, Paragraph, Wrap},
    Frame,
};
use crate::agents::planner_bot::{PlanningSession, PlanningPhase, PlannerBot};
use crate::types::prd::PRDDocument;

pub struct PlanView {
    /// Current planning session (None if not started)
    session: Option<PlanningSession>,

    /// Input buffer for user typing
    input: String,

    /// Scroll position for chat history
    chat_scroll: u16,

    /// Scroll position for PRD preview
    prd_scroll: u16,

    /// Whether input is focused
    input_focused: bool,
}

impl PlanView {
    pub fn new() -> Self {
        Self {
            session: None,
            input: String::new(),
            chat_scroll: 0,
            prd_scroll: 0,
            input_focused: true,
        }
    }

    /// Start a new planning session
    pub fn start_session(&mut self, planner: &PlannerBot<impl LLMProvider>) {
        self.session = Some(planner.start_session(None));
    }

    /// Resume an existing session
    pub fn resume_session(&mut self, session: PlanningSession) {
        self.session = Some(session);
    }

    pub fn render(&self, frame: &mut Frame, area: Rect) {
        // Split into left (chat) and right (PRD preview)
        let chunks = Layout::default()
            .direction(Direction::Horizontal)
            .constraints([
                Constraint::Percentage(60),  // Chat
                Constraint::Percentage(40),  // PRD Preview
            ])
            .split(area);

        self.render_chat_panel(frame, chunks[0]);
        self.render_prd_panel(frame, chunks[1]);
    }

    fn render_chat_panel(&self, frame: &mut Frame, area: Rect) {
        // Split chat area into header, messages, and input
        let chunks = Layout::default()
            .direction(Direction::Vertical)
            .constraints([
                Constraint::Length(3),   // Phase indicator
                Constraint::Min(10),     // Messages
                Constraint::Length(3),   // Input
            ])
            .split(area);

        // Phase indicator
        self.render_phase_indicator(frame, chunks[0]);

        // Chat messages
        self.render_messages(frame, chunks[1]);

        // Input bar
        self.render_input(frame, chunks[2]);
    }

    fn render_phase_indicator(&self, frame: &mut Frame, area: Rect) {
        let phases = ["Discovery", "Scoping", "Technical", "Milestones", "Review"];
        let current_phase = self.session.as_ref()
            .map(|s| s.current_phase)
            .unwrap_or(PlanningPhase::Discovery);

        let phase_index = match current_phase {
            PlanningPhase::Discovery => 0,
            PlanningPhase::Scoping => 1,
            PlanningPhase::Technical => 2,
            PlanningPhase::Milestones => 3,
            PlanningPhase::Review => 4,
        };

        let spans: Vec<Span> = phases.iter().enumerate().map(|(i, phase)| {
            let style = if i == phase_index {
                Style::default().fg(Color::Green).add_modifier(Modifier::BOLD)
            } else if i < phase_index {
                Style::default().fg(Color::DarkGray)
            } else {
                Style::default().fg(Color::Gray)
            };

            let separator = if i < phases.len() - 1 { " → " } else { "" };
            Span::styled(format!("{}{}", phase, separator), style)
        }).collect();

        let phase_line = Line::from(spans);
        let block = Block::default()
            .borders(Borders::BOTTOM)
            .title(" Planning Phase ");

        let paragraph = Paragraph::new(phase_line)
            .block(block)
            .alignment(ratatui::layout::Alignment::Center);

        frame.render_widget(paragraph, area);
    }

    fn render_messages(&self, frame: &mut Frame, area: Rect) {
        let messages = self.session.as_ref()
            .map(|s| &s.messages)
            .map(|msgs| {
                msgs.iter().map(|m| {
                    let (prefix, style) = match m.role {
                        MessageRole::User => ("You: ", Style::default().fg(Color::Cyan)),
                        MessageRole::Planner => ("Planner: ", Style::default().fg(Color::Yellow)),
                    };
                    Line::from(vec![
                        Span::styled(prefix, style.add_modifier(Modifier::BOLD)),
                        Span::raw(&m.content),
                    ])
                }).collect::<Vec<_>>()
            })
            .unwrap_or_else(|| vec![
                Line::from(Span::styled(
                    "Welcome to Plan Mode! Describe what you want to build.",
                    Style::default().fg(Color::Gray).add_modifier(Modifier::ITALIC)
                ))
            ]);

        let block = Block::default()
            .borders(Borders::ALL)
            .title(" Conversation ");

        let paragraph = Paragraph::new(messages)
            .block(block)
            .wrap(Wrap { trim: true })
            .scroll((self.chat_scroll, 0));

        frame.render_widget(paragraph, area);
    }

    fn render_input(&self, frame: &mut Frame, area: Rect) {
        let style = if self.input_focused {
            Style::default().fg(Color::White)
        } else {
            Style::default().fg(Color::DarkGray)
        };

        let block = Block::default()
            .borders(Borders::ALL)
            .border_style(style)
            .title(" Your message ");

        let input = Paragraph::new(self.input.as_str())
            .block(block)
            .style(style);

        frame.render_widget(input, area);
    }

    fn render_prd_panel(&self, frame: &mut Frame, area: Rect) {
        let prd_content = self.session.as_ref()
            .map(|s| self.format_prd_preview(&s.prd))
            .unwrap_or_else(|| vec![
                Line::from(Span::styled(
                    "PRD will appear here as you plan...",
                    Style::default().fg(Color::DarkGray)
                ))
            ]);

        let block = Block::default()
            .borders(Borders::ALL)
            .title(" PRD Preview ");

        let paragraph = Paragraph::new(prd_content)
            .block(block)
            .wrap(Wrap { trim: true })
            .scroll((self.prd_scroll, 0));

        frame.render_widget(paragraph, area);
    }

    fn format_prd_preview(&self, prd: &PRDDocument) -> Vec<Line> {
        let mut lines = vec![];

        lines.push(Line::from(Span::styled(
            &prd.title,
            Style::default().add_modifier(Modifier::BOLD)
        )));
        lines.push(Line::from(""));

        if let Some(ref vision) = prd.vision {
            lines.push(Line::from(Span::styled("Vision:", Style::default().fg(Color::Green))));
            lines.push(Line::from(vision.as_str()));
            lines.push(Line::from(""));
        }

        if !prd.technical_decisions.is_empty() {
            lines.push(Line::from(Span::styled(
                format!("Technical Decisions: {}", prd.technical_decisions.len()),
                Style::default().fg(Color::Blue)
            )));
        }

        if !prd.milestones.is_empty() {
            lines.push(Line::from(Span::styled(
                format!("Milestones: {}", prd.milestones.len()),
                Style::default().fg(Color::Magenta)
            )));
            for m in &prd.milestones {
                lines.push(Line::from(format!("  M{}: {}", m.number, m.title)));
            }
        }

        lines
    }
}
```

```rust
use crossterm::event::{KeyCode, KeyEvent, KeyModifiers};

impl PlanView {
    /// Handle keyboard input, returns true if view should update
    pub fn handle_key(&mut self, key: KeyEvent) -> PlanViewAction {
        match key.code {
            KeyCode::Enter => {
                if !self.input.is_empty() {
                    let message = std::mem::take(&mut self.input);
                    return PlanViewAction::SendMessage(message);
                }
                PlanViewAction::None
            }

            KeyCode::Char(c) => {
                self.input.push(c);
                PlanViewAction::Redraw
            }

            KeyCode::Backspace => {
                self.input.pop();
                PlanViewAction::Redraw
            }

            KeyCode::Up if key.modifiers.contains(KeyModifiers::CONTROL) => {
                self.chat_scroll = self.chat_scroll.saturating_sub(1);
                PlanViewAction::Redraw
            }

            KeyCode::Down if key.modifiers.contains(KeyModifiers::CONTROL) => {
                self.chat_scroll = self.chat_scroll.saturating_add(1);
                PlanViewAction::Redraw
            }

            KeyCode::Tab => {
                // Toggle focus between chat and PRD panel for scrolling
                self.input_focused = !self.input_focused;
                PlanViewAction::Redraw
            }

            KeyCode::Esc => {
                PlanViewAction::ExitPlanMode
            }

            _ => PlanViewAction::None,
        }
    }

    /// Add a message from the Planner Bot
    pub fn add_planner_response(&mut self, response: String) {
        if let Some(ref mut session) = self.session {
            session.messages.push(PlanningMessage {
                role: MessageRole::Planner,
                content: response,
            });
            // Auto-scroll to bottom
            self.chat_scroll = u16::MAX;
        }
    }

    /// Get the current session for persistence
    pub fn get_session(&self) -> Option<&PlanningSession> {
        self.session.as_ref()
    }

    /// Check if PRD is ready to finalize
    pub fn can_finalize(&self) -> bool {
        self.session.as_ref()
            .map(|s| s.prd.is_complete() && s.current_phase == PlanningPhase::Review)
            .unwrap_or(false)
    }
}

#[derive(Debug, Clone)]
pub enum PlanViewAction {
    None,
    Redraw,
    SendMessage(String),
    ExitPlanMode,
    FinalizePRD,
}
```

```rust
use crate::tui::views::plan::{PlanView, PlanViewAction};
use crate::agents::planner_bot::PlannerBot;

impl App {
    /// Handle the /plan command
    pub fn enter_plan_mode(&mut self) {
        self.current_view = View::Plan;

        // Create new session if none exists
        if self.plan_view.get_session().is_none() {
            self.plan_view.start_session(&self.planner_bot);
        }
    }

    /// Process plan view actions
    pub async fn handle_plan_action(&mut self, action: PlanViewAction) -> Result<(), AppError> {
        match action {
            PlanViewAction::SendMessage(message) => {
                // Send to Planner Bot
                match self.planner_bot.chat(
                    self.plan_view.get_session_mut().unwrap(),
                    &message
                ).await {
                    Ok(response) => {
                        self.plan_view.add_planner_response(response);
                    }
                    Err(e) => {
                        self.plan_view.add_planner_response(
                            format!("Error: {}", e)
                        );
                    }
                }
            }

            PlanViewAction::ExitPlanMode => {
                // Save session before exiting
                if let Some(session) = self.plan_view.get_session() {
                    save_planning_session(&self.db_pool, session).await?;
                }
                self.current_view = View::Home;
            }

            PlanViewAction::FinalizePRD => {
                if let Some(session) = self.plan_view.get_session_mut() {
                    match self.planner_bot.finalize_prd(session) {
                        Ok(prd) => {
                            save_prd(&self.db_pool, &prd).await?;
                            self.show_notification(&format!(
                                "PRD '{}' approved and saved!",
                                prd.title
                            ));
                        }
                        Err(e) => {
                            self.show_notification(&format!("Cannot finalize: {}", e));
                        }
                    }
                }
            }

            _ => {}
        }

        Ok(())
    }
}
```

```rust
pub enum Command {
    Home,
    Main,
    Feed,
    Logs,
    Tasks,
    Agents,
    Costs,

    // Plan mode commands
    Plan,           // Enter plan mode (resume if session exists)
    PlanNew,        // Start new planning session
    PlanResume,     // Resume existing session (shows list if multiple)
    PlanApprove,    // Finalize and approve current PRD
    PlanExport,     // Export PRD to markdown

    Unknown(String),
}

impl Command {
    pub fn parse(input: &str) -> Self {
        let input = input.trim().to_lowercase();

        match input.as_str() {
            "/home" | "/" => Command::Home,
            "/main" => Command::Main,
            "/feed" => Command::Feed,
            "/logs" => Command::Logs,
            "/tasks" => Command::Tasks,
            "/agents" => Command::Agents,
            "/costs" => Command::Costs,

            "/plan" => Command::Plan,
            "/plan new" => Command::PlanNew,
            "/plan resume" => Command::PlanResume,
            "/plan approve" => Command::PlanApprove,
            "/plan export" => Command::PlanExport,

            other => Command::Unknown(other.to_string()),
        }
    }
}

impl App {
    pub fn execute_command(&mut self, cmd: Command) -> Result<(), AppError> {
        match cmd {
            Command::Plan => {
                self.enter_plan_mode();
            }
            Command::PlanNew => {
                self.plan_view = PlanView::new();
                self.enter_plan_mode();
            }
            Command::PlanResume => {
                // Show list of draft PRDs to resume
                self.show_prd_picker();
            }
            Command::PlanApprove => {
                self.handle_plan_action(PlanViewAction::FinalizePRD);
            }
            Command::PlanExport => {
                self.export_current_prd();
            }
            // ... other commands
            _ => {}
        }
        Ok(())
    }
}
```
