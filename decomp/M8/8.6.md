# Ticket 8.6: PR Merge Operations

> Merge PRs via GitHub API with strategy options and failure handling.

## Goal

Implement PR merge operations that allow agents to:
- Merge a PR via GitHub API
- Check and wait for mergeable status
- Support different merge strategies (merge, squash, rebase)
- Handle merge failures with actionable errors

**Checkpoint**: Agent can check if a PR is mergeable and merge it using the configured strategy.

/// Merge method for pull requests
#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, Default)]
#[serde(rename_all = "lowercase")]
pub enum MergeMethod {
    /// Create a merge commit
    #[default]
    Merge,
    /// Squash all commits into one
    Squash,
    /// Rebase commits onto base branch
    Rebase,
}

impl MergeMethod {
    pub fn as_str(&self) -> &'static str {
        match self {
            MergeMethod::Merge => "merge",
            MergeMethod::Squash => "squash",
            MergeMethod::Rebase => "rebase",
        }
    }
}

// src/github/pr.rs

#[derive(Debug, Clone, Serialize)]
pub struct MergePrRequest {
    /// Title for the merge commit (used with merge and squash)
    #[serde(skip_serializing_if = "Option::is_none")]
    pub commit_title: Option<String>,
    /// Message for the merge commit
    #[serde(skip_serializing_if = "Option::is_none")]
    pub commit_message: Option<String>,
    /// SHA of the head commit to ensure we're merging expected code
    #[serde(skip_serializing_if = "Option::is_none")]
    pub sha: Option<String>,
    /// Merge method
    pub merge_method: MergeMethod,
}

impl Default for MergePrRequest {
    fn default() -> Self {
        Self {
            commit_title: None,
            commit_message: None,
            sha: None,
            merge_method: MergeMethod::Merge,
        }
    }
}

impl MergePrRequest {
    pub fn new(method: MergeMethod) -> Self {
        Self {
            merge_method: method,
            ..Default::default()
        }
    }

    pub fn with_sha(mut self, sha: &str) -> Self {
        self.sha = Some(sha.to_string());
        self
    }

    pub fn with_message(mut self, title: &str, message: &str) -> Self {
        self.commit_title = Some(title.to_string());
        self.commit_message = Some(message.to_string());
        self
    }
}

#[derive(Debug, Clone, Deserialize)]
pub struct MergePrResponse {
    /// SHA of the merge commit
    pub sha: String,
    /// Whether the merge was successful
    pub merged: bool,
    /// Message from GitHub
    pub message: String,
}

/// Result of attempting to merge a PR
#[derive(Debug, Clone)]
pub enum MergePrResult {
    /// PR was merged successfully
    Merged {
        sha: String,
        message: String,
    },
    /// PR is not mergeable (needs update or has conflicts)
    NotMergeable {
        reason: String,
    },
    /// PR has merge conflicts
    HasConflicts,
    /// PR is already merged
    AlreadyMerged,
    /// Head SHA doesn't match (PR was updated)
    HeadMismatch {
        expected: String,
        actual: String,
    },
    /// Other failure
    Failed {
        status: u16,
        message: String,
    },
}

impl MergePrResult {
    pub fn is_success(&self) -> bool {
        matches!(self, MergePrResult::Merged { .. })
    }
}

impl GitHubClient {
    /// Merge a pull request
    pub async fn merge_pr(
        &self,
        owner: &str,
        repo: &str,
        number: u32,
        request: MergePrRequest,
    ) -> Result<MergePrResult, GitHubError> {
        let url = format!(
            "{}/repos/{}/{}/pulls/{}/merge",
            self.base_url, owner, repo, number
        );

        let response = self.client
            .put(&url)
            .headers(self.auth_headers())
            .json(&request)
            .send()
            .await
            .map_err(|e| GitHubError::Network(e.to_string()))?;

        match response.status() {
            reqwest::StatusCode::OK => {
                let result: MergePrResponse = response.json().await
                    .map_err(|e| GitHubError::ParseError(e.to_string()))?;

                tracing::info!(
                    pr = number,
                    sha = %result.sha,
                    method = ?request.merge_method,
                    "PR merged successfully"
                );

                Ok(MergePrResult::Merged {
                    sha: result.sha,
                    message: result.message,
                })
            }
            reqwest::StatusCode::METHOD_NOT_ALLOWED => {
                // 405: PR not mergeable
                let body = response.text().await.unwrap_or_default();

                tracing::warn!(pr = number, "PR not mergeable: {}", body);

                if body.contains("already merged") {
                    Ok(MergePrResult::AlreadyMerged)
                } else {
                    Ok(MergePrResult::NotMergeable { reason: body })
                }
            }
            reqwest::StatusCode::CONFLICT => {
                // 409: Merge conflict or SHA mismatch
                let body = response.text().await.unwrap_or_default();

                tracing::warn!(pr = number, "Merge conflict: {}", body);

                if body.contains("Head branch was modified") {
                    Ok(MergePrResult::HeadMismatch {
                        expected: request.sha.unwrap_or_default(),
                        actual: "unknown".to_string(),
                    })
                } else {
                    Ok(MergePrResult::HasConflicts)
                }
            }
            status => {
                let body = response.text().await.unwrap_or_default();
                Ok(MergePrResult::Failed {
                    status: status.as_u16(),
                    message: body,
                })
            }
        }
    }

    /// Merge PR with default options
    pub async fn merge_pr_simple(
        &self,
        owner: &str,
        repo: &str,
        number: u32,
        method: MergeMethod,
    ) -> Result<MergePrResult, GitHubError> {
        self.merge_pr(owner, repo, number, MergePrRequest::new(method)).await
    }
}

use std::time::Duration;
use tokio::time::sleep;

/// Mergeable status of a PR
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeableStatus {
    /// PR can be merged
    Mergeable,
    /// PR has conflicts that need resolution
    HasConflicts,
    /// PR is blocked (failing checks, review required, etc.)
    Blocked { reason: String },
    /// GitHub is still calculating - try again later
    Unknown,
    /// PR is already merged
    Merged,
    /// PR is closed
    Closed,
}

impl GitHubClient {
    /// Get the current mergeable status of a PR
    pub async fn get_mergeable_status(
        &self,
        owner: &str,
        repo: &str,
        number: u32,
    ) -> Result<MergeableStatus, GitHubError> {
        let pr = self.get_pr(owner, repo, number).await?;

        // Check if already merged or closed
        if pr.merged {
            return Ok(MergeableStatus::Merged);
        }

        if pr.state == PrState::Closed {
            return Ok(MergeableStatus::Closed);
        }

        // Check mergeable status
        match (pr.mergeable, pr.mergeable_state.as_deref()) {
            (Some(true), _) => Ok(MergeableStatus::Mergeable),
            (Some(false), Some("dirty")) => Ok(MergeableStatus::HasConflicts),
            (Some(false), Some(state)) => Ok(MergeableStatus::Blocked {
                reason: state.to_string(),
            }),
            (Some(false), None) => Ok(MergeableStatus::Blocked {
                reason: "unknown".to_string(),
            }),
            (None, _) => Ok(MergeableStatus::Unknown),
        }
    }

    /// Wait for PR to become mergeable (or definitely not)
    ///
    /// Polls until mergeable status is known (not Unknown).
    /// Returns immediately if already merged, closed, or has definite status.
    pub async fn wait_for_mergeable(
        &self,
        owner: &str,
        repo: &str,
        number: u32,
        timeout: Duration,
        poll_interval: Duration,
    ) -> Result<MergeableStatus, GitHubError> {
        let start = std::time::Instant::now();

        loop {
            let status = self.get_mergeable_status(owner, repo, number).await?;

            match status {
                MergeableStatus::Unknown => {
                    // Still calculating - check timeout and retry
                    if start.elapsed() >= timeout {
                        tracing::warn!(
                            pr = number,
                            elapsed = ?start.elapsed(),
                            "Timeout waiting for mergeable status"
                        );
                        return Ok(MergeableStatus::Unknown);
                    }

                    tracing::debug!(
                        pr = number,
                        "Mergeable status unknown, retrying in {:?}",
                        poll_interval
                    );

                    sleep(poll_interval).await;
                }
                _ => {
                    // Definite status - return it
                    tracing::debug!(
                        pr = number,
                        status = ?status,
                        "Got mergeable status"
                    );
                    return Ok(status);
                }
            }
        }
    }

    /// Wait for PR to be mergeable with default settings
    pub async fn wait_for_mergeable_default(
        &self,
        owner: &str,
        repo: &str,
        number: u32,
    ) -> Result<MergeableStatus, GitHubError> {
        self.wait_for_mergeable(
            owner,
            repo,
            number,
            Duration::from_secs(60),
            Duration::from_secs(2),
        ).await
    }
}

        assert_eq!(MergeableStatus::Mergeable, MergeableStatus::Mergeable);
        assert_ne!(MergeableStatus::Mergeable, MergeableStatus::HasConflicts);
    }
}

// src/github/types.rs

/// Repository merge settings
#[derive(Debug, Clone, Deserialize)]
pub struct RepoMergeSettings {
    /// Allow merge commits
    pub allow_merge_commit: bool,
    /// Allow squash merging
    pub allow_squash_merge: bool,
    /// Allow rebase merging
    pub allow_rebase_merge: bool,
    /// Automatically delete head branches after merge
    pub delete_branch_on_merge: Option<bool>,
}

// src/github/pr.rs

impl GitHubClient {
    /// Get repository merge settings
    pub async fn get_repo_merge_settings(
        &self,
        owner: &str,
        repo: &str,
    ) -> Result<RepoMergeSettings, GitHubError> {
        let url = format!(
            "{}/repos/{}/{}",
            self.base_url, owner, repo
        );

        let response = self.client
            .get(&url)
            .headers(self.auth_headers())
            .send()
            .await
            .map_err(|e| GitHubError::Network(e.to_string()))?;

        #[derive(Deserialize)]
        struct RepoResponse {
            allow_merge_commit: bool,
            allow_squash_merge: bool,
            allow_rebase_merge: bool,
            delete_branch_on_merge: Option<bool>,
        }

        let repo_data: RepoResponse = self.handle_response(response).await?;

        Ok(RepoMergeSettings {
            allow_merge_commit: repo_data.allow_merge_commit,
            allow_squash_merge: repo_data.allow_squash_merge,
            allow_rebase_merge: repo_data.allow_rebase_merge,
            delete_branch_on_merge: repo_data.delete_branch_on_merge,
        })
    }

    /// Check if a merge method is allowed for a repository
    pub async fn is_merge_method_allowed(
        &self,
        owner: &str,
        repo: &str,
        method: MergeMethod,
    ) -> Result<bool, GitHubError> {
        let settings = self.get_repo_merge_settings(owner, repo).await?;

        Ok(match method {
            MergeMethod::Merge => settings.allow_merge_commit,
            MergeMethod::Squash => settings.allow_squash_merge,
            MergeMethod::Rebase => settings.allow_rebase_merge,
        })
    }

    /// Get the best available merge method
    ///
    /// Returns the preferred method if allowed, otherwise falls back to first available.
    pub async fn get_available_merge_method(
        &self,
        owner: &str,
        repo: &str,
        preferred: MergeMethod,
    ) -> Result<Option<MergeMethod>, GitHubError> {
        let settings = self.get_repo_merge_settings(owner, repo).await?;

        // Check if preferred is available
        let preferred_allowed = match preferred {
            MergeMethod::Merge => settings.allow_merge_commit,
            MergeMethod::Squash => settings.allow_squash_merge,
            MergeMethod::Rebase => settings.allow_rebase_merge,
        };

        if preferred_allowed {
            return Ok(Some(preferred));
        }

        // Fallback order: squash > merge > rebase
        if settings.allow_squash_merge {
            tracing::info!(
                preferred = ?preferred,
                fallback = ?MergeMethod::Squash,
                "Using fallback merge method"
            );
            return Ok(Some(MergeMethod::Squash));
        }

        if settings.allow_merge_commit {
            tracing::info!(
                preferred = ?preferred,
                fallback = ?MergeMethod::Merge,
                "Using fallback merge method"
            );
            return Ok(Some(MergeMethod::Merge));
        }

        if settings.allow_rebase_merge {
            tracing::info!(
                preferred = ?preferred,
                fallback = ?MergeMethod::Rebase,
                "Using fallback merge method"
            );
            return Ok(Some(MergeMethod::Rebase));
        }

        // No merge methods allowed (shouldn't happen in practice)
        Ok(None)
    }

    /// Merge a PR using the best available method
    pub async fn merge_pr_auto(
        &self,
        owner: &str,
        repo: &str,
        number: u32,
        preferred_method: MergeMethod,
    ) -> Result<MergePrResult, GitHubError> {
        let method = self.get_available_merge_method(owner, repo, preferred_method)
            .await?
            .ok_or_else(|| GitHubError::Other(
                anyhow::anyhow!("No merge methods are enabled for this repository")
            ))?;

        self.merge_pr_simple(owner, repo, number, method).await
    }
}

        assert_eq!(MergeMethod::Merge.as_str(), "merge");
        assert_eq!(MergeMethod::Rebase.as_str(), "rebase");
    }
}

use thiserror::Error;

/// Detailed error for merge failures
#[derive(Error, Debug, Clone)]
pub enum MergeError {
    #[error("PR #{number} has merge conflicts")]
    Conflicts { number: u32 },

    #[error("PR #{number} is not mergeable: {reason}")]
    NotMergeable { number: u32, reason: String },

    #[error("PR #{number} was updated (expected {expected}, got {actual})")]
    HeadChanged {
        number: u32,
        expected: String,
        actual: String,
    },

    #[error("PR #{number} is already merged")]
    AlreadyMerged { number: u32 },

    #[error("PR #{number} is closed")]
    Closed { number: u32 },

    #[error("required status checks are failing for PR #{number}")]
    ChecksFailing { number: u32 },

    #[error("review approval required for PR #{number}")]
    ReviewRequired { number: u32 },

    #[error("branch protection prevents merge of PR #{number}: {reason}")]
    BranchProtection { number: u32, reason: String },

    #[error("merge method {method:?} not allowed for this repository")]
    MethodNotAllowed { method: MergeMethod },

    #[error("rate limited, retry after {retry_after:?}")]
    RateLimited { retry_after: Option<Duration> },

    #[error("API error: {message}")]
    Api { status: u16, message: String },

    #[error("network error: {0}")]
    Network(String),
}

impl MergeError {
    /// Whether this error is retryable
    pub fn is_retryable(&self) -> bool {
        matches!(self,
            MergeError::RateLimited { .. } |
            MergeError::Network(_)
        )
    }

    /// Whether this error requires conflict resolution
    pub fn needs_conflict_resolution(&self) -> bool {
        matches!(self, MergeError::Conflicts { .. })
    }

    /// Whether the PR needs to be updated before retry
    pub fn needs_refresh(&self) -> bool {
        matches!(self, MergeError::HeadChanged { .. })
    }
}

impl GitHubClient {
    /// Merge with retries for transient failures
    pub async fn merge_pr_with_retry(
        &self,
        owner: &str,
        repo: &str,
        number: u32,
        method: MergeMethod,
        max_retries: u32,
    ) -> Result<MergePrResult, MergeError> {
        let mut attempts = 0;

        loop {
            attempts += 1;

            match self.merge_pr_simple(owner, repo, number, method).await {
                Ok(result) => return Ok(result),
                Err(e) => {
                    let merge_error = self.classify_merge_error(number, e);

                    if !merge_error.is_retryable() || attempts >= max_retries {
                        return Err(merge_error);
                    }

                    let delay = match &merge_error {
                        MergeError::RateLimited { retry_after } => {
                            retry_after.unwrap_or(Duration::from_secs(60))
                        }
                        _ => Duration::from_secs(2 * attempts as u64),
                    };

                    tracing::warn!(
                        pr = number,
                        attempt = attempts,
                        delay = ?delay,
                        error = %merge_error,
                        "Merge failed, retrying"
                    );

                    sleep(delay).await;
                }
            }
        }
    }

    /// Classify a GitHub error into a specific merge error
    fn classify_merge_error(&self, number: u32, error: GitHubError) -> MergeError {
        match error {
            GitHubError::Network(msg) => MergeError::Network(msg),
            GitHubError::RateLimited { retry_after } => {
                MergeError::RateLimited {
                    retry_after: Some(Duration::from_secs(retry_after.unwrap_or(60) as u64)),
                }
            }
            GitHubError::ApiError { status, message } => {
                // Parse common error messages
                let msg_lower = message.to_lowercase();

                if msg_lower.contains("conflict") {
                    MergeError::Conflicts { number }
                } else if msg_lower.contains("status check") {
                    MergeError::ChecksFailing { number }
                } else if msg_lower.contains("review") {
                    MergeError::ReviewRequired { number }
                } else if msg_lower.contains("branch protection") || msg_lower.contains("protected") {
                    MergeError::BranchProtection {
                        number,
                        reason: message,
                    }
                } else {
                    MergeError::Api { status, message }
                }
            }
            _ => MergeError::Api {
                status: 0,
                message: error.to_string(),
            },
        }
    }

    /// Attempt merge with full error classification
    pub async fn try_merge_pr(
        &self,
        owner: &str,
        repo: &str,
        number: u32,
        method: MergeMethod,
    ) -> Result<MergePrResult, MergeError> {
        // First check if PR is in a state where merge makes sense
        let status = self.get_mergeable_status(owner, repo, number)
            .await
            .map_err(|e| MergeError::Api {
                status: 0,
                message: e.to_string(),
            })?;

        match status {
            MergeableStatus::Merged => {
                return Err(MergeError::AlreadyMerged { number });
            }
            MergeableStatus::Closed => {
                return Err(MergeError::Closed { number });
            }
            MergeableStatus::HasConflicts => {
                return Err(MergeError::Conflicts { number });
            }
            MergeableStatus::Blocked { reason } => {
                if reason.contains("behind") {
                    return Err(MergeError::NotMergeable {
                        number,
                        reason: "Branch is behind base - needs update".to_string(),
                    });
                }
            }
            MergeableStatus::Unknown => {
                tracing::debug!(pr = number, "Mergeable status unknown, attempting merge anyway");
            }
            MergeableStatus::Mergeable => {
                // Good to go
            }
        }

        // Attempt the merge
        self.merge_pr_simple(owner, repo, number, method)
            .await
            .map_err(|e| self.classify_merge_error(number, e))
    }
}
