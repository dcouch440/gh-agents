# Ticket 11.4: Session Tracking

> Track user work sessions

## Goal

Track work sessions - when the user starts nexor, what they accomplish, and session duration. This enables session-based analytics and helps users understand their workflow patterns.

**Checkpoint**: Sessions are recorded on app start/exit, session history viewable in `/stats`.

---

## Context

A "session" is a single run of the nexor application. Tracking sessions helps answer questions like "How long did I work today?" and "What did I accomplish this session?"

**Key files**:
- `src/db/sessions.rs` - Session persistence (create this)
- `src/db/mod.rs` - Module exports
- `src/tui/app.rs` - Session start/end hooks
- `migrations/XXX_sessions.sql` - New table

**Dependencies**:
- Requires M1: Database infrastructure

// src/db/sessions.rs

use chrono::{DateTime, Utc};
use sqlx::SqlitePool;
use uuid::Uuid;

#[derive(Debug, Clone)]
pub struct Session {
    pub id: String,
    pub started_at: DateTime<Utc>,
    pub ended_at: Option<DateTime<Utc>>,
    pub tasks_completed: i32,
    pub tasks_failed: i32,
    pub total_cost: f64,
    pub notes: Option<String>,
}

impl Session {
    pub fn duration(&self) -> Option<chrono::Duration> {
        self.ended_at.map(|end| end - self.started_at)
    }

    pub fn duration_formatted(&self) -> String {
        match self.duration() {
            Some(d) => {
                let hours = d.num_hours();
                let mins = d.num_minutes() % 60;
                if hours > 0 {
                    format!("{}h {}m", hours, mins)
                } else {
                    format!("{}m", mins)
                }
            }
            None => "ongoing".to_string(),
        }
    }
}

pub struct SessionStore {
    pool: SqlitePool,
}

impl SessionStore {
    pub fn new(pool: SqlitePool) -> Self {
        Self { pool }
    }

    /// Start a new session
    pub async fn start_session(&self) -> Result<Session, sqlx::Error> {
        let session = Session {
            id: Uuid::new_v4().to_string(),
            started_at: Utc::now(),
            ended_at: None,
            tasks_completed: 0,
            tasks_failed: 0,
            total_cost: 0.0,
            notes: None,
        };

        sqlx::query!(
            r#"
            INSERT INTO sessions (id, started_at, ended_at, tasks_completed, tasks_failed, total_cost)
            VALUES (?, ?, NULL, 0, 0, 0.0)
            "#,
            session.id,
            session.started_at.to_rfc3339(),
        )
        .execute(&self.pool)
        .await?;

        tracing::info!(session_id = %session.id, "Session started");

        Ok(session)
    }

    /// End a session and compute final stats
    pub async fn end_session(&self, session_id: &str) -> Result<(), sqlx::Error> {
        let ended_at = Utc::now().to_rfc3339();

        // Get session start time
        let session = sqlx::query!(
            "SELECT started_at FROM sessions WHERE id = ?",
            session_id
        )
        .fetch_one(&self.pool)
        .await?;

        // Compute stats for this session
        let stats = sqlx::query!(
            r#"
            SELECT
                COALESCE(SUM(CASE WHEN status = 'completed' THEN 1 ELSE 0 END), 0) as "completed!: i32",
                COALESCE(SUM(CASE WHEN status = 'failed' THEN 1 ELSE 0 END), 0) as "failed!: i32"
            FROM tasks
            WHERE updated_at >= ?
            "#,
            session.started_at
        )
        .fetch_one(&self.pool)
        .await?;

        let cost = sqlx::query!(
            r#"
            SELECT COALESCE(SUM(cost_usd), 0.0) as "cost!: f64"
            FROM cost_records
            WHERE timestamp >= ?
            "#,
            session.started_at
        )
        .fetch_one(&self.pool)
        .await?;

        sqlx::query!(
            r#"
            UPDATE sessions
            SET ended_at = ?, tasks_completed = ?, tasks_failed = ?, total_cost = ?
            WHERE id = ?
            "#,
            ended_at,
            stats.completed,
            stats.failed,
            cost.cost,
            session_id
        )
        .execute(&self.pool)
        .await?;

        tracing::info!(
            session_id = %session_id,
            duration_mins = ?(Utc::now() - DateTime::parse_from_rfc3339(&session.started_at).unwrap().with_timezone(&Utc)).num_minutes(),
            tasks_completed = stats.completed,
            cost = cost.cost,
            "Session ended"
        );

        Ok(())
    }

    /// Get a session by ID
    pub async fn get_session(&self, id: &str) -> Result<Option<Session>, sqlx::Error> {
        let row = sqlx::query!(
            r#"
            SELECT id, started_at, ended_at, tasks_completed, tasks_failed, total_cost, notes
            FROM sessions
            WHERE id = ?
            "#,
            id
        )
        .fetch_optional(&self.pool)
        .await?;

        Ok(row.map(|r| Session {
            id: r.id,
            started_at: DateTime::parse_from_rfc3339(&r.started_at)
                .unwrap()
                .with_timezone(&Utc),
            ended_at: r.ended_at.map(|s|
                DateTime::parse_from_rfc3339(&s).unwrap().with_timezone(&Utc)
            ),
            tasks_completed: r.tasks_completed,
            tasks_failed: r.tasks_failed,
            total_cost: r.total_cost,
            notes: r.notes,
        }))
    }
}
```

// In app.rs

pub struct App {
    // ... existing fields ...
    session_store: SessionStore,
    current_session_id: Option<String>,
}

impl App {
    pub async fn new(pool: SqlitePool) -> Result<Self, Error> {
        let session_store = SessionStore::new(pool.clone());

        // Start new session
        let session = session_store.start_session().await?;

        Ok(Self {
            // ... other fields ...
            session_store,
            current_session_id: Some(session.id),
        })
    }

    pub async fn shutdown(&mut self) -> Result<(), Error> {
        // End session before exit
        if let Some(session_id) = &self.current_session_id {
            self.session_store.end_session(session_id).await?;
        }
        Ok(())
    }
}

// In main.rs, ensure shutdown is called

async fn main() -> Result<()> {
    let mut app = App::new(pool).await?;

    // Install panic hook to try to end session
    let session_id = app.current_session_id.clone();
    let pool_for_panic = pool.clone();
    let default_hook = std::panic::take_hook();
    std::panic::set_hook(Box::new(move |info| {
        // Try to end session on panic
        if let Some(ref id) = session_id {
            let rt = tokio::runtime::Handle::current();
            let store = SessionStore::new(pool_for_panic.clone());
            let _ = rt.block_on(store.end_session(id));
        }
        default_hook(info);
    }));

    let result = app.run().await;

    // Clean shutdown
    app.shutdown().await?;

    result
}
```

impl SessionStore {
    /// Get recent sessions
    pub async fn recent_sessions(&self, limit: i32) -> Result<Vec<Session>, sqlx::Error> {
        let rows = sqlx::query!(
            r#"
            SELECT id, started_at, ended_at, tasks_completed, tasks_failed, total_cost, notes
            FROM sessions
            WHERE ended_at IS NOT NULL
            ORDER BY started_at DESC
            LIMIT ?
            "#,
            limit
        )
        .fetch_all(&self.pool)
        .await?;

        Ok(rows.into_iter().map(|r| Session {
            id: r.id,
            started_at: DateTime::parse_from_rfc3339(&r.started_at)
                .unwrap()
                .with_timezone(&Utc),
            ended_at: r.ended_at.map(|s|
                DateTime::parse_from_rfc3339(&s).unwrap().with_timezone(&Utc)
            ),
            tasks_completed: r.tasks_completed,
            tasks_failed: r.tasks_failed,
            total_cost: r.total_cost,
            notes: r.notes,
        }).collect())
    }

    /// Get session summary stats
    pub async fn session_stats(&self, since: Option<DateTime<Utc>>) -> Result<SessionStats, sqlx::Error> {
        let since_str = since
            .map(|d| d.to_rfc3339())
            .unwrap_or_else(|| "1970-01-01T00:00:00Z".to_string());

        let row = sqlx::query!(
            r#"
            SELECT
                COUNT(*) as "count!: i64",
                COALESCE(AVG(
                    (julianday(ended_at) - julianday(started_at)) * 1440
                ), 0) as "avg_duration_mins!: f64",
                COALESCE(SUM(total_cost), 0.0) as "total_cost!: f64"
            FROM sessions
            WHERE ended_at IS NOT NULL AND started_at >= ?
            "#,
            since_str
        )
        .fetch_one(&self.pool)
        .await?;

        Ok(SessionStats {
            session_count: row.count,
            avg_duration_mins: row.avg_duration_mins,
            total_cost: row.total_cost,
        })
    }
}

#[derive(Debug, Clone)]
pub struct SessionStats {
    pub session_count: i64,
    pub avg_duration_mins: f64,
    pub total_cost: f64,
}

impl StatsView {
    fn render_sessions_section(&self, frame: &mut Frame, area: Rect) {
        let block = Block::default()
            .title("Sessions")
            .borders(Borders::ALL);

        let inner = block.inner(area);
        frame.render_widget(block, area);

        if let Some(ref stats) = self.session_stats {
            let lines = vec![
                Line::from(vec![
                    Span::raw("Total: "),
                    Span::styled(
                        stats.session_count.to_string(),
                        Style::default().fg(Color::Cyan),
                    ),
                ]),
                Line::from(vec![
                    Span::raw("Avg Duration: "),
                    Span::styled(
                        format!("{:.0}m", stats.avg_duration_mins),
                        Style::default().fg(Color::Yellow),
                    ),
                ]),
                Line::from(""),
                Line::from(Span::styled("Recent:", Style::default().add_modifier(Modifier::BOLD))),
            ];

            // Add recent sessions
            let mut all_lines = lines;
            for session in &self.recent_sessions {
                let date = session.started_at.format("%m/%d").to_string();
                all_lines.push(Line::from(vec![
                    Span::raw(format!("  {} ", date)),
                    Span::styled(session.duration_formatted(), Style::default().fg(Color::DarkGray)),
                    Span::raw(format!(" {}✓ ", session.tasks_completed)),
                    Span::styled(format!("${:.2}", session.total_cost), Style::default().fg(Color::Green)),
                ]));
            }

            let paragraph = Paragraph::new(all_lines);
            frame.render_widget(paragraph, inner);
        }
    }
}
