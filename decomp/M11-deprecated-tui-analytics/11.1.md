use chrono::{DateTime, Duration, Utc};
use sqlx::SqlitePool;

pub struct Analytics {
    pool: SqlitePool,
}

#[derive(Debug, Clone)]
pub struct CostSummary {
    pub total_usd: f64,
    pub input_tokens: i64,
    pub output_tokens: i64,
    pub request_count: i64,
}

#[derive(Debug, Clone)]
pub struct CostByTier {
    pub tier: String,
    pub total_usd: f64,
    pub request_count: i64,
}

#[derive(Debug, Clone)]
pub struct CostByModel {
    pub model_id: String,
    pub total_usd: f64,
    pub input_tokens: i64,
    pub output_tokens: i64,
}

#[derive(Debug, Clone)]
pub struct CostByDay {
    pub date: String,
    pub total_usd: f64,
}

impl Analytics {
    pub fn new(pool: SqlitePool) -> Self {
        Self { pool }
    }

    /// Get total cost summary for a time period
    pub async fn cost_summary(&self, since: Option<DateTime<Utc>>) -> Result<CostSummary, sqlx::Error> {
        let since_str = since
            .map(|d| d.to_rfc3339())
            .unwrap_or_else(|| "1970-01-01T00:00:00Z".to_string());

        let row = sqlx::query!(
            r#"
            SELECT
                COALESCE(SUM(cost_usd), 0.0) as "total_usd!: f64",
                COALESCE(SUM(input_tokens), 0) as "input_tokens!: i64",
                COALESCE(SUM(output_tokens), 0) as "output_tokens!: i64",
                COUNT(*) as "request_count!: i64"
            FROM cost_records
            WHERE timestamp >= ?
            "#,
            since_str
        )
        .fetch_one(&self.pool)
        .await?;

        Ok(CostSummary {
            total_usd: row.total_usd,
            input_tokens: row.input_tokens,
            output_tokens: row.output_tokens,
            request_count: row.request_count,
        })
    }

    /// Get costs broken down by agent tier
    pub async fn costs_by_tier(&self, since: Option<DateTime<Utc>>) -> Result<Vec<CostByTier>, sqlx::Error> {
        let since_str = since
            .map(|d| d.to_rfc3339())
            .unwrap_or_else(|| "1970-01-01T00:00:00Z".to_string());

        let rows = sqlx::query!(
            r#"
            SELECT
                agent_tier as "tier!: String",
                COALESCE(SUM(cost_usd), 0.0) as "total_usd!: f64",
                COUNT(*) as "request_count!: i64"
            FROM cost_records
            WHERE timestamp >= ?
            GROUP BY agent_tier
            ORDER BY total_usd DESC
            "#,
            since_str
        )
        .fetch_all(&self.pool)
        .await?;

        Ok(rows.into_iter().map(|r| CostByTier {
            tier: r.tier,
            total_usd: r.total_usd,
            request_count: r.request_count,
        }).collect())
    }

    /// Get costs broken down by model
    pub async fn costs_by_model(&self, since: Option<DateTime<Utc>>) -> Result<Vec<CostByModel>, sqlx::Error> {
        let since_str = since
            .map(|d| d.to_rfc3339())
            .unwrap_or_else(|| "1970-01-01T00:00:00Z".to_string());

        let rows = sqlx::query!(
            r#"
            SELECT
                model_id as "model_id!: String",
                COALESCE(SUM(cost_usd), 0.0) as "total_usd!: f64",
                COALESCE(SUM(input_tokens), 0) as "input_tokens!: i64",
                COALESCE(SUM(output_tokens), 0) as "output_tokens!: i64"
            FROM cost_records
            WHERE timestamp >= ?
            GROUP BY model_id
            ORDER BY total_usd DESC
            "#,
            since_str
        )
        .fetch_all(&self.pool)
        .await?;

        Ok(rows.into_iter().map(|r| CostByModel {
            model_id: r.model_id,
            total_usd: r.total_usd,
            input_tokens: r.input_tokens,
            output_tokens: r.output_tokens,
        }).collect())
    }

    /// Get daily cost totals for trend chart
    pub async fn costs_by_day(&self, days: u32) -> Result<Vec<CostByDay>, sqlx::Error> {
        let since = Utc::now() - Duration::days(days as i64);
        let since_str = since.to_rfc3339();

        let rows = sqlx::query!(
            r#"
            SELECT
                date(timestamp) as "date!: String",
                COALESCE(SUM(cost_usd), 0.0) as "total_usd!: f64"
            FROM cost_records
            WHERE timestamp >= ?
            GROUP BY date(timestamp)
            ORDER BY date ASC
            "#,
            since_str
        )
        .fetch_all(&self.pool)
        .await?;

        Ok(rows.into_iter().map(|r| CostByDay {
            date: r.date,
            total_usd: r.total_usd,
        }).collect())
    }
}

#[derive(Debug, Clone)]
pub struct TaskMetrics {
    pub total: i64,
    pub pending: i64,
    pub in_progress: i64,
    pub completed: i64,
    pub failed: i64,
    pub success_rate: f64,
}

#[derive(Debug, Clone)]
pub struct TaskDuration {
    pub task_id: String,
    pub duration_secs: i64,
}

impl Analytics {
    /// Get task counts by status
    pub async fn task_metrics(&self, since: Option<DateTime<Utc>>) -> Result<TaskMetrics, sqlx::Error> {
        let since_str = since
            .map(|d| d.to_rfc3339())
            .unwrap_or_else(|| "1970-01-01T00:00:00Z".to_string());

        let rows = sqlx::query!(
            r#"
            SELECT
                status as "status!: String",
                COUNT(*) as "count!: i64"
            FROM tasks
            WHERE created_at >= ?
            GROUP BY status
            "#,
            since_str
        )
        .fetch_all(&self.pool)
        .await?;

        let mut metrics = TaskMetrics {
            total: 0,
            pending: 0,
            in_progress: 0,
            completed: 0,
            failed: 0,
            success_rate: 0.0,
        };

        for row in rows {
            metrics.total += row.count;
            match row.status.as_str() {
                "pending" => metrics.pending = row.count,
                "in_progress" => metrics.in_progress = row.count,
                "completed" => metrics.completed = row.count,
                "failed" => metrics.failed = row.count,
                _ => {}
            }
        }

        let denominator = metrics.completed + metrics.failed;
        if denominator > 0 {
            metrics.success_rate = (metrics.completed as f64 / denominator as f64) * 100.0;
        }

        Ok(metrics)
    }

    /// Calculate average task duration from task_events
    pub async fn avg_task_duration(&self, since: Option<DateTime<Utc>>) -> Result<Option<f64>, sqlx::Error> {
        let since_str = since
            .map(|d| d.to_rfc3339())
            .unwrap_or_else(|| "1970-01-01T00:00:00Z".to_string());

        // Get duration between 'started' and 'completed' events
        let row = sqlx::query!(
            r#"
            SELECT AVG(
                julianday(completed.timestamp) - julianday(started.timestamp)
            ) * 86400 as "avg_secs: f64"
            FROM task_events started
            JOIN task_events completed ON started.task_id = completed.task_id
            WHERE started.event_type = 'started'
              AND completed.event_type = 'completed'
              AND started.timestamp >= ?
            "#,
            since_str
        )
        .fetch_one(&self.pool)
        .await?;

        Ok(row.avg_secs)
    }
}

#[derive(Debug, Clone)]
pub struct AgentUtilization {
    pub agent_id: String,
    pub tier: String,
    pub tasks_completed: i64,
    pub tasks_failed: i64,
    pub total_cost: f64,
}

#[derive(Debug, Clone)]
pub struct EscalationStats {
    pub from_tier: String,
    pub to_tier: String,
    pub count: i64,
}

impl Analytics {
    /// Get per-agent utilization stats
    pub async fn agent_utilization(&self, since: Option<DateTime<Utc>>) -> Result<Vec<AgentUtilization>, sqlx::Error> {
        let since_str = since
            .map(|d| d.to_rfc3339())
            .unwrap_or_else(|| "1970-01-01T00:00:00Z".to_string());

        let rows = sqlx::query!(
            r#"
            SELECT
                a.id as "agent_id!: String",
                a.tier as "tier!: String",
                COALESCE(completed.count, 0) as "tasks_completed!: i64",
                COALESCE(failed.count, 0) as "tasks_failed!: i64",
                COALESCE(costs.total, 0.0) as "total_cost!: f64"
            FROM agents a
            LEFT JOIN (
                SELECT assigned_agent, COUNT(*) as count
                FROM tasks
                WHERE status = 'completed' AND updated_at >= ?
                GROUP BY assigned_agent
            ) completed ON a.id = completed.assigned_agent
            LEFT JOIN (
                SELECT assigned_agent, COUNT(*) as count
                FROM tasks
                WHERE status = 'failed' AND updated_at >= ?
                GROUP BY assigned_agent
            ) failed ON a.id = failed.assigned_agent
            LEFT JOIN (
                SELECT agent_id, SUM(cost_usd) as total
                FROM cost_records
                WHERE timestamp >= ?
                GROUP BY agent_id
            ) costs ON a.id = costs.agent_id
            ORDER BY tasks_completed DESC
            "#,
            since_str,
            since_str,
            since_str
        )
        .fetch_all(&self.pool)
        .await?;

        Ok(rows.into_iter().map(|r| AgentUtilization {
            agent_id: r.agent_id,
            tier: r.tier,
            tasks_completed: r.tasks_completed,
            tasks_failed: r.tasks_failed,
            total_cost: r.total_cost,
        }).collect())
    }

    /// Get escalation statistics
    pub async fn escalation_stats(&self, since: Option<DateTime<Utc>>) -> Result<Vec<EscalationStats>, sqlx::Error> {
        let since_str = since
            .map(|d| d.to_rfc3339())
            .unwrap_or_else(|| "1970-01-01T00:00:00Z".to_string());

        let rows = sqlx::query!(
            r#"
            SELECT
                details as "details!: String",
                COUNT(*) as "count!: i64"
            FROM task_events
            WHERE event_type = 'escalated' AND timestamp >= ?
            GROUP BY details
            "#,
            since_str
        )
        .fetch_all(&self.pool)
        .await?;

        // Parse escalation details (format: "from_tier -> to_tier")
        Ok(rows.into_iter().filter_map(|r| {
            let parts: Vec<&str> = r.details.split(" -> ").collect();
            if parts.len() == 2 {
                Some(EscalationStats {
                    from_tier: parts[0].to_string(),
                    to_tier: parts[1].to_string(),
                    count: r.count,
                })
            } else {
                None
            }
        }).collect())
    }
}
