use ratatui::{
    layout::{Constraint, Direction, Layout, Rect},
    style::{Color, Modifier, Style},
    text::{Line, Span},
    widgets::{Block, Borders, Paragraph},
    Frame,
};
use crate::db::analytics::{Analytics, CostSummary, TaskMetrics, CostByDay};

pub struct StatsView {
    /// Selected time period
    period: TimePeriod,
    /// Cached metrics
    task_metrics: Option<TaskMetrics>,
    cost_summary: Option<CostSummary>,
    daily_costs: Vec<CostByDay>,
    /// Loading state
    loading: bool,
}

#[derive(Debug, Clone, Copy, PartialEq)]
pub enum TimePeriod {
    Week,
    Month,
    AllTime,
}

impl TimePeriod {
    pub fn label(&self) -> &'static str {
        match self {
            TimePeriod::Week => "Last 7 days",
            TimePeriod::Month => "Last 30 days",
            TimePeriod::AllTime => "All time",
        }
    }

    pub fn days(&self) -> Option<u32> {
        match self {
            TimePeriod::Week => Some(7),
            TimePeriod::Month => Some(30),
            TimePeriod::AllTime => None,
        }
    }
}

impl StatsView {
    pub fn new() -> Self {
        Self {
            period: TimePeriod::Week,
            task_metrics: None,
            cost_summary: None,
            daily_costs: Vec::new(),
            loading: true,
        }
    }

    pub fn set_data(
        &mut self,
        task_metrics: TaskMetrics,
        cost_summary: CostSummary,
        daily_costs: Vec<CostByDay>,
    ) {
        self.task_metrics = Some(task_metrics);
        self.cost_summary = Some(cost_summary);
        self.daily_costs = daily_costs;
        self.loading = false;
    }

    pub fn cycle_period(&mut self) {
        self.period = match self.period {
            TimePeriod::Week => TimePeriod::Month,
            TimePeriod::Month => TimePeriod::AllTime,
            TimePeriod::AllTime => TimePeriod::Week,
        };
        self.loading = true;
    }

    pub fn render(&self, frame: &mut Frame, area: Rect) {
        // Main layout: header + content
        let chunks = Layout::default()
            .direction(Direction::Vertical)
            .constraints([
                Constraint::Length(3), // Header
                Constraint::Min(0),    // Content
            ])
            .split(area);

        self.render_header(frame, chunks[0]);
        self.render_content(frame, chunks[1]);
    }

    fn render_header(&self, frame: &mut Frame, area: Rect) {
        let title = format!("Usage Statistics - {}", self.period.label());
        let header = Paragraph::new(title)
            .style(Style::default().add_modifier(Modifier::BOLD))
            .block(Block::default().borders(Borders::BOTTOM));
        frame.render_widget(header, area);
    }

    fn render_content(&self, frame: &mut Frame, area: Rect) {
        if self.loading {
            let loading = Paragraph::new("Loading...")
                .style(Style::default().fg(Color::DarkGray));
            frame.render_widget(loading, area);
            return;
        }

        // 2x2 grid layout
        let rows = Layout::default()
            .direction(Direction::Vertical)
            .constraints([Constraint::Percentage(50), Constraint::Percentage(50)])
            .split(area);

        let top_cols = Layout::default()
            .direction(Direction::Horizontal)
            .constraints([Constraint::Percentage(50), Constraint::Percentage(50)])
            .split(rows[0]);

        let bottom_cols = Layout::default()
            .direction(Direction::Horizontal)
            .constraints([Constraint::Percentage(50), Constraint::Percentage(50)])
            .split(rows[1]);

        self.render_tasks_section(frame, top_cols[0]);
        self.render_costs_section(frame, top_cols[1]);
        self.render_tokens_section(frame, bottom_cols[0]);
        self.render_trend_section(frame, bottom_cols[1]);
    }
}
```

impl StatsView {
    fn render_tasks_section(&self, frame: &mut Frame, area: Rect) {
        let block = Block::default()
            .title("Tasks")
            .borders(Borders::ALL);

        let inner = block.inner(area);
        frame.render_widget(block, area);

        if let Some(ref metrics) = self.task_metrics {
            let success_color = if metrics.success_rate >= 90.0 {
                Color::Green
            } else if metrics.success_rate >= 70.0 {
                Color::Yellow
            } else {
                Color::Red
            };

            let lines = vec![
                Line::from(vec![
                    Span::raw("Completed: "),
                    Span::styled(
                        metrics.completed.to_string(),
                        Style::default().fg(Color::Green),
                    ),
                ]),
                Line::from(vec![
                    Span::raw("Failed: "),
                    Span::styled(
                        metrics.failed.to_string(),
                        Style::default().fg(Color::Red),
                    ),
                ]),
                Line::from(vec![
                    Span::raw("In Progress: "),
                    Span::styled(
                        metrics.in_progress.to_string(),
                        Style::default().fg(Color::Yellow),
                    ),
                ]),
                Line::from(""),
                Line::from(vec![
                    Span::raw("Success Rate: "),
                    Span::styled(
                        format!("{:.1}%", metrics.success_rate),
                        Style::default().fg(success_color).add_modifier(Modifier::BOLD),
                    ),
                ]),
            ];

            let paragraph = Paragraph::new(lines);
            frame.render_widget(paragraph, inner);
        }
    }
}

impl StatsView {
    fn render_costs_section(&self, frame: &mut Frame, area: Rect) {
        let block = Block::default()
            .title("Costs")
            .borders(Borders::ALL);

        let inner = block.inner(area);
        frame.render_widget(block, area);

        if let Some(ref summary) = self.cost_summary {
            let lines = vec![
                Line::from(vec![
                    Span::raw("Total: "),
                    Span::styled(
                        format!("${:.2}", summary.total_usd),
                        Style::default().fg(Color::Cyan).add_modifier(Modifier::BOLD),
                    ),
                ]),
                Line::from(""),
                Line::from(vec![
                    Span::styled("│ ", Style::default().fg(Color::DarkGray)),
                    Span::raw("Orchestrator: "),
                    Span::styled(
                        format!("${:.2}", summary.orchestrator_cost),
                        Style::default().fg(Color::Magenta),
                    ),
                ]),
                Line::from(vec![
                    Span::styled("│ ", Style::default().fg(Color::DarkGray)),
                    Span::raw("Worker: "),
                    Span::styled(
                        format!("${:.2}", summary.worker_cost),
                        Style::default().fg(Color::Blue),
                    ),
                ]),
                Line::from(vec![
                    Span::styled("│ ", Style::default().fg(Color::DarkGray)),
                    Span::raw("Utility: "),
                    Span::styled(
                        format!("${:.2}", summary.utility_cost),
                        Style::default().fg(Color::Green),
                    ),
                ]),
            ];

            let paragraph = Paragraph::new(lines);
            frame.render_widget(paragraph, inner);
        }
    }

    fn render_tokens_section(&self, frame: &mut Frame, area: Rect) {
        let block = Block::default()
            .title("Tokens")
            .borders(Borders::ALL);

        let inner = block.inner(area);
        frame.render_widget(block, area);

        if let Some(ref summary) = self.cost_summary {
            let format_tokens = |n: i64| -> String {
                if n >= 1_000_000 {
                    format!("{:.1}M", n as f64 / 1_000_000.0)
                } else if n >= 1_000 {
                    format!("{:.1}K", n as f64 / 1_000.0)
                } else {
                    n.to_string()
                }
            };

            let total = summary.input_tokens + summary.output_tokens;
            let avg = if summary.request_count > 0 {
                total / summary.request_count
            } else {
                0
            };

            let lines = vec![
                Line::from(vec![
                    Span::raw("Input: "),
                    Span::styled(format_tokens(summary.input_tokens), Style::default().fg(Color::Cyan)),
                ]),
                Line::from(vec![
                    Span::raw("Output: "),
                    Span::styled(format_tokens(summary.output_tokens), Style::default().fg(Color::Green)),
                ]),
                Line::from(""),
                Line::from(vec![
                    Span::raw("Avg/Request: "),
                    Span::styled(format_tokens(avg), Style::default().fg(Color::Yellow)),
                ]),
                Line::from(vec![
                    Span::raw("Requests: "),
                    Span::raw(summary.request_count.to_string()),
                ]),
            ];

            let paragraph = Paragraph::new(lines);
            frame.render_widget(paragraph, inner);
        }
    }
}

impl StatsView {
    fn render_trend_section(&self, frame: &mut Frame, area: Rect) {
        let block = Block::default()
            .title("Cost Trend")
            .borders(Borders::ALL);

        let inner = block.inner(area);
        frame.render_widget(block, area);

        if self.daily_costs.is_empty() {
            let msg = Paragraph::new("No data")
                .style(Style::default().fg(Color::DarkGray));
            frame.render_widget(msg, inner);
            return;
        }

        // Find max for scaling
        let max_cost = self.daily_costs
            .iter()
            .map(|d| d.total_usd)
            .fold(0.0_f64, f64::max);

        if max_cost == 0.0 {
            let msg = Paragraph::new("$0.00")
                .style(Style::default().fg(Color::DarkGray));
            frame.render_widget(msg, inner);
            return;
        }

        let chart_height = inner.height.saturating_sub(2) as usize;
        let chart_width = inner.width.saturating_sub(2) as usize;

        // Build ASCII chart
        let mut lines: Vec<Line> = Vec::new();

        // Y-axis label
        lines.push(Line::from(format!("${:.2}", max_cost)));

        // Chart bars
        let bar_width = chart_width / self.daily_costs.len().max(1);
        for row in (0..chart_height).rev() {
            let threshold = (row as f64 / chart_height as f64) * max_cost;
            let mut spans = Vec::new();

            for day in &self.daily_costs {
                let char = if day.total_usd >= threshold { "█" } else { " " };
                spans.push(Span::styled(
                    char.repeat(bar_width.max(1)),
                    Style::default().fg(Color::Cyan),
                ));
            }

            lines.push(Line::from(spans));
        }

        // X-axis labels (day abbreviations)
        let day_labels: Vec<Span> = self.daily_costs
            .iter()
            .map(|d| {
                let short = &d.date[5..7]; // MM from YYYY-MM-DD
                Span::raw(format!("{:^width$}", short, width = bar_width.max(1)))
            })
            .collect();
        lines.push(Line::from(day_labels));

        let chart = Paragraph::new(lines);
        frame.render_widget(chart, inner);
    }
}
```
