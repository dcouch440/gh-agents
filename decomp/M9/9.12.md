impl App {
    /// Generate status info for menu header
    pub fn get_menu_status(&self) -> MenuStatus {
        MenuStatus {
            production_state: self.production_state_string(),
            current_milestone: self.milestone_status_string(),
            pending_changes: self.pending_change_count(),
        }
    }

    fn production_state_string(&self) -> String {
        match &self.mode {
            AppMode::Normal => {
                if self.is_production_paused() {
                    "Paused".to_string()
                } else {
                    "Running".to_string()
                }
            }
            AppMode::Refactor(_) => "Refactor Mode".to_string(),
        }
    }

    fn milestone_status_string(&self) -> String {
        let current = self.current_milestone(); // e.g., "M2"
        match self.milestone_limit() {
            Some(limit) => format!("{} (limit: M{})", current, limit),
            None => current.to_string(),
        }
    }

    fn pending_change_count(&self) -> usize {
        // Count pending changes from refactor mode state
        match &self.mode {
            AppMode::Refactor(state) => state.pending_changes.len(),
            AppMode::Normal => 0,
        }
    }

    // These may need to be added or may exist:
    fn is_production_paused(&self) -> bool
    fn current_milestone(&self) -> String
    fn milestone_limit(&self) -> Option<u8>
}

impl App {
    /// Open the interactive menu
    pub async fn open_menu(&mut self) -> Result<()> {
        // Build controller with current menu state
        let menus = build_menu_tree();
        let mut controller = MenuController::new(menus);

        // Set enabled states based on current app state
        self.configure_menu_items(&mut controller);

        // Get status for display
        let status = self.get_menu_status();

        // Create terminal for menu rendering
        let mut terminal = Terminal::new()?;
        controller.open();

        // Menu event loop
        while controller.is_open() {
            // Draw
            terminal.draw(|frame| {
                if let Some(menu) = controller.current_menu() {
                    let widget = MenuWidget::new(menu, controller.state(), &status);
                    let size = menu_size(menu, &status);
                    let area = centered_rect(size.0, size.1, frame.area());
                    frame.render_widget(widget, area);
                }
            })?;

            // Handle input
            if let Some(Event::Key(key)) = terminal.poll_event(Duration::from_millis(100))? {
                if let Some(action) = controller.handle_key(key.code) {
                    terminal.restore()?;
                    self.execute_menu_action(action).await?;
                    return Ok(());
                }
            }
        }

        terminal.restore()?;
        Ok(())
    }

    fn configure_menu_items(&self, controller: &mut MenuController) {
        // Enable/disable based on state
        let in_refactor = self.mode.is_refactor();
        let has_changes = self.pending_change_count() > 0;
        let is_paused = self.is_production_paused();

        controller.set_enabled("start_production", is_paused);
        controller.set_enabled("pause_production", !is_paused && !in_refactor);
        controller.set_enabled("exit_refactor", in_refactor);
        controller.set_enabled("cancel_refactor", in_refactor);
        controller.set_enabled("apply_changes", has_changes);
        controller.set_enabled("discard_changes", has_changes);
        controller.set_enabled("review_changes", has_changes);

        // Update dynamic labels
        if has_changes {
            controller.update_label(
                "pending_changes_submenu",
                &format!("Pending Changes ({})", self.pending_change_count())
            );
        }
    }

impl App {
    async fn execute_menu_action(&mut self, action: MenuAction) -> Result<()> {
        match action {
            // Production control
            MenuAction::StartProduction => {
                self.resume_production().await?;
            }
            MenuAction::PauseProduction => {
                self.pause_production().await?;
            }
            MenuAction::SetMilestoneLimit(milestone) => {
                set_milestone_limit(&self.db, Some(milestone)).await?;
                self.cached_milestone_limit = Some(milestone);
            }
            MenuAction::ClearMilestoneLimit => {
                set_milestone_limit(&self.db, None).await?;
                self.cached_milestone_limit = None;
            }

            // Refactor mode
            MenuAction::ExitRefactorMode => {
                self.exit_refactor_mode().await?;
            }
            MenuAction::CancelRefactor => {
                self.cancel_refactor_mode().await?;  // Exit without applying
            }
            MenuAction::ApplyChanges => {
                self.apply_refactor_changes().await?;
            }
            MenuAction::DiscardChanges => {
                self.discard_refactor_changes().await?;
            }
            MenuAction::ReviewChanges => {
                self.review_pending_changes().await?;
