# Ticket 13.2: Settings Page

> User preferences and application configuration.

## Goal

Settings page for configuring API keys, preferences, and application behavior.

**Checkpoint**: Settings save and persist, changes take effect immediately.

---

## Dependencies

- M11 (React Foundation)
- M10.5 (Auth for user context)

---

## Slices

### Slice 13.2.1: Add Settings API Endpoints

Add to Rust server:
- `GET /api/settings` - Get current settings
- `PUT /api/settings` - Update settings
- `POST /api/settings/api-keys` - Add API key
- `DELETE /api/settings/api-keys/:provider` - Remove API key

**Settings schema**:
```rust
#[derive(Serialize, Deserialize)]
pub struct Settings {
    pub theme: String,           // "dark" | "light" | "system"
    pub notifications: bool,
    pub auto_scroll: bool,
    pub font_size: u8,           // 12-18
    pub default_model: String,
    pub api_keys: Vec<ApiKeyInfo>,
}

#[derive(Serialize, Deserialize)]
pub struct ApiKeyInfo {
    pub provider: String,        // "anthropic" | "openai"
    pub key_preview: String,     // "sk-...abc" (last 3 chars)
    pub added_at: String,
}
```

### Slice 13.2.2: Create SettingsPage

**Create `ui/src/pages/SettingsPage.tsx`**:
```tsx
import { useEffect, useState } from 'react';
import { api } from '../api/client';
import { Save, Key, Trash2 } from 'lucide-react';

interface Settings {
  theme: string;
  notifications: boolean;
  auto_scroll: boolean;
  font_size: number;
  default_model: string;
  api_keys: Array<{ provider: string; key_preview: string; added_at: string }>;
}

export function SettingsPage() {
  const [settings, setSettings] = useState<Settings | null>(null);
  const [saving, setSaving] = useState(false);
  const [newKey, setNewKey] = useState({ provider: 'anthropic', key: '' });

  useEffect(() => {
    api.settings.get().then(setSettings);
  }, []);

  const handleSave = async () => {
    if (!settings) return;
    setSaving(true);
    await api.settings.update(settings);
    setSaving(false);
  };

  const handleAddKey = async () => {
    if (!newKey.key) return;
    await api.settings.addApiKey(newKey.provider, newKey.key);
    setNewKey({ provider: 'anthropic', key: '' });
    // Refresh settings
    const updated = await api.settings.get();
    setSettings(updated);
  };

  const handleRemoveKey = async (provider: string) => {
    await api.settings.removeApiKey(provider);
    const updated = await api.settings.get();
    setSettings(updated);
  };

  if (!settings) {
    return <div className="text-text-secondary">Loading settings...</div>;
  }

  return (
    <div className="max-w-2xl space-y-6">
      <div className="flex items-center justify-between">
        <h2 className="text-lg font-medium text-text-primary">Settings</h2>
        <button
          onClick={handleSave}
          disabled={saving}
          className="flex items-center gap-2 px-4 py-2 bg-accent text-white rounded
                     disabled:opacity-50"
        >
          <Save size={16} />
          {saving ? 'Saving...' : 'Save Changes'}
        </button>
      </div>

      {/* Appearance */}
      <section className="bg-bg-secondary rounded-lg border border-border p-4">
        <h3 className="text-sm font-medium text-text-primary mb-4">Appearance</h3>
        <div className="space-y-4">
          <div>
            <label className="block text-sm text-text-secondary mb-1">Theme</label>
            <select
              value={settings.theme}
              onChange={(e) => setSettings({ ...settings, theme: e.target.value })}
              className="w-full bg-bg-primary border border-border rounded px-3 py-2
                         text-text-primary"
            >
              <option value="dark">Dark</option>
              <option value="light">Light</option>
              <option value="system">System</option>
            </select>
          </div>
          <div>
            <label className="block text-sm text-text-secondary mb-1">
              Font Size: {settings.font_size}px
            </label>
            <input
              type="range"
              min={12}
              max={18}
              value={settings.font_size}
              onChange={(e) => setSettings({ ...settings, font_size: Number(e.target.value) })}
              className="w-full"
            />
          </div>
        </div>
      </section>

      {/* Behavior */}
      <section className="bg-bg-secondary rounded-lg border border-border p-4">
        <h3 className="text-sm font-medium text-text-primary mb-4">Behavior</h3>
        <div className="space-y-3">
          <label className="flex items-center gap-3">
            <input
              type="checkbox"
              checked={settings.notifications}
              onChange={(e) => setSettings({ ...settings, notifications: e.target.checked })}
              className="rounded border-border"
            />
            <span className="text-text-primary">Enable notifications</span>
          </label>
          <label className="flex items-center gap-3">
            <input
              type="checkbox"
              checked={settings.auto_scroll}
              onChange={(e) => setSettings({ ...settings, auto_scroll: e.target.checked })}
              className="rounded border-border"
            />
            <span className="text-text-primary">Auto-scroll feed</span>
          </label>
        </div>
      </section>

      {/* API Keys */}
      <section className="bg-bg-secondary rounded-lg border border-border p-4">
        <h3 className="text-sm font-medium text-text-primary mb-4">API Keys</h3>

        {/* Existing keys */}
        {settings.api_keys.length > 0 && (
          <div className="space-y-2 mb-4">
            {settings.api_keys.map((key) => (
              <div
                key={key.provider}
                className="flex items-center justify-between p-3 bg-bg-primary rounded"
              >
                <div className="flex items-center gap-3">
                  <Key size={16} className="text-text-tertiary" />
                  <div>
                    <div className="text-text-primary capitalize">{key.provider}</div>
                    <div className="text-sm text-text-tertiary font-mono">{key.key_preview}</div>
                  </div>
                </div>
                <button
                  onClick={() => handleRemoveKey(key.provider)}
                  className="p-2 text-text-tertiary hover:text-status-error"
                >
                  <Trash2 size={16} />
                </button>
              </div>
            ))}
          </div>
        )}

        {/* Add new key */}
        <div className="flex gap-2">
          <select
            value={newKey.provider}
            onChange={(e) => setNewKey({ ...newKey, provider: e.target.value })}
            className="bg-bg-primary border border-border rounded px-3 py-2 text-text-primary"
          >
            <option value="anthropic">Anthropic</option>
            <option value="openai">OpenAI</option>
          </select>
          <input
            type="password"
            value={newKey.key}
            onChange={(e) => setNewKey({ ...newKey, key: e.target.value })}
            placeholder="API key..."
            className="flex-1 bg-bg-primary border border-border rounded px-3 py-2
                       text-text-primary placeholder-text-tertiary font-mono"
          />
          <button
            onClick={handleAddKey}
            disabled={!newKey.key}
            className="px-4 py-2 bg-accent text-white rounded disabled:opacity-50"
          >
            Add
          </button>
        </div>
      </section>

      {/* Model Selection */}
      <section className="bg-bg-secondary rounded-lg border border-border p-4">
        <h3 className="text-sm font-medium text-text-primary mb-4">Default Model</h3>
        <select
          value={settings.default_model}
          onChange={(e) => setSettings({ ...settings, default_model: e.target.value })}
          className="w-full bg-bg-primary border border-border rounded px-3 py-2
                     text-text-primary"
        >
          <option value="claude-sonnet-4-20250514">Claude Sonnet 4</option>
          <option value="claude-opus-4-20250514">Claude Opus 4</option>
          <option value="gpt-4o">GPT-4o</option>
        </select>
      </section>
    </div>
  );
}
```

### Slice 13.2.3: Persist Settings with Local Storage Sync

Add settings persistence hook:

**Create `ui/src/hooks/useSettings.ts`**:
```tsx
import { create } from 'zustand';
import { persist } from 'zustand/middleware';

interface SettingsState {
  theme: string;
  fontSize: number;
  autoScroll: boolean;
  notifications: boolean;
  setTheme: (theme: string) => void;
  setFontSize: (size: number) => void;
  setAutoScroll: (enabled: boolean) => void;
  setNotifications: (enabled: boolean) => void;
}

export const useSettings = create<SettingsState>()(
  persist(
    (set) => ({
      theme: 'dark',
      fontSize: 14,
      autoScroll: true,
      notifications: true,
      setTheme: (theme) => set({ theme }),
      setFontSize: (fontSize) => set({ fontSize }),
      setAutoScroll: (autoScroll) => set({ autoScroll }),
      setNotifications: (notifications) => set({ notifications }),
    }),
    {
      name: 'nexor-settings',
    }
  )
);
```

Apply settings to document:
```tsx
// In App.tsx or root component
useEffect(() => {
  const { fontSize, theme } = useSettings.getState();
  document.documentElement.style.fontSize = `${fontSize}px`;
  document.documentElement.setAttribute('data-theme', theme);
}, []);
```
