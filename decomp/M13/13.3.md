### Slice 13.3.1: Add Mobile Navigation

**Update `ui/src/components/Layout.tsx`**:
```tsx
import { useState } from 'react';
import { Outlet, NavLink } from 'react-router-dom';
import { Menu, X, MessageSquare, Activity, CheckSquare, Bot, Settings, BarChart } from 'lucide-react';

const navItems = [
  { to: '/', icon: MessageSquare, label: 'Chat' },
  { to: '/feed', icon: Activity, label: 'Feed' },
  { to: '/tasks', icon: CheckSquare, label: 'Tasks' },
  { to: '/agents', icon: Bot, label: 'Agents' },
  { to: '/analytics', icon: BarChart, label: 'Analytics' },
  { to: '/settings', icon: Settings, label: 'Settings' },
];

export function Layout() {
  const [mobileMenuOpen, setMobileMenuOpen] = useState(false);

  return (
    <div className="h-screen flex flex-col md:flex-row bg-bg-primary">
      {/* Mobile header */}
      <header className="md:hidden flex items-center justify-between p-4 border-b border-border">
        <span className="font-mono text-text-primary font-medium">nexor</span>
        <button
          onClick={() => setMobileMenuOpen(!mobileMenuOpen)}
          className="p-2 text-text-secondary hover:text-text-primary"
        >
          {mobileMenuOpen ? <X size={24} /> : <Menu size={24} />}
        </button>
      </header>

      {/* Mobile menu overlay */}
      {mobileMenuOpen && (
        <div
          className="md:hidden fixed inset-0 z-40 bg-black/50"
          onClick={() => setMobileMenuOpen(false)}
        />
      )}

      {/* Sidebar - hidden on mobile unless menu open */}
      <aside
        className={`
          fixed md:static inset-y-0 left-0 z-50
          w-64 md:w-16 lg:w-48
          bg-bg-secondary border-r border-border
          transform transition-transform duration-200
          ${mobileMenuOpen ? 'translate-x-0' : '-translate-x-full md:translate-x-0'}
        `}
      >
        {/* Logo - desktop only */}
        <div className="hidden md:flex items-center justify-center h-14 border-b border-border">
          <span className="font-mono text-text-primary font-medium lg:block md:hidden">nexor</span>
          <span className="font-mono text-text-primary font-medium md:block lg:hidden">n</span>
        </div>

        {/* Navigation */}
        <nav className="p-2 space-y-1 mt-4 md:mt-0">
          {navItems.map(({ to, icon: Icon, label }) => (
            <NavLink
              key={to}
              to={to}
              onClick={() => setMobileMenuOpen(false)}
              className={({ isActive }) =>
                `flex items-center gap-3 px-3 py-2 rounded text-sm
                 ${isActive
                   ? 'bg-accent/10 text-accent'
                   : 'text-text-secondary hover:text-text-primary hover:bg-bg-tertiary'
                 }`
              }
            >
              <Icon size={18} />
              <span className="md:hidden lg:inline">{label}</span>
            </NavLink>
          ))}
        </nav>
      </aside>

      {/* Main content */}
      <main className="flex-1 overflow-auto">
        <div className="p-4 md:p-6 max-w-6xl mx-auto">
          <Outlet />
        </div>
      </main>
    </div>
  );
}
```

**Update `ui/src/components/ChatMessage.tsx`**:
```tsx
export function ChatMessage({ message }: ChatMessageProps) {
  const isUser = message.role === 'user';

  return (
    <div className={`flex gap-3 ${isUser ? 'flex-row-reverse' : ''}`}>
      {/* Avatar - smaller on mobile */}
      <div
        className={`w-8 h-8 md:w-10 md:h-10 rounded-full flex items-center justify-center
                    flex-shrink-0 ${isUser ? 'bg-accent' : 'bg-bg-tertiary'}`}
      >
        {isUser ? <User size={16} /> : <Bot size={16} />}
      </div>

      {/* Message content */}
      <div
        className={`max-w-[85%] md:max-w-[75%] rounded-lg px-3 py-2 md:px-4 md:py-3
                    ${isUser
                      ? 'bg-accent text-white'
                      : 'bg-bg-secondary text-text-primary'
                    }`}
      >
        <div className="text-sm md:text-base whitespace-pre-wrap break-words">
          {message.content}
        </div>
      </div>
    </div>
  );
}
```

  /* Ensure minimum touch target size */

    min-width: 44px;
