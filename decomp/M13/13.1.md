# Ticket 13.1: Analytics Dashboard

> Visualize agent performance and task metrics.

## Goal

Dashboard showing key metrics: task completion rates, agent utilization, cost tracking.

**Checkpoint**: Dashboard displays charts, metrics update in real-time.

---

## Dependencies

- M11 (React Foundation)
- M12.3 (Tasks View for data)
- M12.4 (Agents View for data)

---

## Slices

### Slice 13.1.1: Add Analytics API Endpoint

Add to Rust server `src/server/api.rs`:
- `GET /api/analytics/summary` - Overall stats
- `GET /api/analytics/tasks` - Task metrics over time
- `GET /api/analytics/agents` - Agent performance metrics
- `GET /api/analytics/costs` - Token usage and costs

**Response schema**:
```rust
#[derive(Serialize)]
pub struct AnalyticsSummary {
    pub total_tasks: u64,
    pub completed_tasks: u64,
    pub failed_tasks: u64,
    pub active_agents: u64,
    pub total_tokens: u64,
    pub estimated_cost: f64,
}
```

### Slice 13.1.2: Create AnalyticsPage

**Create `ui/src/pages/AnalyticsPage.tsx`**:
```tsx
import { useEffect, useState } from 'react';
import { api } from '../api/client';
import { StatCard } from '../components/StatCard';
import { TaskChart } from '../components/TaskChart';
import { AgentChart } from '../components/AgentChart';

interface Summary {
  total_tasks: number;
  completed_tasks: number;
  failed_tasks: number;
  active_agents: number;
  total_tokens: number;
  estimated_cost: number;
}

export function AnalyticsPage() {
  const [summary, setSummary] = useState<Summary | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    api.analytics.summary().then((data) => {
      setSummary(data);
      setLoading(false);
    });
  }, []);

  if (loading) {
    return <div className="text-text-secondary">Loading analytics...</div>;
  }

  const completionRate = summary
    ? Math.round((summary.completed_tasks / summary.total_tasks) * 100) || 0
    : 0;

  return (
    <div className="space-y-6">
      <h2 className="text-lg font-medium text-text-primary">Analytics</h2>

      {/* Summary Cards */}
      <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
        <StatCard
          label="Total Tasks"
          value={summary?.total_tasks ?? 0}
        />
        <StatCard
          label="Completion Rate"
          value={`${completionRate}%`}
          trend={completionRate > 80 ? 'up' : completionRate < 50 ? 'down' : 'neutral'}
        />
        <StatCard
          label="Active Agents"
          value={summary?.active_agents ?? 0}
        />
        <StatCard
          label="Est. Cost"
          value={`$${(summary?.estimated_cost ?? 0).toFixed(2)}`}
        />
      </div>

      {/* Charts */}
      <div className="grid md:grid-cols-2 gap-6">
        <div className="bg-bg-secondary rounded-lg border border-border p-4">
          <h3 className="text-sm font-medium text-text-secondary mb-4">
            Tasks Over Time
          </h3>
          <TaskChart />
        </div>
        <div className="bg-bg-secondary rounded-lg border border-border p-4">
          <h3 className="text-sm font-medium text-text-secondary mb-4">
            Agent Utilization
          </h3>
          <AgentChart />
        </div>
      </div>
    </div>
  );
}
```

### Slice 13.1.3: Create Chart Components

**Commands**:
```bash
npm install recharts
```

**Create `ui/src/components/StatCard.tsx`**:
```tsx
import { TrendingUp, TrendingDown, Minus } from 'lucide-react';

interface StatCardProps {
  label: string;
  value: string | number;
  trend?: 'up' | 'down' | 'neutral';
}

export function StatCard({ label, value, trend }: StatCardProps) {
  const TrendIcon = trend === 'up' ? TrendingUp : trend === 'down' ? TrendingDown : Minus;
  const trendColor = trend === 'up' ? 'text-status-success' : trend === 'down' ? 'text-status-error' : 'text-text-tertiary';

  return (
    <div className="bg-bg-secondary rounded-lg border border-border p-4">
      <div className="text-sm text-text-secondary">{label}</div>
      <div className="flex items-center gap-2 mt-1">
        <span className="text-2xl font-mono text-text-primary">{value}</span>
        {trend && <TrendIcon size={16} className={trendColor} />}
      </div>
    </div>
  );
}
```

**Create `ui/src/components/TaskChart.tsx`**:
```tsx
import { useEffect, useState } from 'react';
import { AreaChart, Area, XAxis, YAxis, Tooltip, ResponsiveContainer } from 'recharts';
import { api } from '../api/client';

interface DataPoint {
  date: string;
  completed: number;
  failed: number;
}

export function TaskChart() {
  const [data, setData] = useState<DataPoint[]>([]);

  useEffect(() => {
    api.analytics.tasks().then(setData);
  }, []);

  return (
    <ResponsiveContainer width="100%" height={200}>
      <AreaChart data={data}>
        <XAxis
          dataKey="date"
          stroke="#8b949e"
          fontSize={12}
          tickLine={false}
        />
        <YAxis
          stroke="#8b949e"
          fontSize={12}
          tickLine={false}
          axisLine={false}
        />
        <Tooltip
          contentStyle={{
            backgroundColor: '#161b22',
            border: '1px solid #30363d',
            borderRadius: '6px',
          }}
        />
        <Area
          type="monotone"
          dataKey="completed"
          stroke="#3fb950"
          fill="#3fb95020"
          stackId="1"
        />
        <Area
          type="monotone"
          dataKey="failed"
          stroke="#f85149"
          fill="#f8514920"
```

### Slice 13.1.4: Add Real-time Updates

Subscribe to WebSocket for live metric updates:
```tsx
useEffect(() => {
  wsClient.subscribe(['analytics']);

  const handleUpdate = (data: Partial<Summary>) => {
    setSummary(prev => prev ? { ...prev, ...data } : null);
  };

  wsClient.on('analytics_update', handleUpdate);

  return () => {
    wsClient.off('analytics_update', handleUpdate);
    wsClient.unsubscribe(['analytics']);
  };
}, []);
```
