# Ticket 10.2: File Editor Widget

> Full text editing using tui-textarea

## Goal

Create a full-featured text editor widget that allows users to edit files directly within the TUI. Uses nano-style keybindings for familiarity, tracks modification state, and supports undo/redo.

**Checkpoint**: Can open a file, edit content, undo/redo changes, search, and go to specific line.

```rust
use ratatui::{layout::Rect, Frame};
use std::path::PathBuf;
use tui_textarea::{Input, Key, TextArea};

pub struct FileEditor<'a> {
    /// File path being edited
    path: PathBuf,
    /// Original content (for detecting changes)
    original_content: String,
    /// The text area widget
    textarea: TextArea<'a>,
    /// Whether the file has been modified
    modified: bool,
}

impl<'a> FileEditor<'a> {
    pub fn new(path: PathBuf, content: String) -> Self {
        let lines: Vec<String> = content.lines().map(String::from).collect();
        let mut textarea = TextArea::new(lines);

        // Configure the text area
        textarea.set_cursor_line_style(ratatui::style::Style::default());
        textarea.set_line_number_style(ratatui::style::Style::default().fg(ratatui::style::Color::DarkGray));

        Self {
            path,
            original_content: content,
            textarea,
            modified: false,
        }
    }

    pub fn path(&self) -> &PathBuf {
        &self.path
    }

    pub fn is_modified(&self) -> bool {
        self.modified
    }

    pub fn content(&self) -> String {
        self.textarea.lines().join("\n")
    }

    pub fn handle_input(&mut self, input: Input) -> EditorAction {
        match input {
            Input { key: Key::Char('x'), ctrl: true, .. } => {
                EditorAction::Exit
            }
            Input { key: Key::Char('o'), ctrl: true, .. } => {
                EditorAction::Save
            }
            _ => {
                let before = self.textarea.lines().join("\n");
                self.textarea.input(input);
                let after = self.textarea.lines().join("\n");

                if before != after {
                    self.modified = self.original_content != after;
                }

                EditorAction::Continue
            }
        }
    }

    pub fn render(&self, frame: &mut Frame, area: Rect) {
        frame.render_widget(self.textarea.widget(), area);
    }
}

#[derive(Debug, Clone, PartialEq)]
pub enum EditorAction {
    Continue,
    Save,
    Exit,
    GoToLine,
    Search,
}
```

```rust
impl<'a> FileEditor<'a> {
    pub fn handle_input(&mut self, input: Input) -> EditorAction {
        match input {
            // Exit
            Input { key: Key::Char('x'), ctrl: true, .. } => {
                EditorAction::Exit
            }
            // Save
            Input { key: Key::Char('o'), ctrl: true, .. } => {
                EditorAction::Save
            }
            // Cut line
            Input { key: Key::Char('k'), ctrl: true, .. } => {
                self.cut_line();
                EditorAction::Continue
            }
            // Paste (uncut)
            Input { key: Key::Char('u'), ctrl: true, .. } => {
                self.paste();
                EditorAction::Continue
            }
            // Go to line
            Input { key: Key::Char('g'), ctrl: true, .. } => {
                EditorAction::GoToLine
            }
            // Search
            Input { key: Key::Char('w'), ctrl: true, .. } => {
                EditorAction::Search
            }
            // Regular input
            _ => {
                let before = self.textarea.lines().join("\n");
                self.textarea.input(input);
                let after = self.textarea.lines().join("\n");

                if before != after {
                    self.modified = self.original_content != after;
                }

                EditorAction::Continue
            }
        }
    }

    fn cut_line(&mut self) {
        // Store current line in clipboard
        let (row, _) = self.textarea.cursor();
        if let Some(line) = self.textarea.lines().get(row) {
            self.clipboard = Some(line.clone());
        }
        // Delete the line
        self.textarea.delete_line_by_head();
        self.textarea.delete_line_by_end();
        self.modified = true;
    }

    fn paste(&mut self) {
        if let Some(ref text) = self.clipboard {
            self.textarea.insert_str(text);
            self.modified = true;
        }
    }

    fn render_help_bar(&self, frame: &mut Frame, area: Rect) {
        let help = "^X Exit  ^O Save  ^K Cut  ^U Paste  ^G GoTo  ^W Search";
        let widget = Paragraph::new(help)
            .style(Style::default().bg(Color::DarkGray).fg(Color::White));
        frame.render_widget(widget, area);
    }
}
```

```rust
impl<'a> FileEditor<'a> {
    fn render_status_bar(&self, frame: &mut Frame, area: Rect) {
        let (row, col) = self.textarea.cursor();
        let total_lines = self.textarea.lines().len();

        let modified_indicator = if self.modified { " [Modified]" } else { "" };

        let status = format!(
            " {} | Ln {}, Col {} | {} lines{}",
            self.path.display(),
            row + 1,
            col + 1,
            total_lines,
            modified_indicator,
        );

        let style = if self.modified {
            Style::default().bg(Color::Yellow).fg(Color::Black)
        } else {
            Style::default().bg(Color::DarkGray).fg(Color::White)
        };

        let widget = Paragraph::new(status).style(style);
        frame.render_widget(widget, area);
    }

    pub fn mark_saved(&mut self) {
        self.original_content = self.content();
        self.modified = false;
    }
}
```

```rust
pub struct FileEditor<'a> {
    // ... existing fields ...

    /// Active modal (GoToLine or Search)
    modal: Option<EditorModal>,
    /// Search query
    search_query: String,
    /// Search matches: (line, start_col, end_col)
    search_matches: Vec<(usize, usize, usize)>,
    /// Current match index
    current_match: Option<usize>,
}

#[derive(Debug, Clone)]
pub enum EditorModal {
    GoToLine { input: String },
    Search { input: String },
}

impl<'a> FileEditor<'a> {
    pub fn enter_goto_mode(&mut self) {
        self.modal = Some(EditorModal::GoToLine { input: String::new() });
    }

    pub fn enter_search_mode(&mut self) {
        self.modal = Some(EditorModal::Search { input: self.search_query.clone() });
    }

    pub fn handle_modal_input(&mut self, input: Input) -> bool {
        match &mut self.modal {
            Some(EditorModal::GoToLine { input: line_input }) => {
                match input.key {
                    Key::Enter => {
                        if let Ok(line_num) = line_input.parse::<usize>() {
                            self.goto_line(line_num);
                        }
                        self.modal = None;
                        true
                    }
                    Key::Esc => {
                        self.modal = None;
                        true
                    }
                    Key::Char(c) if c.is_ascii_digit() => {
                        line_input.push(c);
                        true
                    }
                    Key::Backspace => {
                        line_input.pop();
                        true
                    }
                    _ => true,
                }
            }
            Some(EditorModal::Search { input: search_input }) => {
                match input.key {
                    Key::Enter => {
                        self.search_query = search_input.clone();
                        self.perform_search();
                        self.next_match();
                        true
                    }
                    Key::Esc => {
                        self.modal = None;
                        true
                    }
                    Key::Char(c) => {
                        search_input.push(c);
                        self.search_query = search_input.clone();
                        self.perform_search();
                        true
                    }
                    Key::Backspace => {
                        search_input.pop();
                        self.search_query = search_input.clone();
                        self.perform_search();
                        true
                    }
                    _ => true,
                }
            }
            None => false,
        }
    }

    fn goto_line(&mut self, line: usize) {
        let target = line.saturating_sub(1); // Convert to 0-indexed
        let max_line = self.textarea.lines().len().saturating_sub(1);
        let target = target.min(max_line);
        self.textarea.move_cursor(tui_textarea::CursorMove::Jump(target as u16, 0));
    }

    fn perform_search(&mut self) {
        self.search_matches.clear();
        if self.search_query.is_empty() {
            return;
        }

        let query_lower = self.search_query.to_lowercase();
        for (line_idx, line) in self.textarea.lines().iter().enumerate() {
            let line_lower = line.to_lowercase();
            let mut col = 0;
            while let Some(pos) = line_lower[col..].find(&query_lower) {
                let start = col + pos;
                let end = start + self.search_query.len();
                self.search_matches.push((line_idx, start, end));
                col = end;
            }
        }
    }

    fn next_match(&mut self) {
        if self.search_matches.is_empty() {
            self.current_match = None;
            return;
        }

        let next = match self.current_match {
            Some(i) => (i + 1) % self.search_matches.len(),
            None => 0,
        };
        self.current_match = Some(next);

        if let Some(&(line, col, _)) = self.search_matches.get(next) {
            self.textarea.move_cursor(tui_textarea::CursorMove::Jump(line as u16, col as u16));
        }
    }
}
```
