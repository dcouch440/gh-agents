```rust
// src/tui/commands/file_commands.rs

use crate::execution::files::FileOps;
use crate::tui::views::{FileViewer, AppView};
use std::path::PathBuf;

pub struct ViewCommand;

impl ViewCommand {
    pub async fn execute(
        args: &str,
        file_ops: &FileOps,
        project_root: &PathBuf,
    ) -> Result<AppView, CommandError> {
        let path_str = args.trim();
        if path_str.is_empty() {
            return Err(CommandError::MissingArgument("path".to_string()));
        }

        let path = Self::resolve_path(path_str, project_root);

        let content = file_ops.read_file(&path)
            .await
            .map_err(|e| CommandError::FileError(e.to_string()))?;

        let viewer = FileViewer::new(path, content);
        Ok(AppView::FileViewer(viewer))
    }

    fn resolve_path(path_str: &str, project_root: &PathBuf) -> PathBuf {
        let path = PathBuf::from(path_str);
        if path.is_absolute() {
            path
        } else {
            project_root.join(path)
        }
    }
}

#[derive(Debug, thiserror::Error)]
pub enum CommandError {
    #[error("missing argument: {0}")]
    MissingArgument(String),
    #[error("file error: {0}")]
    FileError(String),
    #[error("invalid command: {0}")]
    InvalidCommand(String),
}

// In commands/mod.rs, add to router:
pub async fn handle_command(
    input: &str,
    ctx: &mut CommandContext,
) -> Result<Option<AppView>, CommandError> {
    let parts: Vec<&str> = input.splitn(2, ' ').collect();
    let command = parts[0];
    let args = parts.get(1).unwrap_or(&"");

    match command {
        "/view" => {
            let view = ViewCommand::execute(args, &ctx.file_ops, &ctx.project_root).await?;
            Ok(Some(view))
        }
        // ... other commands
        _ => Ok(None),
    }
}
```

```rust
pub struct EditCommand;

impl EditCommand {
    pub async fn execute(
        args: &str,
        file_ops: &FileOps,
        project_root: &PathBuf,
    ) -> Result<AppView, CommandError> {
        let path_str = args.trim();
        if path_str.is_empty() {
            return Err(CommandError::MissingArgument("path".to_string()));
        }

        let path = Self::resolve_path(path_str, project_root);

        let content = match file_ops.read_file(&path).await {
            Ok(content) => content,
            Err(e) if e.to_string().contains("not found") => {
                // New file - start with empty content
                tracing::info!(path = %path.display(), "Creating new file");
                String::new()
            }
            Err(e) => return Err(CommandError::FileError(e.to_string())),
        };

        let editor = FileEditor::new(path, content);
        Ok(AppView::FileEditor(editor))
    }

    fn resolve_path(path_str: &str, project_root: &PathBuf) -> PathBuf {
        let path = PathBuf::from(path_str);
        if path.is_absolute() {
            path
        } else {
            project_root.join(path)
        }
    }
}

// Add to router:
"/edit" => {
    let view = EditCommand::execute(args, &ctx.file_ops, &ctx.project_root).await?;
    Ok(Some(view))
}
```

```rust
pub struct FilesCommand;

impl FilesCommand {
    pub fn execute(
        args: &str,
        project_root: &PathBuf,
    ) -> Result<AppView, CommandError> {
        let path = if args.trim().is_empty() {
            project_root.clone()
        } else {
            let path = PathBuf::from(args.trim());
            if path.is_absolute() {
                path
            } else {
                project_root.join(path)
            }
        };

        if !path.is_dir() {
            return Err(CommandError::NotADirectory(path.display().to_string()));
        }

        let browser = FileBrowser::new(path)
            .map_err(|e| CommandError::IoError(e.to_string()))?;

        Ok(AppView::FileBrowser(browser))
    }
}

// Add to router:
"/files" => {
    let view = FilesCommand::execute(args, &ctx.project_root)?;
    Ok(Some(view))
}
```

### Slice 10.6.5: Add File Quick-Open from Agent Task Context

**Do this**:
- In task view, show file actions next to context_files
- Allow clicking/selecting to open in viewer/editor
- Add keyboard shortcuts in task view

**Create/modify**:
- `src/tui/views/task_view.rs`

**Code to add**:
```rust
pub struct TaskView {
    task: Task,
    selected_file_idx: Option<usize>,
}

impl TaskView {
    pub fn render_context_files(&self, frame: &mut Frame, area: Rect) {
        let files: Vec<ListItem> = self.task.context_files
            .iter()
            .enumerate()
            .map(|(idx, path)| {
                let style = if Some(idx) == self.selected_file_idx {
                    Style::default().bg(Color::Blue).fg(Color::White)
                } else {
                    Style::default()
                };

                ListItem::new(Line::from(vec![
                    Span::raw("📄 "),
                    Span::styled(path.display().to_string(), style),
                    Span::raw(" "),
                    Span::styled("[v]iew", Style::default().fg(Color::DarkGray)),
                    Span::raw(" "),
                    Span::styled("[e]dit", Style::default().fg(Color::DarkGray)),
                    Span::raw(" "),
                    Span::styled("[d]iff", Style::default().fg(Color::DarkGray)),
                ]))
            })
            .collect();

        let list = List::new(files)
            .block(Block::default().title("Context Files").borders(Borders::ALL));

        frame.render_widget(list, area);
    }

    pub fn handle_input(&mut self, input: Input) -> TaskViewAction {
        match input.key {
            Key::Up => {
                if let Some(idx) = self.selected_file_idx {
                    self.selected_file_idx = Some(idx.saturating_sub(1));
                } else if !self.task.context_files.is_empty() {
                    self.selected_file_idx = Some(0);
                }
                TaskViewAction::Continue
            }
            Key::Down => {
                if let Some(idx) = self.selected_file_idx {
                    let max = self.task.context_files.len().saturating_sub(1);
                    self.selected_file_idx = Some((idx + 1).min(max));
                } else if !self.task.context_files.is_empty() {
                    self.selected_file_idx = Some(0);
                }
                TaskViewAction::Continue
            }
            Key::Char('v') => {
                if let Some(idx) = self.selected_file_idx {
                    if let Some(path) = self.task.context_files.get(idx) {
                        return TaskViewAction::ViewFile(path.clone());
                    }
                }
                TaskViewAction::Continue
            }
            Key::Char('e') => {
                if let Some(idx) = self.selected_file_idx {
                    if let Some(path) = self.task.context_files.get(idx) {
                        return TaskViewAction::EditFile(path.clone());
                    }
                }
                TaskViewAction::Continue
            }
            Key::Char('d') => {
                if let Some(idx) = self.selected_file_idx {
                    if let Some(path) = self.task.context_files.get(idx) {
                        return TaskViewAction::DiffFile(path.clone());
                    }
                }
                TaskViewAction::Continue
            }
            _ => TaskViewAction::Continue,
        }
    }
}

pub enum TaskViewAction {
    Continue,
    ViewFile(PathBuf),
    EditFile(PathBuf),
    DiffFile(PathBuf),
}
```
