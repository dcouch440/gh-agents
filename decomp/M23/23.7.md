### Slice 23.7.2: API Client Methods

**Modify `cli/src/api/client.ts`**:

```typescript
// Add to existing api object or create gate-specific functions:

export async function listGates(status?: GateStatus): Promise<Gate[]> {
  const query = status ? `?status=${status}` : '';
  return fetchApi<Gate[]>(`/gates${query}`);
}

export async function getGate(id: string): Promise<GateDetail> {
  return fetchApi<GateDetail>(`/gates/${id}`);
}

export async function getGateMessages(gateId: string): Promise<GateMessage[]> {
  return fetchApi<GateMessage[]>(`/gates/${gateId}/messages`);
}

export async function sendGateMessage(gateId: string, content: string): Promise<GateMessage> {
  return fetchApi<GateMessage>(`/gates/${gateId}/messages`, {
    method: 'POST',
    body: JSON.stringify({ content }),
  });
}

export async function updateGateStatus(gateId: string, status: 'resolved' | 'skipped'): Promise<Gate> {
  return fetchApi<Gate>(`/gates/${gateId}/status`, {
    method: 'PATCH',
    body: JSON.stringify({ status }),
  });
}

export async function getProjectTree(): Promise<TreeNode[]> {
  return fetchApi<TreeNode[]>('/tree');
}
```
