# Ticket 11.5: Export & Reports

> Export data for external analysis

## Goal

// src/export/csv.rs
use crate::db::analytics::Analytics;
use chrono::{DateTime, Utc};
use std::io::Write;
use std::path::Path;

pub struct CsvExporter<'a> {
    analytics: &'a Analytics,
}

impl<'a> CsvExporter<'a> {
    pub fn new(analytics: &'a Analytics) -> Self {
        Self { analytics }
    }

    pub async fn export_costs(
        &self,
        output: &Path,
        since: Option<DateTime<Utc>>,
    ) -> Result<ExportResult, ExportError> {
        let since_str = since
            .map(|d| d.to_rfc3339())
            .unwrap_or_else(|| "1970-01-01T00:00:00Z".to_string());

        // Query raw cost records
        let records = sqlx::query!(
            r#"
            SELECT
                id, task_id, agent_id, agent_tier, model_id,
                input_tokens, output_tokens, cost_usd, timestamp
            FROM cost_records
            WHERE timestamp >= ?
            ORDER BY timestamp ASC
            "#,
            since_str
        )
        .fetch_all(&self.analytics.pool())
        .await
        .map_err(ExportError::Database)?;

        let mut file = std::fs::File::create(output)
            .map_err(ExportError::Io)?;

        // Write header
        writeln!(
            file,
            "timestamp,task_id,agent_id,tier,model,input_tokens,output_tokens,cost_usd"
        ).map_err(ExportError::Io)?;

        // Write rows
        for record in &records {
            writeln!(
                file,
                "{},{},{},{},{},{},{},{:.6}",
                record.timestamp,
                record.task_id.as_deref().unwrap_or(""),
                record.agent_id,
                record.agent_tier,
                record.model_id,
                record.input_tokens,
                record.output_tokens,
                record.cost_usd,
            ).map_err(ExportError::Io)?;
        }

        Ok(ExportResult {
            path: output.to_path_buf(),
            rows: records.len(),
            format: ExportFormat::Csv,
        })
    }

    pub async fn export_tasks(
        &self,
        output: &Path,
        since: Option<DateTime<Utc>>,
    ) -> Result<ExportResult, ExportError> {
        let since_str = since
            .map(|d| d.to_rfc3339())
            .unwrap_or_else(|| "1970-01-01T00:00:00Z".to_string());

        let tasks = sqlx::query!(
            r#"
            SELECT
                id, title, status, assigned_tier, assigned_agent,
                priority, created_at, updated_at
            FROM tasks
            WHERE created_at >= ?
            ORDER BY created_at ASC
            "#,
            since_str
        )
        .fetch_all(&self.analytics.pool())
        .await
        .map_err(ExportError::Database)?;

        let mut file = std::fs::File::create(output)
            .map_err(ExportError::Io)?;

        writeln!(
            file,
            "id,title,status,tier,agent,priority,created_at,updated_at"
        ).map_err(ExportError::Io)?;

        for task in &tasks {
            // Escape title for CSV
            let escaped_title = task.title.replace('"', "\"\"");
            writeln!(
                file,
                "{},\"{}\",{},{},{},{},{},{}",
                task.id,
                escaped_title,
                task.status,
                task.assigned_tier,
                task.assigned_agent.as_deref().unwrap_or(""),
                task.priority,
                task.created_at,
                task.updated_at,
            ).map_err(ExportError::Io)?;
        }

        Ok(ExportResult {
            path: output.to_path_buf(),
            rows: tasks.len(),
            format: ExportFormat::Csv,
        })
    }

    pub async fn export_sessions(
        &self,
        output: &Path,
        since: Option<DateTime<Utc>>,
    ) -> Result<ExportResult, ExportError> {
        let since_str = since
            .map(|d| d.to_rfc3339())
            .unwrap_or_else(|| "1970-01-01T00:00:00Z".to_string());

        let sessions = sqlx::query!(
            r#"
            SELECT
                id, started_at, ended_at, tasks_completed, tasks_failed, total_cost
            FROM sessions
            WHERE started_at >= ?
            ORDER BY started_at ASC
            "#,
            since_str
        )
        .fetch_all(&self.analytics.pool())
        .await
        .map_err(ExportError::Database)?;

        let mut file = std::fs::File::create(output)
            .map_err(ExportError::Io)?;

        writeln!(
            file,
            "id,started_at,ended_at,tasks_completed,tasks_failed,total_cost"
        ).map_err(ExportError::Io)?;

        for session in &sessions {
            writeln!(
                file,
                "{},{},{},{},{},{:.4}",
                session.id,
                session.started_at,
                session.ended_at.as_deref().unwrap_or(""),
                session.tasks_completed,
                session.tasks_failed,
                session.total_cost,
            ).map_err(ExportError::Io)?;
        }

        Ok(ExportResult {
            path: output.to_path_buf(),
            rows: sessions.len(),
            format: ExportFormat::Csv,
        })
    }
}

#[derive(Debug)]
pub struct ExportResult {
    pub path: PathBuf,
    pub rows: usize,
    pub format: ExportFormat,
}

#[derive(Debug, thiserror::Error)]
pub enum ExportError {
    #[error("database error: {0}")]
    Database(#[from] sqlx::Error),
    #[error("io error: {0}")]
    Io(#[from] std::io::Error),
}

// src/export/json.rs

use crate::db::analytics::{Analytics, CostSummary, TaskMetrics, CostByTier, CostByModel};
use crate::db::sessions::{Session, SessionStats};
use chrono::{DateTime, Utc};
use serde::Serialize;
use std::path::Path;

pub struct JsonExporter<'a> {
    analytics: &'a Analytics,
}

#[derive(Debug, Serialize)]
pub struct AnalyticsSnapshot {
    pub exported_at: DateTime<Utc>,
    pub period: PeriodInfo,
    pub summary: SummarySection,
    pub costs: CostsSection,
    pub tasks: TasksSection,
    pub sessions: SessionsSection,
}

#[derive(Debug, Serialize)]
pub struct PeriodInfo {
    pub from: Option<DateTime<Utc>>,
    pub to: DateTime<Utc>,
}

#[derive(Debug, Serialize)]
pub struct SummarySection {
    pub total_cost_usd: f64,
    pub total_tokens: i64,
    pub total_requests: i64,
    pub tasks_completed: i64,
    pub tasks_failed: i64,
    pub success_rate: f64,
}

#[derive(Debug, Serialize)]
pub struct CostsSection {
    pub by_tier: Vec<CostByTier>,
    pub by_model: Vec<CostByModel>,
}

#[derive(Debug, Serialize)]
pub struct TasksSection {
    pub total: i64,
    pub by_status: std::collections::HashMap<String, i64>,
}

#[derive(Debug, Serialize)]
pub struct SessionsSection {
    pub total: i64,
    pub avg_duration_mins: f64,
    pub recent: Vec<SessionSummary>,
}

#[derive(Debug, Serialize)]
pub struct SessionSummary {
    pub id: String,
    pub started_at: DateTime<Utc>,
    pub duration_mins: Option<i64>,
    pub tasks_completed: i32,
    pub cost_usd: f64,
}

impl<'a> JsonExporter<'a> {
    pub fn new(analytics: &'a Analytics) -> Self {
        Self { analytics }
    }

    pub async fn export_snapshot(
        &self,
        output: &Path,
        since: Option<DateTime<Utc>>,
        pretty: bool,
    ) -> Result<ExportResult, ExportError> {
        // Gather all data
        let cost_summary = self.analytics.cost_summary(since).await?;
        let task_metrics = self.analytics.task_metrics(since).await?;
        let by_tier = self.analytics.costs_by_tier(since).await?;
        let by_model = self.analytics.costs_by_model(since).await?;
        let session_stats = self.analytics.session_stats(since).await?;
        let recent_sessions = self.analytics.recent_sessions(10).await?;

        let snapshot = AnalyticsSnapshot {
            exported_at: Utc::now(),
            period: PeriodInfo {
                from: since,
                to: Utc::now(),
            },
            summary: SummarySection {
                total_cost_usd: cost_summary.total_usd,
                total_tokens: cost_summary.input_tokens + cost_summary.output_tokens,
                total_requests: cost_summary.request_count,
                tasks_completed: task_metrics.completed,
                tasks_failed: task_metrics.failed,
                success_rate: task_metrics.success_rate,
            },
            costs: CostsSection {
                by_tier,
                by_model,
            },
            tasks: TasksSection {
                total: task_metrics.total,
                by_status: [
                    ("pending".to_string(), task_metrics.pending),
                    ("in_progress".to_string(), task_metrics.in_progress),
                    ("completed".to_string(), task_metrics.completed),
                    ("failed".to_string(), task_metrics.failed),
                ].into_iter().collect(),
            },
            sessions: SessionsSection {
                total: session_stats.session_count,
                avg_duration_mins: session_stats.avg_duration_mins,
                recent: recent_sessions.into_iter().map(|s| SessionSummary {
                    id: s.id,
                    started_at: s.started_at,
                    duration_mins: s.duration().map(|d| d.num_minutes()),
                    tasks_completed: s.tasks_completed,
                    cost_usd: s.total_cost,
                }).collect(),
            },
        };

        let json = if pretty {
            serde_json::to_string_pretty(&snapshot)?
        } else {
            serde_json::to_string(&snapshot)?
        };

        std::fs::write(output, json)?;

        Ok(ExportResult {
            path: output.to_path_buf(),
            rows: 1, // Single snapshot
            format: ExportFormat::Json,
        })
    }
}

### Slice 11.5.3: Add /export Command with Format Options

**Do this**:

**Create/modify**:
- `src/tui/commands/export.rs`
- `src/tui/commands/mod.rs`

**Code**:
```rust
// src/tui/commands/export.rs

use crate::export::{ExportFormat, ExportDataType, ExportOptions};
use crate::export::csv::CsvExporter;
use crate::export::json::JsonExporter;
use chrono::{Duration, Utc};
use std::path::PathBuf;

pub struct ExportCommand;

impl ExportCommand {
    pub fn parse(args: &str) -> Result<ExportOptions, CommandError> {
        let parts: Vec<&str> = args.split_whitespace().collect();

        let mut options = ExportOptions {
            format: ExportFormat::Csv,
            output_path: None,
            since: None,
            data_type: ExportDataType::Costs,
        };

        let mut i = 0;
        while i < parts.len() {
            match parts[i] {
                "costs" => options.data_type = ExportDataType::Costs,
                "tasks" => options.data_type = ExportDataType::Tasks,
                "sessions" => options.data_type = ExportDataType::Sessions,
                "all" => options.data_type = ExportDataType::All,
                "--format" | "-f" => {
                    i += 1;
                    if i < parts.len() {
                        options.format = match parts[i] {
                            "csv" => ExportFormat::Csv,
                            "json" => ExportFormat::Json,
                            _ => return Err(CommandError::InvalidFormat(parts[i].to_string())),
                        };
                    }
                }
                "--period" | "-p" => {
                    i += 1;
                    if i < parts.len() {
                        options.since = match parts[i] {
                            "today" => Some(Utc::now() - Duration::days(1)),
                            "week" => Some(Utc::now() - Duration::days(7)),
                            "month" => Some(Utc::now() - Duration::days(30)),
                            "all" => None,
                            _ => return Err(CommandError::InvalidPeriod(parts[i].to_string())),
                        };
                    }
                }
                "--output" | "-o" => {
                    i += 1;
                    if i < parts.len() {
                        options.output_path = Some(PathBuf::from(parts[i]));
                    }
                }
                _ => {}
            }
            i += 1;
        }

        Ok(options)
    }

    pub async fn execute(
        options: ExportOptions,
        analytics: &Analytics,
    ) -> Result<ExportResult, ExportError> {
        // Create exports directory if needed
        let exports_dir = PathBuf::from(".nexor/exports");
        std::fs::create_dir_all(&exports_dir)?;

        let timestamp = Utc::now().format("%Y%m%d_%H%M%S");
        let extension = match options.format {
            ExportFormat::Csv => "csv",
            ExportFormat::Json => "json",
        };

        let output_path = options.output_path.unwrap_or_else(|| {
            let filename = match options.data_type {
                ExportDataType::Costs => format!("costs_{}.{}", timestamp, extension),
                ExportDataType::Tasks => format!("tasks_{}.{}", timestamp, extension),
                ExportDataType::Sessions => format!("sessions_{}.{}", timestamp, extension),
                ExportDataType::All => format!("analytics_{}.{}", timestamp, extension),
            };
            exports_dir.join(filename)
        });

        match (options.format, options.data_type) {
            (ExportFormat::Csv, ExportDataType::Costs) => {
                CsvExporter::new(analytics).export_costs(&output_path, options.since).await
            }
            (ExportFormat::Csv, ExportDataType::Tasks) => {
                CsvExporter::new(analytics).export_tasks(&output_path, options.since).await
            }
            (ExportFormat::Csv, ExportDataType::Sessions) => {
                CsvExporter::new(analytics).export_sessions(&output_path, options.since).await
            }
            (ExportFormat::Json, _) => {
                JsonExporter::new(analytics).export_snapshot(&output_path, options.since, true).await
            }
            _ => Err(ExportError::UnsupportedCombination),
        }
    }
}

// Add to command router
"/export" => {
    let options = ExportCommand::parse(args)?;
    let result = ExportCommand::execute(options, &ctx.analytics).await?;
    ctx.notifications.push(Notification::success(
        format!("Exported {} rows to {}", result.rows, result.path.display())
    ));
    Ok(None)
}
```
