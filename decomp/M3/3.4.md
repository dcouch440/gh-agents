> Configurable agent personalities.

## Goal

use crate::types::{AgentTier, AgentPersona, CommunicationStyle};

/// Default system prompt for Orchestrator agents
const DEFAULT_ORCHESTRATOR_PROMPT: &str = r#"You are a senior software architect working with a team of AI agents.
Your role is to:
- Break down complex problems into vertical slices
- Each slice should be independently deployable and valuable
- Explain your reasoning and ask clarifying questions
- Review work from other agents before approval
- Make architectural decisions and resolve conflicts

Communicate in a collaborative, thoughtful manner. Brainstorm with
the user when needed. Always consider the bigger picture."#;

/// Default system prompt for Worker agents
const DEFAULT_WORKER_PROMPT: &str = r#"You are a focused software developer. Your role is to:
- Implement code changes as specified in your task
- Write clean, well-tested code
- Report progress naturally as you work
- Ask for more context only when truly needed
- Submit work for review when complete

Stay heads-down on your task. Keep updates brief but informative."#;

/// Default system prompt for Utility agents
const DEFAULT_UTILITY_PROMPT: &str = r#"You handle quick, well-defined tasks:
- Format code according to project style
- Run linters and fix issues
- Generate boilerplate from templates
- Update documentation

Be brief. Report only completions and errors.
Example: "Formatted 3 files" or "Lint error in src/main.rs:42""#;

/// Manages agent personas
pub struct PersonaManager {
    /// Default personas by tier
    defaults: std::collections::HashMap<AgentTier, AgentPersona>,
    /// Project-specific overrides
    overrides: std::collections::HashMap<AgentTier, AgentPersona>,
}

impl PersonaManager {
    /// Create a new persona manager with default personas
    pub fn new() -> Self {
        let mut defaults = std::collections::HashMap::new();

        defaults.insert(
            AgentTier::Orchestrator,
            AgentPersona {
                name: "Arch".to_string(),
                system_prompt: DEFAULT_ORCHESTRATOR_PROMPT.to_string(),
                style: CommunicationStyle::Formal,
            },
        );

        defaults.insert(
            AgentTier::Worker,
            AgentPersona {
                name: "Dev".to_string(),
                system_prompt: DEFAULT_WORKER_PROMPT.to_string(),
                style: CommunicationStyle::Technical,
            },
        );

        defaults.insert(
            AgentTier::Utility,
            AgentPersona {
                name: "Helper".to_string(),
                system_prompt: DEFAULT_UTILITY_PROMPT.to_string(),
                style: CommunicationStyle::Casual,
            },
        );

        Self {
            defaults,
            overrides: std::collections::HashMap::new(),
        }
    }

    /// Get the persona for a tier (with any overrides applied)
    pub fn get_persona(&self, tier: AgentTier) -> AgentPersona {
        self.overrides
            .get(&tier)
            .or_else(|| self.defaults.get(&tier))
            .cloned()
            .unwrap_or_else(|| self.default_persona(tier))
    }

    /// Get the default persona for a tier
    fn default_persona(&self, tier: AgentTier) -> AgentPersona {
        // Fallback if somehow not in defaults map
        AgentPersona {
            name: format!("{:?}", tier),
            system_prompt: "You are an AI assistant.".to_string(),
            style: CommunicationStyle::Formal,
        }
    }

    /// Get all default personas
    pub fn defaults(&self) -> &std::collections::HashMap<AgentTier, AgentPersona> {
        &self.defaults
    }
}

impl Default for PersonaManager {
    fn default() -> Self {
        Self::new()
    }
}

#[cfg(test)]
mod tests {

        assert_eq!(worker.name, "Dev");

use crate::types::ProjectConfig;

impl PersonaManager {
    /// Load persona overrides from project configuration
    pub fn load_project_overrides(&mut self, config: &ProjectConfig) {
        // Check for orchestrator override
        if let Some(ref persona_config) = config.personas {
            if let Some(ref orch) = persona_config.orchestrator {
                let default = self.defaults.get(&AgentTier::Orchestrator)
                    .cloned()
                    .unwrap_or_else(|| self.default_persona(AgentTier::Orchestrator));

                self.overrides.insert(
                    AgentTier::Orchestrator,
                    merge_persona_config(default, orch),
                );
            }

            if let Some(ref worker) = persona_config.worker {
                let default = self.defaults.get(&AgentTier::Worker)
                    .cloned()
                    .unwrap_or_else(|| self.default_persona(AgentTier::Worker));

                self.overrides.insert(
                    AgentTier::Worker,
                    merge_persona_config(default, worker),
                );
            }

            if let Some(ref utility) = persona_config.utility {
                let default = self.defaults.get(&AgentTier::Utility)
                    .cloned()
                    .unwrap_or_else(|| self.default_persona(AgentTier::Utility));

                self.overrides.insert(
                    AgentTier::Utility,
                    merge_persona_config(default, utility),
                );
            }
        }
    }

    /// Clear all overrides (return to defaults)
    pub fn clear_overrides(&mut self) {
        self.overrides.clear();
    }

    /// Check if a tier has an override
    pub fn has_override(&self, tier: AgentTier) -> bool {
        self.overrides.contains_key(&tier)
    }
}

/// Persona configuration from project config (all fields optional)
#[derive(Debug, Clone, serde::Deserialize)]
pub struct PersonaConfig {
    pub name: Option<String>,
    pub system_prompt: Option<String>,
    pub style: Option<CommunicationStyle>,
}

/// Merge a partial config into a default persona
fn merge_persona_config(mut default: AgentPersona, config: &PersonaConfig) -> AgentPersona {
    if let Some(ref name) = config.name {
        default.name = name.clone();
    }
    if let Some(ref prompt) = config.system_prompt {
        default.system_prompt = prompt.clone();
    }
    if let Some(style) = config.style {
        default.style = style;
    }
    default
}

// Add to src/agents/persona.rs

/// Context for building a system prompt
#[derive(Debug, Clone)]
pub struct PromptContext {
    /// Task description
    pub task_description: Option<String>,
    /// Project conventions (e.g., from CLAUDE.md)
    pub conventions: Option<String>,
    /// Additional context to include
    pub additional_context: Option<String>,
}

impl Default for PromptContext {
    fn default() -> Self {
        Self {
            task_description: None,
            conventions: None,
            additional_context: None,
        }
    }
}

impl PersonaManager {
    /// Build a complete system prompt for an agent
    ///
    /// Combines the persona's base prompt with task context and conventions.
    pub fn build_system_prompt(&self, tier: AgentTier, context: &PromptContext) -> String {
        let persona = self.get_persona(tier);
        let mut parts = Vec::new();

        // Base persona prompt
        parts.push(format!("## Your Role\n\n{}", persona.system_prompt));

        // Project conventions
        if let Some(ref conventions) = context.conventions {
            if !conventions.is_empty() {
                parts.push(format!("## Project Conventions\n\n{}", conventions));
            }
        }

        // Task-specific context
        if let Some(ref task) = context.task_description {
            if !task.is_empty() {
                parts.push(format!("## Current Task\n\n{}", task));
            }
        }

        // Additional context
        if let Some(ref additional) = context.additional_context {
            if !additional.is_empty() {
                parts.push(format!("## Additional Context\n\n{}", additional));
            }
        }

        // Communication style guidance
        let style_guidance = match persona.style {
            CommunicationStyle::Formal => "Communicate professionally and precisely.",
            CommunicationStyle::Casual => "Keep responses brief and conversational.",
            CommunicationStyle::Technical => "Focus on technical accuracy and code.",
            CommunicationStyle::Friendly => "Be helpful and approachable.",
        };
        parts.push(format!("## Communication Style\n\n{}", style_guidance));

        parts.join("\n\n---\n\n")
    }

    /// Build a minimal prompt (persona only, no context)
    pub fn build_base_prompt(&self, tier: AgentTier) -> String {
        let persona = self.get_persona(tier);
        persona.system_prompt.clone()
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn build_system_prompt_includes_all_parts() {
        let manager = PersonaManager::new();

        let context = PromptContext {
            task_description: Some("Implement a new feature".to_string()),
            conventions: Some("Use snake_case for functions".to_string()),
            additional_context: None,
        };

        let prompt = manager.build_system_prompt(AgentTier::Worker, &context);

        assert!(prompt.contains("## Your Role"));
        assert!(prompt.contains("focused software developer")); // From default worker prompt
        assert!(prompt.contains("## Project Conventions"));
        assert!(prompt.contains("snake_case"));
        assert!(prompt.contains("## Current Task"));
        assert!(prompt.contains("Implement a new feature"));
        assert!(prompt.contains("## Communication Style"));
    }

    #[test]
    fn build_prompt_handles_empty_context() {
        let manager = PersonaManager::new();
        let context = PromptContext::default();

        let prompt = manager.build_system_prompt(AgentTier::Orchestrator, &context);

        assert!(prompt.contains("## Your Role"));
        // Should not contain empty sections
        assert!(!prompt.contains("## Project Conventions\n\n\n"));
    }
}
