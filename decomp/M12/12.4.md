### Slice 12.4.1: Create AgentsPage

**Do this**:
- Create AgentsPage component
- Group agents by tier
- Use CSS modules for styling
- Add proper useEffect cleanup for WebSocket

**Create `ui/src/pages/AgentsPage/AgentsPage.tsx`**:
```tsx
import { useEffect, useState } from 'react';
import { api, Agent } from '../../api/client';
import { AgentCard } from '../../components/AgentCard';
import { wsClient } from '../../api/websocket';
import styles from './AgentsPage.module.css';

export function AgentsPage() {
  const [agents, setAgents] = useState<Agent[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    let mounted = true;

    api.agents.list().then((data) => {
      if (mounted) {
        setAgents(data);
        setLoading(false);
      }
    });

    return () => {
      mounted = false;
    };
  }, []);

  useEffect(() => {
    let mounted = true;

    wsClient.subscribe(['agents']);

    const handleUpdate = (data: Agent) => {
      if (mounted) {
        setAgents((prev) =>
          prev.map((a) => (a.id === data.id ? data : a))
        );
      }
    };

    wsClient.on('agent_update', handleUpdate);

    return () => {
      mounted = false;
      wsClient.off('agent_update', handleUpdate);
      wsClient.unsubscribe(['agents']);
    };
  }, []);

  // Group by tier
  const byTier = agents.reduce((acc, agent) => {
    const tier = agent.tier;
    if (!acc[tier]) acc[tier] = [];
    acc[tier].push(agent);
    return acc;
  }, {} as Record<string, Agent[]>);

  if (loading) {
    return <div className={styles.loading}>Loading...</div>;
  }

  return (
    <div className={styles.container}>
      <h2 className={styles.title}>Agent Pool</h2>

      <div className={styles.tierGroups}>
        {Object.entries(byTier).map(([tier, tierAgents]) => (
          <div key={tier} className={styles.tierGroup}>
            <h3 className={styles.tierTitle}>
              {tier}s ({tierAgents.filter((a) => a.status === 'working').length}/
              {tierAgents.length} active)
            </h3>
            <div className={styles.agentGrid}>
              {tierAgents.map((agent) => (
                <AgentCard key={agent.id} agent={agent} />
              ))}
            </div>
          </div>
        ))}
      </div>
    </div>
  );
}
```

### Slice 12.4.2: Create AgentCard Component

**Do this**:
- Create AgentCard component
- Show agent status and current task
- Use CSS modules
- Add loading/working state indicators

**Create `ui/src/components/AgentCard/AgentCard.tsx`**:
```tsx
import { Agent } from '../../api/client';
import { StatusDot } from '../StatusDot';
import { Bot, Loader } from 'lucide-react';
import styles from './AgentCard.module.css';

interface AgentCardProps {
  agent: Agent;
}

export function AgentCard({ agent }: AgentCardProps) {
  const isWorking = agent.status === 'working';

  return (
    <div className={styles.card}>
      <div className={styles.header}>
        <div className={`${styles.iconWrapper} ${isWorking ? styles.iconWorking : ''}`}>
          {isWorking ? (
            <Loader size={20} className={styles.spinner} />
          ) : (
            <Bot size={20} className={styles.icon} />
          )}
        </div>
        <div className={styles.info}>
          <div className={styles.nameRow}>
            <span className={styles.name}>{agent.id}</span>
            <StatusDot
              status={isWorking ? 'active' : 'idle'}
              pulse={isWorking}
            />
          </div>
          <span className={styles.tier}>{agent.tier}</span>
        </div>
      </div>
      {agent.current_task && (
        <div className={styles.task}>
          <span className={styles.taskLabel}>Working on:</span>
          <p className={styles.taskName}>{agent.current_task}</p>
        </div>
      )}
    </div>
  );
}
```

```tsx
import { AgentCard } from '../../components/AgentCard';
import { wsClient } from '../../api/websocket';
import styles from './AgentsPage.module.css';

export function AgentsPage() {
  const [agents, setAgents] = useState<Agent[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    let mounted = true;

    api.agents.list().then((data) => {
      if (mounted) {
        setAgents(data);
        setLoading(false);
      }
    });

    return () => {
      mounted = false;
    };
  }, []);

  useEffect(() => {
    let mounted = true;

    wsClient.subscribe(['agents']);

    const handleUpdate = (data: Agent) => {
      if (mounted) {
        setAgents((prev) =>
          prev.map((a) => (a.id === data.id ? data : a))
```
