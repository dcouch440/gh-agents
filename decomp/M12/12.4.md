## Slices

### Slice 12.4.1: Create AgentsPage

**Create `ui/src/pages/AgentsPage.tsx`**:
```tsx
import { useEffect, useState } from 'react';
import { api, Agent } from '../api/client';
import { AgentCard } from '../components/AgentCard';
import { wsClient } from '../api/websocket';

export function AgentsPage() {
  const [agents, setAgents] = useState<Agent[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    api.agents.list().then((data) => {
      setAgents(data);
      setLoading(false);
    });
  }, []);

  useEffect(() => {
    wsClient.subscribe(['agents']);

    const handleUpdate = (data: Agent) => {
      setAgents((prev) =>
        prev.map((a) => (a.id === data.id ? data : a))
      );
    };

    wsClient.on('agent_update', handleUpdate);

    return () => {
      wsClient.off('agent_update', handleUpdate);
      wsClient.unsubscribe(['agents']);
    };
  }, []);

  // Group by tier
  const byTier = agents.reduce((acc, agent) => {
    const tier = agent.tier;
    if (!acc[tier]) acc[tier] = [];
    acc[tier].push(agent);
    return acc;
  }, {} as Record<string, Agent[]>);

  return (
    <div>
      <h2 className="text-lg font-medium text-text-primary mb-4">Agent Pool</h2>

      {loading ? (
        <div className="text-text-secondary">Loading...</div>
      ) : (
        <div className="space-y-6">
          {Object.entries(byTier).map(([tier, tierAgents]) => (
            <div key={tier}>
              <h3 className="text-sm font-medium text-text-secondary mb-2 capitalize">
                {tier}s ({tierAgents.filter(a => a.status === 'working').length}/{tierAgents.length} active)
              </h3>
              <div className="grid gap-3 md:grid-cols-2 lg:grid-cols-3">
                {tierAgents.map((agent) => (
                  <AgentCard key={agent.id} agent={agent} />
                ))}
              </div>
            </div>
          ))}
        </div>
      )}
    </div>
  );
}
```

### Slice 12.4.2: Create AgentCard Component

**Create `ui/src/components/AgentCard.tsx`**:
```tsx
import { Agent } from '../api/client';
import { StatusDot } from './StatusDot';
import { Bot, Loader } from 'lucide-react';

interface AgentCardProps {
  agent: Agent;
}

export function AgentCard({ agent }: AgentCardProps) {
  const isWorking = agent.status === 'working';

  return (
    <div className="p-4 bg-bg-secondary rounded-lg border border-border">
      <div className="flex items-center gap-3">
        <div className={`p-2 rounded-full ${isWorking ? 'bg-accent-secondary/20' : 'bg-bg-tertiary'}`}>
          {isWorking ? (
            <Loader size={20} className="animate-spin text-accent-secondary" />
          ) : (
            <Bot size={20} className="text-text-tertiary" />
          )}
        </div>
        <div>
          <div className="flex items-center gap-2">
            <span className="font-medium text-text-primary">{agent.id}</span>
            <StatusDot
              status={isWorking ? 'active' : 'idle'}
              pulse={isWorking}
            />
          </div>
          <span className="text-sm text-text-secondary capitalize">{agent.tier}</span>
        </div>
      </div>
      {agent.current_task && (
        <div className="mt-3 pt-3 border-t border-border">
          <span className="text-xs text-text-tertiary">Working on:</span>
          <p className="text-sm text-text-secondary truncate">{agent.current_task}</p>
        </div>
      )}
    </div>
  );
}
```
