### Slice 12.3.1: Create TasksPage with Task List

**Do this**:
- Create TasksPage component
- Display task list with filters
- Use CSS modules for styling
- Add proper useEffect cleanup for WebSocket

**Create `ui/src/pages/TasksPage/TasksPage.tsx`**:
```tsx
import { useEffect, useState } from 'react';
import { api, Task } from '../../api/client';
import { TaskCard } from '../../components/TaskCard';
import { wsClient } from '../../api/websocket';
import { Button } from '../../components/Button';
import styles from './TasksPage.module.css';

export function TasksPage() {
  const [tasks, setTasks] = useState<Task[]>([]);
  const [loading, setLoading] = useState(true);
  const [filter, setFilter] = useState<string | null>(null);

  useEffect(() => {
    let mounted = true;

    api.tasks.list({ status: filter || undefined }).then((data) => {
      if (mounted) {
        setTasks(data);
        setLoading(false);
      }
    });

    return () => {
      mounted = false;
    };
  }, [filter]);

  useEffect(() => {
    let mounted = true;

    wsClient.subscribe(['tasks']);

    const handleUpdate = (data: { id: string; status: string }) => {
      if (mounted) {
        setTasks((prev) =>
          prev.map((t) => (t.id === data.id ? { ...t, status: data.status } : t))
        );
      }
    };

    wsClient.on('task_update', handleUpdate);

    return () => {
      mounted = false;
      wsClient.off('task_update', handleUpdate);
      wsClient.unsubscribe(['tasks']);
    };
  }, []);

  const filteredTasks = tasks.filter((t) => !filter || t.status === filter);

  const filters = ['all', 'pending', 'in_progress', 'completed', 'failed'];

  return (
    <div className={styles.container}>
      {/* Filters */}
      <div className={styles.filters}>
        {filters.map((status) => (
          <Button
            key={status}
            onClick={() => setFilter(status === 'all' ? null : status)}
            variant={
              (filter === status || (status === 'all' && !filter))
                ? 'primary'
                : 'secondary'
            }
            size="sm"
          >
            {status.replace('_', ' ')}
          </Button>
        ))}
      </div>

      {/* Task list */}
      {loading ? (
        <div className={styles.loading}>Loading...</div>
      ) : filteredTasks.length === 0 ? (
        <div className={styles.emptyState}>No tasks found</div>
      ) : (
        <div className={styles.taskList}>
          {filteredTasks.map((task) => (
            <TaskCard key={task.id} task={task} />
          ))}
        </div>
      )}
    </div>
  );
}
```

### Slice 12.3.2: Create TaskCard Component

**Do this**:
- Create TaskCard component
- Show task details with status indicator
- Use CSS modules
- Support click handler

**Create `ui/src/components/TaskCard/TaskCard.tsx`**:
```tsx
import { Task } from '../../api/client';
import { StatusDot } from '../StatusDot';
import { Clock, User } from 'lucide-react';
import styles from './TaskCard.module.css';

interface TaskCardProps {
  task: Task;
  onClick?: () => void;
}

export function TaskCard({ task, onClick }: TaskCardProps) {
  const getStatusType = (status: string): 'success' | 'error' | 'active' | 'idle' => {
    switch (status) {
      case 'completed':
        return 'success';
      case 'failed':
        return 'error';
      case 'in_progress':
        return 'active';
      default:
        return 'idle';
    }
  };

  return (
    <div onClick={onClick} className={styles.card}>
      <div className={styles.content}>
        <div className={styles.main}>
          <div className={styles.statusRow}>
            <StatusDot
              status={getStatusType(task.status)}
              pulse={task.status === 'in_progress'}
            />
            <span className={styles.status}>
              {task.status.replace('_', ' ')}
            </span>
          </div>
          <h3 className={styles.title}>{task.title}</h3>
          <p className={styles.description}>
            {task.description}
          </p>
        </div>
        <div className={styles.meta}>
          <div className={styles.metaItem}>
            <Clock size={12} />
            {new Date(task.created_at).toLocaleDateString()}
          </div>
          {task.assigned_agent && (
            <div className={styles.metaItem}>
              <User size={12} />
              {task.assigned_agent}
            </div>
          )}
        </div>
      </div>
    </div>
  );
}
```

### Slice 12.3.5: Add StatusDot Utility Component

```tsx
import styles from './StatusDot.module.css';

interface StatusDotProps {
  status: 'success' | 'error' | 'active' | 'idle';
  pulse?: boolean;
}

export function StatusDot({ status, pulse = false }: StatusDotProps) {
  return (
    <div
      className={`${styles.dot} ${styles[status]} ${pulse ? styles.pulse : ''}`}
      aria-label={`Status: ${status}`}
    />
  );
}
```
