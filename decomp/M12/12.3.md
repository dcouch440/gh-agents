### Slice 12.3.1: Create TasksPage with Task List

**Create `ui/src/pages/TasksPage.tsx`**:
```tsx
import { useEffect, useState } from 'react';
import { api, Task } from '../api/client';
import { TaskCard } from '../components/TaskCard';
import { wsClient } from '../api/websocket';

export function TasksPage() {
  const [tasks, setTasks] = useState<Task[]>([]);
  const [loading, setLoading] = useState(true);
  const [filter, setFilter] = useState<string | null>(null);

  useEffect(() => {
    api.tasks.list({ status: filter || undefined }).then((data) => {
      setTasks(data);
      setLoading(false);
    });
  }, [filter]);

  useEffect(() => {
    wsClient.subscribe(['tasks']);

    const handleUpdate = (data: { id: string; status: string }) => {
      setTasks((prev) =>
        prev.map((t) => (t.id === data.id ? { ...t, status: data.status } : t))
      );
    };

    wsClient.on('task_update', handleUpdate);

    return () => {
      wsClient.off('task_update', handleUpdate);
      wsClient.unsubscribe(['tasks']);
    };
  }, []);

  const filteredTasks = tasks.filter((t) => !filter || t.status === filter);

  return (
    <div>
      {/* Filters */}
      <div className="flex gap-2 mb-4">
        {['all', 'pending', 'in_progress', 'completed', 'failed'].map((status) => (
          <button
            key={status}
            onClick={() => setFilter(status === 'all' ? null : status)}
            className={`px-3 py-1 rounded text-sm ${
              (filter === status || (status === 'all' && !filter))
                ? 'bg-accent text-white'
                : 'bg-bg-secondary text-text-secondary hover:text-text-primary'
            }`}
          >
            {status.replace('_', ' ')}
          </button>
        ))}
      </div>

      {/* Task list */}
      {loading ? (
        <div className="text-text-secondary">Loading...</div>
      ) : filteredTasks.length === 0 ? (
        <div className="text-text-secondary text-center py-12">No tasks found</div>
      ) : (
        <div className="space-y-3">
          {filteredTasks.map((task) => (
            <TaskCard key={task.id} task={task} />
          ))}
        </div>
      )}
    </div>
  );
}
```

### Slice 12.3.2: Create TaskCard Component

**Create `ui/src/components/TaskCard.tsx`**:
```tsx
import { Task } from '../api/client';
import { StatusDot } from './StatusDot';
import { Clock, User } from 'lucide-react';

interface TaskCardProps {
  task: Task;
  onClick?: () => void;
}

export function TaskCard({ task, onClick }: TaskCardProps) {
  const getStatusColor = (status: string) => {
    switch (status) {
      case 'completed': return 'success';
      case 'failed': return 'error';
      case 'in_progress': return 'active';
      default: return 'idle';
    }
  };

  return (
    <div
      onClick={onClick}
      className="p-4 bg-bg-secondary rounded-lg border border-border
                 hover:border-border/80 cursor-pointer transition-colors"
    >
      <div className="flex items-start justify-between gap-4">
        <div className="flex-1">
          <div className="flex items-center gap-2 mb-1">
            <StatusDot status={getStatusColor(task.status)} pulse={task.status === 'in_progress'} />
            <span className="text-sm text-text-secondary capitalize">
              {task.status.replace('_', ' ')}
            </span>
          </div>
          <h3 className="text-text-primary font-medium">{task.title}</h3>
          <p className="text-text-secondary text-sm mt-1 line-clamp-2">
            {task.description}
          </p>
        </div>
        <div className="text-right text-xs text-text-tertiary">
          <div className="flex items-center gap-1">
            <Clock size={12} />
            {new Date(task.created_at).toLocaleDateString()}
          </div>
          {task.assigned_agent && (
            <div className="flex items-center gap-1 mt-1">
              <User size={12} />
              {task.assigned_agent}
            </div>
          )}
        </div>
      </div>
    </div>
  );
}
```

### Slice 12.3.3: Create TaskDetail Modal

**Create `ui/src/components/TaskDetail.tsx`**:

```tsx
import { Task } from '../api/client';
import { X, Clock, User, Flag } from 'lucide-react';

interface TaskDetailProps {
  task: Task;
  onClose: () => void;
}

export function TaskDetail({ task, onClose }: TaskDetailProps) {
  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center">

      <div className="relative bg-bg-secondary rounded-lg max-w-lg w-full mx-4 max-h-[80vh] overflow-auto">
        <div className="sticky top-0 bg-bg-secondary border-b border-border p-4 flex justify-between items-center">
          <h2 className="text-lg font-medium text-text-primary">Task Details</h2>
          <button onClick={onClose} className="text-text-secondary hover:text-text-primary">
            <X size={20} />
          </button>
        </div>
        <div className="p-4 space-y-4">
          <div>
            <h3 className="text-xl font-medium text-text-primary">{task.title}</h3>

              <span className="flex items-center gap-1">
                <StatusDot status={task.status === 'completed' ? 'success' : 'idle'} />
                {task.status}

              <span className="flex items-center gap-1">
                <Flag size={14} />
                {task.priority}
              </span>
            </div>
          </div>
          <div>
            <h4 className="text-sm font-medium text-text-secondary mb-1">Description</h4>
            <p className="text-text-primary">{task.description}</p>
          </div>
          <div className="grid grid-cols-2 gap-4 text-sm">
            <div>
              <span className="text-text-secondary">Created</span>
              <p className="text-text-primary">{new Date(task.created_at).toLocaleString()}</p>
            </div>
            {task.assigned_agent && (
              <div>
                <span className="text-text-secondary">Assigned to</span>
                <p className="text-text-primary">{task.assigned_agent}</p>
              </div>
            )}
          </div>
        </div>
      </div>
    </div>
  );
}
```
