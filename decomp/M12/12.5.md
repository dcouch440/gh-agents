### Slice 12.5.2: Create FileBrowser Component

**Do this**:
- Create collapsible file tree
- Support expand/collapse directories
- Use CSS modules for styling
- Add proper cleanup

**Create `ui/src/components/FileBrowser/FileBrowser.tsx`**:
```tsx
import { useState, useEffect } from 'react';
import { ChevronRight, ChevronDown, File, Folder } from 'lucide-react';
import styles from './FileBrowser.module.css';

interface FileNode {
  name: string;
  path: string;
  type: 'file' | 'directory';
  children?: FileNode[];
}

interface FileBrowserProps {
  onSelect: (path: string) => void;
}

export function FileBrowser({ onSelect }: FileBrowserProps) {
  const [tree, setTree] = useState<FileNode[]>([]);
  const [expanded, setExpanded] = useState<Set<string>>(new Set());
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    let mounted = true;

    fetch('/api/files')
      .then((r) => r.json())
      .then((data) => {
        if (mounted) {
          setTree(data);
          setLoading(false);
        }
      })
      .catch((err) => {
        if (mounted) {
          console.error('Failed to load files:', err);
          setLoading(false);
        }
      });

    return () => {
      mounted = false;
    };
  }, []);

  const toggleExpand = (path: string) => {
    setExpanded((prev) => {
      const next = new Set(prev);
      if (next.has(path)) {
        next.delete(path);
      } else {
        next.add(path);
      }
      return next;
    });
  };

  const renderNode = (node: FileNode, depth = 0) => (
    <div key={node.path}>
      <div
        className={styles.node}
        style={{ paddingLeft: `${depth * 16 + 8}px` }}
        onClick={() =>
          node.type === 'file' ? onSelect(node.path) : toggleExpand(node.path)
        }
      >
        {node.type === 'directory' && (
          <span className={styles.chevron}>
            {expanded.has(node.path) ? (
              <ChevronDown size={14} />
            ) : (
              <ChevronRight size={14} />
            )}
          </span>
        )}
        <span className={styles.icon}>
          {node.type === 'directory' ? <Folder size={14} /> : <File size={14} />}
        </span>
        <span className={styles.name}>{node.name}</span>
      </div>
      {node.type === 'directory' &&
        expanded.has(node.path) &&
        node.children?.map((child) => renderNode(child, depth + 1))}
    </div>
  );

  if (loading) {
    return <div className={styles.loading}>Loading files...</div>;
  }

  return (
    <div className={styles.container}>
      {tree.map((node) => renderNode(node))}
    </div>
  );
}
```

import { useState, useEffect } from 'react';
import { CodeBlock } from '../CodeBlock';
import styles from './FileViewer.module.css';

interface FileViewerProps {
  path: string;
  onClose: () => void;
}

export function FileViewer({ path, onClose }: FileViewerProps) {
  const [content, setContent] = useState('');
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    let mounted = true;

    setLoading(true);
    fetch(`/api/files/${path}`)
      .then((r) => r.text())
      .then((data) => {
        if (mounted) {
          setContent(data);
          setLoading(false);
        }
      })
      .catch((err) => {
        if (mounted) {
          console.error('Failed to load file:', err);
          setLoading(false);
        }
      });

    return () => {
      mounted = false;
    };
  }, [path]);

  const getLanguage = (filePath: string): string => {
    const ext = filePath.split('.').pop()?.toLowerCase() || '';
    const langMap: Record<string, string> = {
      ts: 'typescript',
      tsx: 'tsx',
      js: 'javascript',
      jsx: 'jsx',
      rs: 'rust',
      py: 'python',
      go: 'go',
      md: 'markdown',
      json: 'json',
      yaml: 'yaml',
      yml: 'yaml',
      toml: 'toml',
      css: 'css',
      html: 'html',
    };
    return langMap[ext] || 'text';
  };

  return (
    <div className={styles.container}>
      <div className={styles.header}>
        <span className={styles.path}>{path}</span>
        <button onClick={onClose} className={styles.closeButton}>
          Close
        </button>
      </div>
      {loading ? (
        <div className={styles.loading}>Loading...</div>
      ) : (
        <CodeBlock code={content} language={getLanguage(path)} />
      )}
    </div>
  );
}

**Create `ui/src/components/FileEditor/FileEditor.tsx`**:
```tsx
import Editor from '@monaco-editor/react';
import { useState, useEffect } from 'react';
import { Save, X } from 'lucide-react';
import { Button } from '../Button';
import styles from './FileEditor.module.css';

interface FileEditorProps {
  path: string;
  content: string;
  language: string;
  onSave: (content: string) => Promise<void>;
  onClose: () => void;
}

export function FileEditor({
  path,
  content,
  language,
  onSave,
  onClose,
}: FileEditorProps) {
  const [value, setValue] = useState(content);
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState('');
  const modified = value !== content;

  useEffect(() => {
    setValue(content);
  }, [content]);

  const handleSave = async () => {
    setSaving(true);
    setError('');
    try {
      await onSave(value);
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Failed to save');
    } finally {
      setSaving(false);
    }
  };

  return (
    <div className={styles.container}>
      <div className={styles.header}>
        <div className={styles.pathWrapper}>
          <span className={styles.path}>{path}</span>
          {modified && <span className={styles.modifiedBadge}>Modified</span>}
        </div>
        <div className={styles.actions}>
          {error && <span className={styles.error}>{error}</span>}
          <Button
            onClick={handleSave}
            disabled={!modified || saving}
            isLoading={saving}
            variant="primary"
            size="sm"
          >
            <Save size={14} />
            Save
          </Button>
          <button onClick={onClose} className={styles.closeButton}>
            <X size={16} />
          </button>
        </div>
      </div>
      <div className={styles.editorWrapper}>
        <Editor
          height="100%"
          language={language}
          value={value}
          onChange={(v) => setValue(v || '')}
          theme="vs-dark"
          options={{
            minimap: { enabled: false },
            fontSize: 14,
            fontFamily: "'SF Mono', 'Fira Code', monospace",
            lineNumbers: 'on',
            scrollBeyondLastLine: false,
            automaticLayout: true,
          }}
        />
      </div>
    </div>
  );
}
```

### Slice 12.5.5: Create FilesPage to Integrate Components

**Do this**:
- Create page with split layout
- Browser on left, viewer/editor on right
- Handle file selection and editing

**Create `ui/src/pages/FilesPage/FilesPage.tsx`**:
```tsx
import { useState } from 'react';
import { FileBrowser } from '../../components/FileBrowser';
import { FileEditor } from '../../components/FileEditor';
import styles from './FilesPage.module.css';

export function FilesPage() {
  const [selectedFile, setSelectedFile] = useState<string | null>(null);
  const [fileContent, setFileContent] = useState('');

  const handleFileSelect = async (path: string) => {
    try {
      const response = await fetch(`/api/files/${path}`);
      const content = await response.text();
      setFileContent(content);
      setSelectedFile(path);
    } catch (err) {
      console.error('Failed to load file:', err);
    }
  };

  const handleSave = async (content: string) => {
    if (!selectedFile) return;

    const response = await fetch(`/api/files/${selectedFile}`, {
      method: 'PUT',
      headers: { 'Content-Type': 'text/plain' },
      body: content,
    });

    if (!response.ok) {
      throw new Error('Failed to save file');
    }

    setFileContent(content);
  };

  const getLanguage = (filePath: string): string => {
    const ext = filePath.split('.').pop()?.toLowerCase() || '';
    const langMap: Record<string, string> = {
      ts: 'typescript',
      tsx: 'tsx',
      js: 'javascript',
      jsx: 'jsx',
      rs: 'rust',
      py: 'python',
      go: 'go',
      md: 'markdown',
      json: 'json',
      yaml: 'yaml',
      yml: 'yaml',
      toml: 'toml',
      css: 'css',
      html: 'html',
    };
    return langMap[ext] || 'text';
  };

  return (
    <div className={styles.container}>
      <div className={styles.browser}>
        <FileBrowser onSelect={handleFileSelect} />
      </div>
      <div className={styles.content}>
        {selectedFile ? (
          <FileEditor
            path={selectedFile}
            content={fileContent}
            language={getLanguage(selectedFile)}
            onSave={handleSave}
            onClose={() => setSelectedFile(null)}
          />
        ) : (
          <div className={styles.emptyState}>
            <p>Select a file to view or edit</p>
          </div>
        )}
      </div>
    </div>
  );
}
```
