# Ticket 12.6: Diff Viewer

> View git diff for changes made by agents.

## Goal

Side-by-side or unified diff view for file changes.

**Checkpoint**: Can view diff, additions/deletions highlighted.

---

## Slices

### Slice 12.6.1: Add Diff API Endpoint

Add to Rust server:
- `GET /api/git/diff` - Get current diff
- `GET /api/git/diff/:path` - Get diff for specific file

---

### Slice 12.6.2: Create DiffViewer Component

**Commands**:
```bash
npm install react-diff-viewer-continued
```

**Create `ui/src/components/DiffViewer.tsx`**:
```tsx
import ReactDiffViewer from 'react-diff-viewer-continued';

interface DiffViewerProps {
  oldValue: string;
  newValue: string;
  fileName: string;
}

export function DiffViewer({ oldValue, newValue, fileName }: DiffViewerProps) {
  return (
    <div className="h-full overflow-auto">
      <div className="p-2 bg-bg-secondary border-b border-border">
        <span className="text-sm text-text-primary">{fileName}</span>
      </div>
      <ReactDiffViewer
        oldValue={oldValue}
        newValue={newValue}
        splitView={true}
        useDarkTheme={true}
        styles={{
          variables: {
            dark: {
              diffViewerBackground: '#0d1117',
              addedBackground: '#1a4721',
              removedBackground: '#5c2222',
              wordAddedBackground: '#2ea04326',
              wordRemovedBackground: '#f8514926',
              addedGutterBackground: '#2ea04326',
              removedGutterBackground: '#f8514926',
              gutterBackground: '#161b22',
              gutterColor: '#8b949e',
              codeFoldBackground: '#161b22',
            },
          },
        }}
      />
    </div>
  );
}
```
