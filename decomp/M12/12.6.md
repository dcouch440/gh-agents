### Slice 12.6.3: Create DiffPage to Display Git Changes

**Do this**:
- Create page to list and view diffs
- Fetch from API
- Use CSS modules
- Add proper cleanup

**Create `ui/src/pages/DiffPage/DiffPage.tsx`**:
```tsx
import { useState, useEffect } from 'react';
import { DiffViewer } from '../../components/DiffViewer';
import { Button } from '../../components/Button';
import styles from './DiffPage.module.css';

interface FileDiff {
  path: string;
  oldContent: string;
  newContent: string;
}

export function DiffPage() {
  const [diffs, setDiffs] = useState<FileDiff[]>([]);
  const [loading, setLoading] = useState(true);
  const [selectedFile, setSelectedFile] = useState<string | null>(null);
  const [splitView, setSplitView] = useState(true);

  useEffect(() => {
    let mounted = true;

    fetch('/api/git/diff')
      .then((r) => r.json())
      .then((data) => {
        if (mounted) {
          setDiffs(data);
          setLoading(false);
        }
      })
      .catch((err) => {
        if (mounted) {
          console.error('Failed to load diffs:', err);
          setLoading(false);
        }
      });

    return () => {
      mounted = false;
    };
  }, []);

  const selectedDiff = selectedFile
    ? diffs.find((d) => d.path === selectedFile)
    : null;

  if (loading) {
    return <div className={styles.loading}>Loading changes...</div>;
  }

  if (diffs.length === 0) {
    return (
      <div className={styles.emptyState}>
        <p>No changes to display</p>
      </div>
    );
  }

  return (
    <div className={styles.container}>
      <div className={styles.sidebar}>
        <div className={styles.header}>
          <h2 className={styles.title}>Changed Files</h2>
          <span className={styles.count}>{diffs.length}</span>
        </div>
        <div className={styles.fileList}>
          {diffs.map((diff) => (
            <button
              key={diff.path}
              onClick={() => setSelectedFile(diff.path)}
              className={`${styles.fileItem} ${
                selectedFile === diff.path ? styles.fileItemActive : ''
              }`}
            >
              {diff.path}
            </button>
          ))}
        </div>
      </div>

      <div className={styles.content}>
        {selectedDiff ? (
          <>
            <div className={styles.toolbar}>
              <Button
                onClick={() => setSplitView(!splitView)}
                variant="secondary"
                size="sm"
              >
                {splitView ? 'Unified View' : 'Split View'}
              </Button>
            </div>
            <DiffViewer
              oldValue={selectedDiff.oldContent}
              newValue={selectedDiff.newContent}
              fileName={selectedDiff.path}
              splitView={splitView}
            />
          </>
        ) : (
          <div className={styles.placeholder}>
            <p>Select a file to view changes</p>
          </div>
        )}
      </div>
    </div>
  );
}
```

### Slice 12.6.4: Add File-Specific Diff Support

**Do this**:
- Support viewing diff for single file
- Add route parameter handling
- Use same DiffViewer component

**Update `ui/src/pages/DiffPage/DiffPage.tsx`** to support file parameter:
```tsx
import { useState, useEffect } from 'react';
import { useParams } from 'react-router-dom';
import { DiffViewer } from '../../components/DiffViewer';
import { Button } from '../../components/Button';
import styles from './DiffPage.module.css';

interface FileDiff {
  path: string;
  oldContent: string;
  newContent: string;
}

export function DiffPage() {
  const { filePath } = useParams<{ filePath?: string }>();
  const [diffs, setDiffs] = useState<FileDiff[]>([]);
  const [loading, setLoading] = useState(true);
  const [selectedFile, setSelectedFile] = useState<string | null>(
    filePath || null
  );
  const [splitView, setSplitView] = useState(true);

  useEffect(() => {
    let mounted = true;

    const endpoint = filePath ? `/api/git/diff/${filePath}` : '/api/git/diff';

    fetch(endpoint)
      .then((r) => r.json())
      .then((data) => {
        if (mounted) {
          const diffArray = Array.isArray(data) ? data : [data];
          setDiffs(diffArray);
          setLoading(false);
          if (filePath && diffArray.length > 0) {
            setSelectedFile(diffArray[0].path);
          }
        }
      })
      .catch((err) => {
        if (mounted) {
          console.error('Failed to load diffs:', err);
          setLoading(false);
        }
      });

    return () => {
      mounted = false;
    };
  }, [filePath]);

  const selectedDiff = selectedFile
    ? diffs.find((d) => d.path === selectedFile)
    : null;

  if (loading) {
    return <div className={styles.loading}>Loading changes...</div>;
  }

  if (diffs.length === 0) {
    return (
      <div className={styles.emptyState}>
        <p>No changes to display</p>
      </div>
    );
  }

  return (
    <div className={styles.container}>
      {diffs.length > 1 && (
        <div className={styles.sidebar}>
          <div className={styles.header}>
            <h2 className={styles.title}>Changed Files</h2>
            <span className={styles.count}>{diffs.length}</span>
          </div>
          <div className={styles.fileList}>
            {diffs.map((diff) => (
              <button
                key={diff.path}
                onClick={() => setSelectedFile(diff.path)}
                className={`${styles.fileItem} ${
                  selectedFile === diff.path ? styles.fileItemActive : ''
                }`}
              >
                {diff.path}
              </button>
            ))}
          </div>
        </div>
      )}

      <div className={styles.content}>
        {selectedDiff ? (
          <>
            <div className={styles.toolbar}>
              <Button
                onClick={() => setSplitView(!splitView)}
                variant="secondary"
                size="sm"
              >
                {splitView ? 'Unified View' : 'Split View'}
              </Button>
            </div>
            <DiffViewer
              oldValue={selectedDiff.oldContent}
              newValue={selectedDiff.newContent}
              fileName={selectedDiff.path}
              splitView={splitView}
            />
          </>
        ) : (
          <div className={styles.placeholder}>
            <p>Select a file to view changes</p>
          </div>
        )}
      </div>
    </div>
  );
}
```
