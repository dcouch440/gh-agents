### Slice 12.2.1: Create FeedPage with Scrollable Feed

**Do this**:
- Create FeedPage component
- Display feed items in reverse chronological order
- Auto-scroll to newest items

**Create `ui/src/pages/FeedPage.tsx`**:
```tsx
import { useEffect, useRef } from 'react';
import { useFeed } from '../hooks/useFeed';
import { FeedItem } from '../components/FeedItem';

export function FeedPage() {
  const { items } = useFeed();
  const bottomRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    bottomRef.current?.scrollIntoView({ behavior: 'smooth' });
  }, [items]);

  return (
    <div className="space-y-2">
      <div className="flex items-center justify-between mb-4">
        <h2 className="text-lg font-medium text-text-primary">Agent Activity</h2>
        <span className="text-sm text-text-secondary">
          {items.length} events
        </span>
      </div>

      {items.length === 0 ? (
        <div className="text-center text-text-secondary py-12">
          <p>No agent activity yet</p>
          <p className="text-sm mt-2">Start a task to see agents working</p>
        </div>
      ) : (
        <div className="space-y-1">
          {items.map((item) => (
            <FeedItem key={item.id} item={item} />
          ))}
          <div ref={bottomRef} />
        </div>
      )}
    </div>
  );
}
```

### Slice 12.2.2: Create FeedItem Component

**Do this**:
- Create FeedItem component
- Style differently based on type
- Show timestamp and agent name

**Create `ui/src/components/FeedItem.tsx`**:
```tsx
import { FeedItem as FeedItemType } from '../hooks/useFeed';
import { StatusDot } from './StatusDot';

interface FeedItemProps {
  item: FeedItemType;
}

export function FeedItem({ item }: FeedItemProps) {
  const getAgentColor = (agent: string) => {
    if (agent.toLowerCase().includes('orchestrator')) return 'text-accent';
    if (agent.toLowerCase().includes('worker')) return 'text-accent-secondary';
    return 'text-text-secondary';
  };

  const getTypeStyle = (type: string) => {
    switch (type) {
      case 'milestone':
        return 'border-l-status-success bg-status-success/5';
      case 'error':
        return 'border-l-status-error bg-status-error/5';
      case 'warning':
        return 'border-l-status-warning bg-status-warning/5';
      default:
        return 'border-l-border';
    }
  };

  return (
    <div
      className={`flex gap-3 px-3 py-2 rounded-r border-l-2 font-mono text-sm
                  ${getTypeStyle(item.type)}`}
    >
      {/* Timestamp */}
      <span className="text-text-tertiary flex-shrink-0">
        {new Date(item.timestamp).toLocaleTimeString()}
      </span>

      {/* Status dot */}
      <StatusDot
        status={item.type === 'error' ? 'error' : item.type === 'milestone' ? 'success' : 'active'}
      />

      {/* Agent name */}
      <span className={`font-medium flex-shrink-0 ${getAgentColor(item.agent)}`}>
        {item.agent}
      </span>

      {/* Content */}
      <span className="text-text-primary">
        {item.content}
      </span>
    </div>
  );
}
```

- [ ] Items styled by type
- [ ] Agent name colored
- [ ] Timestamp visible

### Slice 12.2.3: Subscribe to WebSocket Feed Channel

**Do this**:
- Connect to WebSocket on mount
- Subscribe to "feed" channel
- Update items on new events

**Update `ui/src/hooks/useFeed.ts`**:
```tsx
import { useState, useEffect, useCallback } from 'react';
import { wsClient } from '../api/websocket';

export interface FeedItem {
  id: string;
  agent: string;
  content: string;
  type: 'report' | 'milestone' | 'error' | 'warning' | 'system';
  timestamp: string;
}

export function useFeed() {
  const [items, setItems] = useState<FeedItem[]>([]);
  const [connected, setConnected] = useState(false);

  useEffect(() => {
    // Connect WebSocket
    wsClient.connect().then(() => {
      setConnected(true);
      wsClient.subscribe(['feed']);
    });

    // Handle feed events
    const handleFeed = (data: FeedItem) => {
      setItems((prev) => [...prev, data].slice(-200)); // Keep last 200
    };

    wsClient.on('feed', handleFeed);

    return () => {
      wsClient.off('feed', handleFeed);
      wsClient.unsubscribe(['feed']);
    };
  }, []);

  const clear = useCallback(() => {
    setItems([]);
  }, []);

  return { items, connected, clear };
}
```

- [ ] WebSocket connects
- [ ] Feed items appear in real-time

### Slice 12.2.4: Add Auto-scroll with "New Messages" Indicator

**Do this**:
- Detect when user scrolls up
- Show "new messages" indicator
- Click to scroll to bottom

**Update `ui/src/pages/FeedPage.tsx`**:
```tsx
import { useEffect, useRef, useState } from 'react';
import { useFeed } from '../hooks/useFeed';
import { FeedItem } from '../components/FeedItem';
import { ArrowDown } from 'lucide-react';

export function FeedPage() {
  const { items } = useFeed();
  const containerRef = useRef<HTMLDivElement>(null);
  const bottomRef = useRef<HTMLDivElement>(null);
  const [autoScroll, setAutoScroll] = useState(true);
  const [newCount, setNewCount] = useState(0);

  // Check if user has scrolled up
  const handleScroll = () => {
    if (!containerRef.current) return;
    const { scrollTop, scrollHeight, clientHeight } = containerRef.current;
    const isAtBottom = scrollHeight - scrollTop - clientHeight < 50;
    setAutoScroll(isAtBottom);
    if (isAtBottom) setNewCount(0);
  };

  // Auto-scroll on new items
  useEffect(() => {
    if (autoScroll) {
      bottomRef.current?.scrollIntoView({ behavior: 'smooth' });
    } else {
      setNewCount((prev) => prev + 1);
    }
  }, [items, autoScroll]);

  const scrollToBottom = () => {
    bottomRef.current?.scrollIntoView({ behavior: 'smooth' });
    setAutoScroll(true);
    setNewCount(0);
  };

  return (
    <div className="relative h-full">
      <div
        ref={containerRef}
        onScroll={handleScroll}
        className="h-full overflow-y-auto space-y-1"
      >
        {items.map((item) => (
          <FeedItem key={item.id} item={item} />
        ))}
        <div ref={bottomRef} />
      </div>

      {/* New messages indicator */}
      {!autoScroll && newCount > 0 && (
        <button
          onClick={scrollToBottom}
          className="absolute bottom-4 left-1/2 -translate-x-1/2
                     px-4 py-2 bg-accent text-white rounded-full
                     flex items-center gap-2 shadow-lg"
        >
          <ArrowDown size={16} />
          {newCount} new {newCount === 1 ? 'event' : 'events'}
        </button>
      )}
    </div>
  );
}
```

**Verify**:
- [ ] Auto-scroll when at bottom
- [ ] Pauses when user scrolls up
- [ ] "New messages" indicator shows
- [ ] Click scrolls to bottom
