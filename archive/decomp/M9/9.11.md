│   Navigate                  →   │

│   ───────────────────────       │  <- separator item
│   Quit                      q   │  <- item with shortcut

pub struct MenuWidget<'a> {
    menu: &'a Menu,
    state: &'a MenuState,
    status: &'a MenuStatus,
}

impl<'a> MenuWidget<'a> {
    pub fn new(menu: &'a Menu, state: &'a MenuState, status: &'a MenuStatus) -> Self
}

impl Widget for MenuWidget<'_> {
    fn render(self, area: Rect, buf: &mut Buffer) {
        // 1. Draw outer border with title
        // 2. Draw status lines
        // 3. Draw separator
        // 4. Draw menu items with:
        //    - ">" for selected item
        //    - "→" for submenu items
        //    - Shortcut char right-aligned
        //    - Dimmed style for disabled items
        //    - "───" for separators
    }
}

/// Calculate required widget size based on menu content
pub fn menu_size(menu: &Menu, status: &MenuStatus) -> (u16, u16)

/// Center a rect within another rect

impl Terminal {
    /// Create new terminal, enter raw mode and alternate screen
    pub fn new() -> Result<Self>

    /// Restore terminal to normal state
    pub fn restore(&mut self) -> Result<()>

    /// Draw a frame using the provided closure
    pub fn draw<F>(&mut self, f: F) -> Result<()>
    where
        F: FnOnce(&mut Frame)

    pub fn poll_event(&self, timeout: Duration) -> Result<Option<Event>>

    fn drop(&mut self) {
        let _ = self.restore();

### Slice 9.11.3: Menu Controller

**Do this**:
- Create `src/tui/menu/controller.rs` for input handling and navigation
- Handle arrow keys, enter, escape, and shortcut keys

**Implementation**:
```rust
pub struct MenuController {
    menus: HashMap<String, Menu>,
    state: MenuState,
}

impl MenuController {
    /// Create controller with menu tree
    pub fn new(menus: HashMap<String, Menu>) -> Self

    /// Open menu system (shows main menu)
    pub fn open(&mut self)

    /// Close menu system
    pub fn close(&mut self)

    /// Check if menu is open
    pub fn is_open(&self) -> bool

    /// Get current menu being displayed
    pub fn current_menu(&self) -> Option<&Menu>

    /// Get current state
    pub fn state(&self) -> &MenuState

    /// Handle key input, return action if one was triggered
    pub fn handle_key(&mut self, key: KeyCode) -> Option<MenuAction> {
        match key {
            KeyCode::Up => { self.move_selection(-1); None }
            KeyCode::Down => { self.move_selection(1); None }
            KeyCode::Enter => self.select_current(),
            KeyCode::Esc => { self.back_or_close(); None }
            KeyCode::Char(c) => self.handle_shortcut(c),
            _ => None,
        }
    }

    /// Enable/disable a menu item by id
    pub fn set_enabled(&mut self, item_id: &str, enabled: bool)

    /// Update item label (for dynamic content like "Pending Changes (2)")
    pub fn update_label(&mut self, item_id: &str, label: &str)

    // Internal methods
    fn move_selection(&mut self, delta: i32)  // Skip separators and disabled items
    fn select_current(&mut self) -> Option<MenuAction>  // Handle action/submenu/back
    fn back_or_close(&mut self)  // Go back or close if at root
    fn handle_shortcut(&mut self, c: char) -> Option<MenuAction>
    fn find_item_mut(&mut self, item_id: &str) -> Option<&mut MenuItem>
}
```
