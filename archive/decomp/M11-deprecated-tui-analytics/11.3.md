use ratatui::{
    layout::{Constraint, Direction, Layout, Rect},
    style::{Color, Modifier, Style},
    text::{Line, Span},
    widgets::{Block, Borders, Paragraph, Row, Table, TableState, Tabs},
    Frame,
};
use crate::db::analytics::{Analytics, CostByTier, CostByModel};

pub struct CostsView {
    /// Active tab
    active_tab: CostTab,
    /// Time period filter
    period: TimePeriod,
    /// Data for each tab
    by_tier: Vec<CostByTier>,
    by_model: Vec<CostByModel>,
    by_task: Vec<CostByTask>,
    /// Table state for scrolling
    table_state: TableState,
    /// Sort column and direction
    sort: SortConfig,
}

#[derive(Debug, Clone, Copy, PartialEq)]
pub enum CostTab {
    ByTier,
    ByModel,
    ByTask,
}

impl CostTab {
    pub fn all() -> Vec<CostTab> {
        vec![CostTab::ByTier, CostTab::ByModel, CostTab::ByTask]
    }

    pub fn label(&self) -> &'static str {
        match self {
            CostTab::ByTier => "By Tier",
            CostTab::ByModel => "By Model",
            CostTab::ByTask => "By Task",
        }
    }
}

#[derive(Debug, Clone)]
pub struct CostByTask {
    pub task_id: String,
    pub title: String,
    pub cost: f64,
    pub tokens: i64,
}

#[derive(Debug, Clone)]
pub struct SortConfig {
    pub column: usize,
    pub ascending: bool,
}

#[derive(Debug, Clone, Copy, PartialEq)]
pub enum TimePeriod {
    Today,
    Week,
    Month,
    AllTime,
}

impl CostsView {
    pub fn new() -> Self {
        Self {
            active_tab: CostTab::ByTier,
            period: TimePeriod::Week,
            by_tier: Vec::new(),
            by_model: Vec::new(),
            by_task: Vec::new(),
            table_state: TableState::default(),
            sort: SortConfig { column: 1, ascending: false }, // Default: sort by cost desc
        }
    }

    pub fn next_tab(&mut self) {
        self.active_tab = match self.active_tab {
            CostTab::ByTier => CostTab::ByModel,
            CostTab::ByModel => CostTab::ByTask,
            CostTab::ByTask => CostTab::ByTier,
        };
        self.table_state.select(Some(0));
    }

    pub fn prev_tab(&mut self) {
        self.active_tab = match self.active_tab {
            CostTab::ByTier => CostTab::ByTask,
            CostTab::ByModel => CostTab::ByTier,
            CostTab::ByTask => CostTab::ByModel,
        };
        self.table_state.select(Some(0));
    }

    pub fn render(&mut self, frame: &mut Frame, area: Rect) {
        let chunks = Layout::default()
            .direction(Direction::Vertical)
            .constraints([
                Constraint::Length(3), // Tabs
                Constraint::Min(0),    // Table
                Constraint::Length(1), // Help
            ])
            .split(area);

        self.render_tabs(frame, chunks[0]);
        self.render_table(frame, chunks[1]);
        self.render_help(frame, chunks[2]);
    }

    fn render_tabs(&self, frame: &mut Frame, area: Rect) {
        let titles: Vec<Span> = CostTab::all()
            .iter()
            .map(|t| {
                let style = if *t == self.active_tab {
                    Style::default().fg(Color::Yellow).add_modifier(Modifier::BOLD)
                } else {
                    Style::default()
                };
                Span::styled(format!(" {} ", t.label()), style)
            })
            .collect();

        let tabs = Tabs::new(titles)
            .block(Block::default().borders(Borders::BOTTOM))
            .highlight_style(Style::default().fg(Color::Yellow));

        frame.render_widget(tabs, area);
    }
}

impl CostsView {
    fn render_table(&mut self, frame: &mut Frame, area: Rect) {
        match self.active_tab {
            CostTab::ByTier => self.render_tier_table(frame, area),
            CostTab::ByModel => self.render_model_table(frame, area),
            CostTab::ByTask => self.render_task_table(frame, area),
        }
    }

    fn render_tier_table(&mut self, frame: &mut Frame, area: Rect) {
        let total: f64 = self.by_tier.iter().map(|t| t.total_usd).sum();

        let rows: Vec<Row> = self.by_tier
            .iter()
            .map(|tier| {
                let pct = if total > 0.0 {
                    (tier.total_usd / total) * 100.0
                } else {
                    0.0
                };

                let tier_color = match tier.tier.as_str() {
                    "orchestrator" => Color::Magenta,
                    "worker" => Color::Blue,
                    "utility" => Color::Green,
                    _ => Color::White,
                };

                Row::new(vec![
                    Span::styled(&tier.tier, Style::default().fg(tier_color)),
                    Span::raw(format!("${:.4}", tier.total_usd)),
                    Span::raw(tier.request_count.to_string()),
                    Span::raw(format!("{:.1}%", pct)),
                ])
            })
            .collect();

        let header = Row::new(vec!["Tier", "Cost", "Requests", "%"])
            .style(Style::default().add_modifier(Modifier::BOLD))
            .bottom_margin(1);

        let table = Table::new(rows, [
            Constraint::Length(15),
            Constraint::Length(12),
            Constraint::Length(10),
            Constraint::Length(8),
        ])
        .header(header)
        .block(Block::default().borders(Borders::ALL).title("Costs by Tier"))
        .highlight_style(Style::default().bg(Color::DarkGray));

        frame.render_stateful_widget(table, area, &mut self.table_state);
    }

    fn render_model_table(&mut self, frame: &mut Frame, area: Rect) {
        let rows: Vec<Row> = self.by_model
            .iter()
            .map(|model| {
                Row::new(vec![
                    Span::raw(&model.model_id),
                    Span::raw(format!("${:.4}", model.total_usd)),
                    Span::raw(format_tokens(model.input_tokens)),
                    Span::raw(format_tokens(model.output_tokens)),
                ])
            })
            .collect();

        let header = Row::new(vec!["Model", "Cost", "Input", "Output"])
            .style(Style::default().add_modifier(Modifier::BOLD))
            .bottom_margin(1);

        let table = Table::new(rows, [
            Constraint::Min(20),
            Constraint::Length(12),
            Constraint::Length(10),
            Constraint::Length(10),
        ])
        .header(header)
        .block(Block::default().borders(Borders::ALL).title("Costs by Model"))
        .highlight_style(Style::default().bg(Color::DarkGray));

        frame.render_stateful_widget(table, area, &mut self.table_state);
    }

    fn render_task_table(&mut self, frame: &mut Frame, area: Rect) {
        let rows: Vec<Row> = self.by_task
            .iter()
            .map(|task| {
                let title = if task.title.len() > 30 {
                    format!("{}...", &task.title[..27])
                } else {
                    task.title.clone()
                };

                Row::new(vec![
                    Span::raw(&task.task_id[..8]), // Short ID
                    Span::raw(title),
                    Span::raw(format!("${:.4}", task.cost)),
                    Span::raw(format_tokens(task.tokens)),
                ])
            })
            .collect();

        let header = Row::new(vec!["Task ID", "Title", "Cost", "Tokens"])
            .style(Style::default().add_modifier(Modifier::BOLD))
            .bottom_margin(1);

        let table = Table::new(rows, [
            Constraint::Length(10),
            Constraint::Min(20),
            Constraint::Length(12),
            Constraint::Length(10),
        ])
        .header(header)
        .block(Block::default().borders(Borders::ALL).title("Costs by Task"))
        .highlight_style(Style::default().bg(Color::DarkGray));

        frame.render_stateful_widget(table, area, &mut self.table_state);
    }

    pub fn sort_by_column(&mut self, column: usize) {
        if self.sort.column == column {
            self.sort.ascending = !self.sort.ascending;
        } else {
            self.sort.column = column;
            self.sort.ascending = false;
        }
        self.apply_sort();
    }

    fn apply_sort(&mut self) {
        // Sort current tab's data based on sort config
        match self.active_tab {
            CostTab::ByTier => {
                self.by_tier.sort_by(|a, b| {
                    let cmp = a.total_usd.partial_cmp(&b.total_usd).unwrap();
                    if self.sort.ascending { cmp } else { cmp.reverse() }
                });
            }
            // Similar for other tabs...
            _ => {}
        }
    }
}

fn format_tokens(n: i64) -> String {
    if n >= 1_000_000 {
        format!("{:.1}M", n as f64 / 1_000_000.0)
    } else if n >= 1_000 {
        format!("{:.1}K", n as f64 / 1_000.0)
    } else {
        n.to_string()
    }
}

impl TimePeriod {
    pub fn all() -> Vec<TimePeriod> {
        vec![
            TimePeriod::Today,
            TimePeriod::Week,
            TimePeriod::Month,
            TimePeriod::AllTime,
        ]
    }

    pub fn label(&self) -> &'static str {
        match self {
            TimePeriod::Today => "Today",
            TimePeriod::Week => "Last 7 days",
            TimePeriod::Month => "Last 30 days",
            TimePeriod::AllTime => "All time",
        }
    }

    pub fn since(&self) -> Option<DateTime<Utc>> {
        match self {
            TimePeriod::Today => Some(Utc::now() - Duration::days(1)),
            TimePeriod::Week => Some(Utc::now() - Duration::days(7)),
            TimePeriod::Month => Some(Utc::now() - Duration::days(30)),
            TimePeriod::AllTime => None,
        }
    }
}

impl CostsView {
    pub fn cycle_period(&mut self) {
        self.period = match self.period {
            TimePeriod::Today => TimePeriod::Week,
            TimePeriod::Week => TimePeriod::Month,
            TimePeriod::Month => TimePeriod::AllTime,
            TimePeriod::AllTime => TimePeriod::Today,
        };
        // Trigger data refresh
    }

    fn render_help(&self, frame: &mut Frame, area: Rect) {
        let help = format!(
            "Tab: Switch view | ↑↓: Navigate | p: Period ({}) | s: Sort | q: Close",
            self.period.label()
        );
        let widget = Paragraph::new(help)
            .style(Style::default().fg(Color::DarkGray));
        frame.render_widget(widget, area);
    }

    pub fn handle_input(&mut self, key: Key) -> CostsAction {
        match key {
            Key::Tab => {
                self.next_tab();
                CostsAction::Refresh
            }
            Key::BackTab => {
                self.prev_tab();
                CostsAction::Refresh
            }
            Key::Char('p') => {
                self.cycle_period();
                CostsAction::Refresh
            }
            Key::Char('s') => {
                // Cycle sort column
                let max_col = 3;
                self.sort_by_column((self.sort.column + 1) % max_col);
                CostsAction::Continue
            }
            Key::Up => {
                self.table_state.select(
                    self.table_state.selected().map(|i| i.saturating_sub(1))
                );
                CostsAction::Continue
            }
            Key::Down => {
                self.table_state.select(
                    self.table_state.selected().map(|i| i + 1)
                );
                CostsAction::Continue
            }
            Key::Char('q') | Key::Esc => CostsAction::Close,
            _ => CostsAction::Continue,
        }
    }
}

pub enum CostsAction {
    Continue,
    Refresh,
    Close,
}
