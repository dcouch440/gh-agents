struct GateRow {
    id: Uuid,
    level: String,
    target_id: Uuid,
    status: String,
    agent_tier: String,
    created_at: DateTime<Utc>,
    updated_at: DateTime<Utc>,
    resolved_at: Option<DateTime<Utc>>,
}

struct GateMessageRow {
    id: Uuid,
    gate_id: Uuid,
    role: String,
    content: String,
    created_at: DateTime<Utc>,
}

#[async_trait]
impl GateRepo for PgRepo {
    async fn create_gate(&self, gate: &Gate) -> Result<()> {
        sqlx::query(
            "INSERT INTO gates (id, level, target_id, status, agent_tier, created_at, updated_at)
             VALUES ($1, $2, $3, $4, $5, $6, $7)"
        )
        .bind(gate.id.0)
        .bind(gate.level.to_string())
        .bind(gate.target_id)
        .bind(gate.status.to_string())
        .bind(gate.agent_tier.to_string())
        .bind(gate.created_at)
        .bind(gate.updated_at)
        .execute(&self.pool)
        .await?;
        Ok(())
    }
