# Ticket 7.1: File Operations

> Scoped file read/write with path validation and audit logging.

## Goal

Implement safe file operations for agents: reading and writing files within the project directory only. All operations must validate paths to prevent directory traversal attacks and log all actions for auditability.

**Checkpoint**: Agent can read and write files in project directory. Attempts to access files outside project directory are rejected with clear error. All operations appear in audit log.

File operations are the most fundamental execution capability. They must be:

4. **Async**: Non-blocking I/O for performance

```rust
pub use files::{FileOps, FileError};

use std::path::PathBuf;

/// Context for all execution operations
#[derive(Debug, Clone)]
pub struct ExecutionContext {
    /// Root directory for the project (all paths scoped here)
    pub project_root: PathBuf,

    /// Whether to enforce sandboxing
    pub sandboxed: bool,
}

impl ExecutionContext {
    pub fn new(project_root: PathBuf) -> Self {
        Self {
            project_root,
            sandboxed: true,
        }
    }

    /// Check if a path is within the project root
    pub fn is_path_allowed(&self, path: &std::path::Path) -> bool {
        // Canonicalize both paths to resolve symlinks and ..
        let canonical_root = match self.project_root.canonicalize() {
            Ok(p) => p,
            Err(_) => return false,
        };

        let canonical_path = match path.canonicalize() {
            Ok(p) => p,
            Err(_) => {
                // Path doesn't exist yet - check parent
                if let Some(parent) = path.parent() {
                    match parent.canonicalize() {
                        Ok(p) => p.join(path.file_name().unwrap_or_default()),
                        Err(_) => return false,
                    }
                } else {
                    return false;
                }
            }
        };

        canonical_path.starts_with(&canonical_root)
    }
}
```

```rust
use crate::execution::ExecutionContext;
use std::path::{Path, PathBuf};
use thiserror::Error;
use tokio::fs;

#[derive(Error, Debug)]
pub enum FileError {
    #[error("path outside project directory: {path}")]
    PathOutsideProject { path: PathBuf },

    #[error("file not found: {path}")]
    NotFound { path: PathBuf },

    #[error("permission denied: {path}")]
    PermissionDenied { path: PathBuf },

    #[error("io error: {0}")]
    IoError(#[from] std::io::Error),

    #[error("file too large: {path} ({size} bytes, max {max} bytes)")]
    FileTooLarge { path: PathBuf, size: u64, max: u64 },
}

/// Maximum file size to read (10 MB)
const MAX_FILE_SIZE: u64 = 10 * 1024 * 1024;

pub struct FileOps {
    ctx: ExecutionContext,
}

impl FileOps {
    pub fn new(ctx: ExecutionContext) -> Self {
        Self { ctx }
    }

    /// Read a file's contents as a string
    ///
    /// # Arguments
    /// * `path` - Path relative to project root, or absolute path within project
    ///
    /// # Errors
    /// * `PathOutsideProject` if path escapes project directory
    /// * `NotFound` if file doesn't exist
    /// * `FileTooLarge` if file exceeds size limit
    pub async fn read_file(&self, path: impl AsRef<Path>) -> Result<String, FileError> {
        let path = self.resolve_path(path.as_ref())?;

        // Check file size first
        let metadata = fs::metadata(&path).await.map_err(|e| {
            if e.kind() == std::io::ErrorKind::NotFound {
                FileError::NotFound { path: path.clone() }
            } else if e.kind() == std::io::ErrorKind::PermissionDenied {
                FileError::PermissionDenied { path: path.clone() }
            } else {
                FileError::IoError(e)
            }
        })?;

        if metadata.len() > MAX_FILE_SIZE {
            return Err(FileError::FileTooLarge {
                path,
                size: metadata.len(),
                max: MAX_FILE_SIZE,
            });
        }

        let content = fs::read_to_string(&path).await?;

        tracing::debug!(
            path = %path.display(),
            size = content.len(),
            "File read"
        );

        Ok(content)
    }

    /// Resolve a path relative to project root and validate it
    fn resolve_path(&self, path: &Path) -> Result<PathBuf, FileError> {
        let full_path = if path.is_absolute() {
            path.to_path_buf()
        } else {
            self.ctx.project_root.join(path)
        };

        if !self.ctx.is_path_allowed(&full_path) {
            return Err(FileError::PathOutsideProject { path: full_path });
        }

        Ok(full_path)
    }
}
```

```rust
impl FileOps {
    /// Write content to a file
    ///
    /// Creates the file if it doesn't exist, overwrites if it does.
    /// Creates parent directories as needed.
    ///
    /// # Arguments
    /// * `path` - Path relative to project root, or absolute path within project
    /// * `content` - Content to write
    ///
    /// # Errors
    /// * `PathOutsideProject` if path escapes project directory
    pub async fn write_file(
        &self,
        path: impl AsRef<Path>,
        content: impl AsRef<str>,
    ) -> Result<(), FileError> {
        let path = self.resolve_path(path.as_ref())?;

        // Create parent directories if needed
        if let Some(parent) = path.parent() {
            fs::create_dir_all(parent).await?;
        }

        let content = content.as_ref();
        fs::write(&path, content).await?;

        tracing::debug!(
            path = %path.display(),
            size = content.len(),
            "File written"
        );

        Ok(())
    }

    /// Check if a file exists
    pub async fn exists(&self, path: impl AsRef<Path>) -> Result<bool, FileError> {
        let path = self.resolve_path(path.as_ref())?;
        Ok(fs::try_exists(&path).await.unwrap_or(false))
    }

    /// Delete a file
    pub async fn delete_file(&self, path: impl AsRef<Path>) -> Result<(), FileError> {
        let path = self.resolve_path(path.as_ref())?;

        fs::remove_file(&path).await.map_err(|e| {
            if e.kind() == std::io::ErrorKind::NotFound {
                FileError::NotFound { path: path.clone() }
            } else {
                FileError::IoError(e)
            }
        })?;

        tracing::debug!(
            path = %path.display(),
            "File deleted"
        );

        Ok(())
    }

    /// List files in a directory
    pub async fn list_dir(&self, path: impl AsRef<Path>) -> Result<Vec<PathBuf>, FileError> {
        let path = self.resolve_path(path.as_ref())?;

        let mut entries = Vec::new();
        let mut dir = fs::read_dir(&path).await?;

        while let Some(entry) = dir.next_entry().await? {
            entries.push(entry.path());
        }

        Ok(entries)
    }
}
```

```rust
impl ExecutionContext {
    /// Enhanced path validation that handles symlinks safely
    pub fn validate_and_resolve_path(&self, path: &Path) -> Result<PathBuf, PathValidationError> {
        // Normalize the path first (remove . and resolve ..)
        let normalized = self.normalize_path(path)?;

        // Construct the full path
        let full_path = if normalized.is_absolute() {
            normalized
        } else {
            self.project_root.join(&normalized)
        };

        // For existing paths, canonicalize to resolve symlinks
        // For new paths, check the parent
        let resolved = if full_path.exists() {
            full_path.canonicalize()
                .map_err(|e| PathValidationError::ResolutionFailed {
                    path: full_path.clone(),
                    reason: e.to_string()
                })?
        } else {
            // File doesn't exist - validate parent
            let parent = full_path.parent()
                .ok_or_else(|| PathValidationError::NoParent { path: full_path.clone() })?;

            let resolved_parent = if parent.exists() {
                parent.canonicalize()
                    .map_err(|e| PathValidationError::ResolutionFailed {
                        path: parent.to_path_buf(),
                        reason: e.to_string()
                    })?
            } else {
                // Parent doesn't exist either - allow if it would be in project
                // This is for nested directory creation
                self.validate_hypothetical_path(parent)?
            };

            resolved_parent.join(full_path.file_name().unwrap_or_default())
        };

        // Final check: is resolved path within project root?
        let canonical_root = self.project_root.canonicalize()
            .map_err(|e| PathValidationError::InvalidProjectRoot { reason: e.to_string() })?;

        if !resolved.starts_with(&canonical_root) {
            return Err(PathValidationError::OutsideProject {
                path: path.to_path_buf(),
                resolved,
                project_root: canonical_root,
            });
        }

        Ok(resolved)
    }

    /// Normalize a path by removing . and carefully handling ..
    fn normalize_path(&self, path: &Path) -> Result<PathBuf, PathValidationError> {
        let mut components = Vec::new();

        for component in path.components() {
            match component {
                std::path::Component::Normal(c) => components.push(c),
                std::path::Component::CurDir => {} // Skip .
                std::path::Component::ParentDir => {
                    // Only allow .. if it doesn't escape
                    if components.pop().is_none() {
                        return Err(PathValidationError::EscapeAttempt {
                            path: path.to_path_buf(),
                        });
                    }
                }
                std::path::Component::RootDir => {
                    // Absolute path - clear and start fresh
                    components.clear();
                    components.push(std::ffi::OsStr::new("/"));
                }
                std::path::Component::Prefix(p) => {
                    // Windows prefix - keep it
                    components.push(p.as_os_str());
                }
            }
        }

        let normalized: PathBuf = components.iter().collect();
        Ok(normalized)
    }

    /// Validate a path that doesn't exist yet by checking its logical location
    fn validate_hypothetical_path(&self, path: &Path) -> Result<PathBuf, PathValidationError> {
        // Walk up until we find an existing ancestor
        let mut current = path.to_path_buf();
        let mut to_append = Vec::new();

        while !current.exists() {
            if let Some(name) = current.file_name() {
                to_append.push(name.to_os_string());
            }
            current = match current.parent() {
                Some(p) => p.to_path_buf(),
                None => return Err(PathValidationError::NoParent { path: path.to_path_buf() }),
            };
        }

        // Now current exists - canonicalize it
        let resolved_ancestor = current.canonicalize()
            .map_err(|e| PathValidationError::ResolutionFailed {
                path: current,
                reason: e.to_string()
            })?;

        // Rebuild the full path
        let mut result = resolved_ancestor;
        for component in to_append.into_iter().rev() {
            result.push(component);
        }

        Ok(result)
    }
}

#[derive(Error, Debug)]
pub enum PathValidationError {
    #[error("path escapes project directory: {path}")]
    EscapeAttempt { path: PathBuf },

    #[error("path outside project: {path} resolved to {resolved}, project root is {project_root}")]
    OutsideProject { path: PathBuf, resolved: PathBuf, project_root: PathBuf },

    #[error("path has no parent: {path}")]
    NoParent { path: PathBuf },

    #[error("failed to resolve path {path}: {reason}")]
    ResolutionFailed { path: PathBuf, reason: String },

    #[error("invalid project root: {reason}")]
    InvalidProjectRoot { reason: String },
}
```

```rust
use chrono::{DateTime, Utc};
use sqlx::SqlitePool;
use uuid::Uuid;

#[derive(Debug, Clone)]
pub struct FileAuditRecord {
    pub id: Uuid,
    pub task_id: Option<Uuid>,
    pub agent_id: Option<String>,
    pub operation: FileOperation,
    pub path: PathBuf,
    pub result: OperationResult,
    pub bytes: Option<u64>,
    pub timestamp: DateTime<Utc>,
}

#[derive(Debug, Clone, Copy)]
pub enum FileOperation {
    Read,
    Write,
    Delete,
    List,
    Exists,
}

impl std::fmt::Display for FileOperation {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        match self {
            FileOperation::Read => write!(f, "READ"),
            FileOperation::Write => write!(f, "WRITE"),
            FileOperation::Delete => write!(f, "DELETE"),
            FileOperation::List => write!(f, "LIST"),
            FileOperation::Exists => write!(f, "EXISTS"),
        }
    }
}

#[derive(Debug, Clone)]
pub enum OperationResult {
    Success,
    Denied(String),
    Error(String),
}

pub struct AuditedFileOps {
    inner: FileOps,
    pool: SqlitePool,
    task_id: Option<Uuid>,
    agent_id: Option<String>,
}

impl AuditedFileOps {
    pub fn new(
        ctx: ExecutionContext,
        pool: SqlitePool,
        task_id: Option<Uuid>,
        agent_id: Option<String>,
    ) -> Self {
        Self {
            inner: FileOps::new(ctx),
            pool,
            task_id,
            agent_id,
        }
    }

    async fn log_operation(
        &self,
        operation: FileOperation,
        path: &Path,
        result: &OperationResult,
        bytes: Option<u64>,
    ) {
        let record = FileAuditRecord {
            id: Uuid::new_v4(),
            task_id: self.task_id,
            agent_id: self.agent_id.clone(),
            operation,
            path: path.to_path_buf(),
            result: result.clone(),
            bytes,
            timestamp: Utc::now(),
        };

        // Log to tracing
        match &result {
            OperationResult::Success => {
                tracing::info!(
                    task_id = ?self.task_id,
                    agent_id = ?self.agent_id,
                    operation = %operation,
                    path = %path.display(),
                    bytes = ?bytes,
                    "File operation succeeded"
                );
            }
            OperationResult::Denied(reason) => {
                tracing::warn!(
                    task_id = ?self.task_id,
                    agent_id = ?self.agent_id,
                    operation = %operation,
                    path = %path.display(),
                    reason = %reason,
                    "File operation denied"
                );
            }
            OperationResult::Error(err) => {
                tracing::error!(
                    task_id = ?self.task_id,
                    agent_id = ?self.agent_id,
                    operation = %operation,
                    path = %path.display(),
                    error = %err,
                    "File operation failed"
                );
            }
        }

        // Persist to database
        if let Err(e) = self.persist_audit_record(&record).await {
            tracing::error!(error = %e, "Failed to persist audit record");
        }
    }

    async fn persist_audit_record(&self, record: &FileAuditRecord) -> Result<(), sqlx::Error> {
        let result_str = match &record.result {
            OperationResult::Success => "SUCCESS".to_string(),
            OperationResult::Denied(r) => format!("DENIED: {}", r),
            OperationResult::Error(e) => format!("ERROR: {}", e),
        };

        sqlx::query!(
            r#"
            INSERT INTO file_audit_log (id, task_id, agent_id, operation, path, result, bytes, timestamp)
            VALUES (?, ?, ?, ?, ?, ?, ?, ?)
            "#,
            record.id.to_string(),
            record.task_id.map(|u| u.to_string()),
            record.agent_id,
            record.operation.to_string(),
            record.path.to_string_lossy().to_string(),
            result_str,
            record.bytes.map(|b| b as i64),
            record.timestamp.to_rfc3339(),
        )
        .execute(&self.pool)
        .await?;

        Ok(())
    }

    pub async fn read_file(&self, path: impl AsRef<Path>) -> Result<String, FileError> {
        let path = path.as_ref();

        match self.inner.read_file(path).await {
            Ok(content) => {
                self.log_operation(
                    FileOperation::Read,
                    path,
                    &OperationResult::Success,
                    Some(content.len() as u64),
                ).await;
                Ok(content)
            }
            Err(e @ FileError::PathOutsideProject { .. }) => {
                self.log_operation(
                    FileOperation::Read,
                    path,
                    &OperationResult::Denied(e.to_string()),
                    None,
                ).await;
                Err(e)
            }
            Err(e) => {
                self.log_operation(
                    FileOperation::Read,
                    path,
                    &OperationResult::Error(e.to_string()),
                    None,
                ).await;
                Err(e)
            }
        }
    }

    pub async fn write_file(
        &self,
        path: impl AsRef<Path>,
        content: impl AsRef<str>,
    ) -> Result<(), FileError> {
        let path = path.as_ref();
        let content = content.as_ref();

        match self.inner.write_file(path, content).await {
            Ok(()) => {
                self.log_operation(
                    FileOperation::Write,
                    path,
                    &OperationResult::Success,
                    Some(content.len() as u64),
                ).await;
                Ok(())
            }
            Err(e @ FileError::PathOutsideProject { .. }) => {
                self.log_operation(
                    FileOperation::Write,
                    path,
                    &OperationResult::Denied(e.to_string()),
                    None,
                ).await;
                Err(e)
            }
            Err(e) => {
                self.log_operation(
                    FileOperation::Write,
                    path,
                    &OperationResult::Error(e.to_string()),
                    None,
                ).await;
                Err(e)
            }
        }
    }
}
```
