# Ticket 8.5: PR Retrieval & Review

> Fetch PR details, list PRs, and submit reviews via GitHub API.

## Goal

Implement PR retrieval and review capabilities that allow agents to:
- Get detailed information about a specific PR
- List open PRs with filtering and sorting
- Get the files changed in a PR
- Submit reviews (approve, request changes, comment)

**Checkpoint**: Agent can fetch a PR's details, see what files changed, and submit a review.

---

## Context

Before merging PRs, agents need to retrieve and review them. This ticket provides the read operations and review submission needed by the merge queue workflow.

**Key files**:
- `src/github/pr.rs` - Extend with retrieval methods
- `src/github/types.rs` - Add review types

**Dependencies**:
- Requires 8.1 (GitHub API Client) complete
- Uses authentication from 8.0

**References**:
- GitHub PR API: https://docs.github.com/en/rest/pulls/pulls
- GitHub Reviews API: https://docs.github.com/en/rest/pulls/reviews

impl GitHubClient {
    /// Get a pull request by number
    pub async fn get_pr(
        &self,
        owner: &str,
        repo: &str,
        number: u32,
    ) -> Result<GitHubPullRequest, GitHubError> {
        let url = format!(
            "{}/repos/{}/{}/pulls/{}",
            self.base_url, owner, repo, number
        );

        let response = self.client
            .get(&url)
            .headers(self.auth_headers())
            .send()
            .await
            .map_err(|e| GitHubError::Network(e.to_string()))?;

        match response.status() {
            reqwest::StatusCode::OK => {
                response.json().await
                    .map_err(|e| GitHubError::ParseError(e.to_string()))
            }
            reqwest::StatusCode::NOT_FOUND => {
                Err(GitHubError::NotFound {
                    resource: format!("PR #{}", number),
                })
            }
            status => {
                let body = response.text().await.unwrap_or_default();
                Err(GitHubError::ApiError {
                    status: status.as_u16(),
                    message: body,
                })
            }
        }
    }

    /// Check if a PR is mergeable
    /// Note: GitHub may return null for mergeable if still calculating
    pub async fn is_pr_mergeable(
        &self,
        owner: &str,
        repo: &str,
        number: u32,
    ) -> Result<Option<bool>, GitHubError> {
        let pr = self.get_pr(owner, repo, number).await?;
        Ok(pr.mergeable)
    }
}

// src/github/pr.rs

#[derive(Debug, Clone, Default)]
pub struct ListPrsOptions {
    /// Filter by state
    pub state: Option<PrStateFilter>,
    /// Sort by (created, updated, popularity, long-running)
    pub sort: Option<PrSort>,
    /// Sort direction
    pub direction: Option<SortDirection>,
    /// Results per page (max 100)
    pub per_page: Option<u8>,
    /// Page number
    pub page: Option<u32>,
    /// Filter by head branch
    pub head: Option<String>,
    /// Filter by base branch
    pub base: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum PrStateFilter {
    Open,
    Closed,
    All,
}

impl PrStateFilter {
    fn as_str(&self) -> &'static str {
        match self {
            PrStateFilter::Open => "open",
            PrStateFilter::Closed => "closed",
            PrStateFilter::All => "all",
        }
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum PrSort {
    Created,
    Updated,
    Popularity,
    LongRunning,
}

impl PrSort {
    fn as_str(&self) -> &'static str {
        match self {
            PrSort::Created => "created",
            PrSort::Updated => "updated",
            PrSort::Popularity => "popularity",
            PrSort::LongRunning => "long-running",
        }
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SortDirection {
    Asc,
    Desc,
}

impl SortDirection {
    fn as_str(&self) -> &'static str {
        match self {
            SortDirection::Asc => "asc",
            SortDirection::Desc => "desc",
        }
    }
}

impl GitHubClient {
    /// List pull requests for a repository
    pub async fn list_prs(
        &self,
        owner: &str,
        repo: &str,
        options: ListPrsOptions,
    ) -> Result<Vec<GitHubPullRequest>, GitHubError> {
        let mut url = format!(
            "{}/repos/{}/{}/pulls",
            self.base_url, owner, repo
        );

        // Build query params
        let mut params = Vec::new();

        if let Some(state) = options.state {
            params.push(format!("state={}", state.as_str()));
        }
        if let Some(sort) = options.sort {
            params.push(format!("sort={}", sort.as_str()));
        }
        if let Some(dir) = options.direction {
            params.push(format!("direction={}", dir.as_str()));
        }
        if let Some(per_page) = options.per_page {
            params.push(format!("per_page={}", per_page.min(100)));
        }
        if let Some(page) = options.page {
            params.push(format!("page={}", page));
        }
        if let Some(ref head) = options.head {
            params.push(format!("head={}", head));
        }
        if let Some(ref base) = options.base {
            params.push(format!("base={}", base));
        }

        if !params.is_empty() {
            url = format!("{}?{}", url, params.join("&"));
        }

        let response = self.client
            .get(&url)
            .headers(self.auth_headers())
            .send()
            .await
            .map_err(|e| GitHubError::Network(e.to_string()))?;

        self.handle_response(response).await
    }

    /// List all open PRs (convenience method)
    pub async fn list_open_prs(
        &self,
        owner: &str,
        repo: &str,
    ) -> Result<Vec<GitHubPullRequest>, GitHubError> {
        self.list_prs(owner, repo, ListPrsOptions {
            state: Some(PrStateFilter::Open),
            sort: Some(PrSort::Created),
            direction: Some(SortDirection::Asc),
            per_page: Some(100),
            ..Default::default()
        }).await
    }
}

// src/github/types.rs

#[derive(Debug, Clone, Deserialize)]
pub struct PrFile {
    /// Path to the file
    pub filename: String,
    /// Status: added, removed, modified, renamed, copied, changed, unchanged
    pub status: FileStatus,
    /// Number of additions
    pub additions: u32,
    /// Number of deletions
    pub deletions: u32,
    /// Total changes
    pub changes: u32,
    /// Patch content (may be empty for binary files or large diffs)
    pub patch: Option<String>,
    /// Previous filename (for renames)
    pub previous_filename: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, Deserialize)]
#[serde(rename_all = "lowercase")]
pub enum FileStatus {
    Added,
    Removed,
    Modified,
    Renamed,
    Copied,
    Changed,
    Unchanged,
}

// src/github/pr.rs

impl GitHubClient {
    /// Get the files changed in a pull request
    pub async fn get_pr_files(
        &self,
        owner: &str,
        repo: &str,
        number: u32,
    ) -> Result<Vec<PrFile>, GitHubError> {
        let url = format!(
            "{}/repos/{}/{}/pulls/{}/files",
            self.base_url, owner, repo, number
        );

        // GitHub paginates this - get all pages
        let mut all_files = Vec::new();
        let mut page = 1u32;

        loop {
            let page_url = format!("{}?page={}&per_page=100", url, page);

            let response = self.client
                .get(&page_url)
                .headers(self.auth_headers())
                .send()
                .await
                .map_err(|e| GitHubError::Network(e.to_string()))?;

            let files: Vec<PrFile> = self.handle_response(response).await?;

            if files.is_empty() {
                break;
            }

            all_files.extend(files);
            page += 1;

            // Safety limit
            if page > 50 {
                tracing::warn!("PR has too many files, stopping at 5000");
                break;
            }
        }

        tracing::debug!(
            pr = number,
            file_count = all_files.len(),
            "Retrieved PR files"
        );

        Ok(all_files)
    }

    /// Get a summary of changes in a PR
    pub async fn get_pr_change_summary(
        &self,
        owner: &str,
        repo: &str,
        number: u32,
    ) -> Result<PrChangeSummary, GitHubError> {
        let files = self.get_pr_files(owner, repo, number).await?;

        let mut summary = PrChangeSummary::default();

        for file in &files {
            summary.total_files += 1;
            summary.additions += file.additions;
            summary.deletions += file.deletions;

            match file.status {
                FileStatus::Added => summary.files_added += 1,
                FileStatus::Removed => summary.files_removed += 1,
                FileStatus::Modified | FileStatus::Changed => summary.files_modified += 1,
                FileStatus::Renamed => summary.files_renamed += 1,
                _ => {}
            }
        }

        Ok(summary)
    }
}

#[derive(Debug, Clone, Default)]
pub struct PrChangeSummary {
    pub total_files: u32,
    pub files_added: u32,
    pub files_removed: u32,
    pub files_modified: u32,
    pub files_renamed: u32,
    pub additions: u32,
    pub deletions: u32,
}

// src/github/types.rs

#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize)]
#[serde(rename_all = "SCREAMING_SNAKE_CASE")]
pub enum ReviewEvent {
    /// Approve the PR
    Approve,
    /// Request changes before merging
    RequestChanges,
    /// Comment without approving or requesting changes
    Comment,
}

#[derive(Debug, Clone, Serialize)]
pub struct CreateReviewRequest {
    /// Review event type
    pub event: ReviewEvent,
    /// Optional body for the review
    #[serde(skip_serializing_if = "Option::is_none")]
    pub body: Option<String>,
    /// Line comments (optional)
    #[serde(skip_serializing_if = "Vec::is_empty")]
    pub comments: Vec<ReviewComment>,
    /// Commit SHA to review (optional, uses latest if not specified)
    #[serde(skip_serializing_if = "Option::is_none")]
    pub commit_id: Option<String>,
}

#[derive(Debug, Clone, Serialize)]
pub struct ReviewComment {
    /// Path to the file
    pub path: String,
    /// Line number in the diff (not the file)
    pub position: u32,
    /// Comment body
    pub body: String,
}

#[derive(Debug, Clone, Deserialize)]
pub struct GitHubReview {
    pub id: u64,
    pub user: GitHubUser,
    pub body: Option<String>,
    pub state: ReviewState,
    pub html_url: String,
    pub submitted_at: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, Deserialize)]
#[serde(rename_all = "SCREAMING_SNAKE_CASE")]
pub enum ReviewState {
    Approved,
    ChangesRequested,
    Commented,
    Dismissed,
    Pending,
}

// src/github/pr.rs

impl GitHubClient {
    /// Submit a review on a pull request
    pub async fn create_review(
        &self,
        owner: &str,
        repo: &str,
        number: u32,
        review: CreateReviewRequest,
    ) -> Result<GitHubReview, GitHubError> {
        let url = format!(
            "{}/repos/{}/{}/pulls/{}/reviews",
            self.base_url, owner, repo, number
        );

        let response = self.client
            .post(&url)
            .headers(self.auth_headers())
            .json(&review)
            .send()
            .await
            .map_err(|e| GitHubError::Network(e.to_string()))?;

        match response.status() {
            reqwest::StatusCode::OK | reqwest::StatusCode::CREATED => {
                let result = response.json().await
                    .map_err(|e| GitHubError::ParseError(e.to_string()))?;

                tracing::info!(
                    pr = number,
                    event = ?review.event,
                    "Submitted PR review"
                );

                Ok(result)
            }
            reqwest::StatusCode::FORBIDDEN => {
                Err(GitHubError::PermissionDenied {
                    action: "submit review".to_string(),
                })
            }
            reqwest::StatusCode::UNPROCESSABLE_ENTITY => {
                let body = response.text().await.unwrap_or_default();
                Err(GitHubError::ValidationError {
                    message: body,
                })
            }
            status => {
                let body = response.text().await.unwrap_or_default();
                Err(GitHubError::ApiError {
                    status: status.as_u16(),
                    message: body,
                })
            }
        }
    }

    /// Approve a PR with optional comment
    pub async fn approve_pr(
        &self,
        owner: &str,
        repo: &str,
        number: u32,
        comment: Option<&str>,
    ) -> Result<GitHubReview, GitHubError> {
        self.create_review(owner, repo, number, CreateReviewRequest {
            event: ReviewEvent::Approve,
            body: comment.map(String::from),
            comments: Vec::new(),
            commit_id: None,
        }).await
    }

    /// Request changes on a PR
    pub async fn request_changes(
        &self,
        owner: &str,
        repo: &str,
        number: u32,
        reason: &str,
    ) -> Result<GitHubReview, GitHubError> {
        self.create_review(owner, repo, number, CreateReviewRequest {
            event: ReviewEvent::RequestChanges,
            body: Some(reason.to_string()),
            comments: Vec::new(),
            commit_id: None,
        }).await
    }

    /// List reviews on a PR
    pub async fn list_reviews(
        &self,
        owner: &str,
        repo: &str,
        number: u32,
    ) -> Result<Vec<GitHubReview>, GitHubError> {
        let url = format!(
            "{}/repos/{}/{}/pulls/{}/reviews",
            self.base_url, owner, repo, number
        );

        let response = self.client
            .get(&url)
            .headers(self.auth_headers())
            .send()
            .await
            .map_err(|e| GitHubError::Network(e.to_string()))?;

        self.handle_response(response).await
    }
}
