use ratatui::{
    buffer::Buffer,
    layout::Rect,
    style::{Color, Modifier, Style},
    text::{Line, Span},
    widgets::{Block, Borders, List, ListItem, ListState, Widget},

use crate::agents::roles::{Role, RoleCategory, RoleLibrary};

/// Role selector widget with category grouping
pub struct RoleSelector {
    pub library: RoleLibrary,
    pub state: RoleSelectorState,
}

pub struct RoleSelectorState {
    pub selected_category: usize,
    pub selected_role: usize,
    pub expanded_categories: Vec<bool>,
}

impl Default for RoleSelectorState {
    fn default() -> Self {
        Self {
            selected_category: 0,
            selected_role: 0,
            expanded_categories: vec![true, false, false, false], // Analysis expanded by default
        }
    }
}

impl RoleSelector {
    pub fn new(library: RoleLibrary) -> Self {
        Self {
            library,
            state: RoleSelectorState::default(),
        }
    }

    /// Get the currently selected role
    pub fn selected_role(&self) -> Option<&Role> {
        let categories: Vec<_> = self.library.categories().collect();
        let category = categories.get(self.state.selected_category)?;
        let roles = self.library.roles_in_category(category);
        roles.get(self.state.selected_role)
    }

    /// Navigate to next item
    pub fn next(&mut self) {
        // Navigate within category, or to next category
        let categories: Vec<_> = self.library.categories().collect();
        if let Some(category) = categories.get(self.state.selected_category) {
            let roles = self.library.roles_in_category(category);
            if self.state.selected_role < roles.len().saturating_sub(1) {
                self.state.selected_role += 1;
            } else if self.state.selected_category < categories.len() - 1 {
                self.state.selected_category += 1;
                self.state.selected_role = 0;
            }
        }
    }

    /// Navigate to previous item
    pub fn previous(&mut self) {
        if self.state.selected_role > 0 {
            self.state.selected_role -= 1;
        } else if self.state.selected_category > 0 {
            self.state.selected_category -= 1;
            let categories: Vec<_> = self.library.categories().collect();
            if let Some(category) = categories.get(self.state.selected_category) {
                let roles = self.library.roles_in_category(category);
                self.state.selected_role = roles.len().saturating_sub(1);
            }
        }
    }

    /// Toggle category expansion
    pub fn toggle_category(&mut self) {
        if let Some(expanded) = self.state.expanded_categories.get_mut(self.state.selected_category) {
            *expanded = !*expanded;
        }
    }
}

impl Widget for &RoleSelector {
    fn render(self, area: Rect, buf: &mut Buffer) {
        let block = Block::default()
            .title(" Select Role ")
            .borders(Borders::ALL);
        let inner = block.inner(area);
        block.render(area, buf);

        let mut items: Vec<ListItem> = Vec::new();
        let categories: Vec<_> = self.library.categories().collect();

        for (cat_idx, category) in categories.iter().enumerate() {
            // Category header
            let icon = category.icon();
            let expanded = self.state.expanded_categories.get(cat_idx).copied().unwrap_or(false);
            let arrow = if expanded { "▼" } else { "▶" };

            let cat_style = if cat_idx == self.state.selected_category && self.state.selected_role == 0 {
                Style::default().fg(Color::Yellow).add_modifier(Modifier::BOLD)
            } else {
                Style::default().fg(Color::White).add_modifier(Modifier::BOLD)
            };

            items.push(ListItem::new(Line::from(vec![
                Span::styled(format!("{} {} ", arrow, icon), cat_style),
                Span::styled(category.display_name(), cat_style),
            ])));

            // Roles in category (if expanded)
            if expanded {
                let roles = self.library.roles_in_category(category);
                for (role_idx, role) in roles.iter().enumerate() {
                    let is_selected = cat_idx == self.state.selected_category
                        && role_idx == self.state.selected_role;

                    let style = if is_selected {
                        Style::default().fg(Color::Cyan).add_modifier(Modifier::BOLD)
                    } else {
                        Style::default().fg(Color::Gray)
                    };

                    items.push(ListItem::new(Line::from(vec![
                        Span::raw("    ├── "),
                        Span::styled(&role.name, style),
                    ])));

                    // Show description on next line
                    items.push(ListItem::new(Line::from(vec![
                        Span::raw("    │   "),
                        Span::styled(&role.description, Style::default().fg(Color::DarkGray)),
                    ])));
                }
            }
        }

        let list = List::new(items);
        Widget::render(list, inner, buf);
    }
}

// Add to src/tui/views/roles.rs

use std::collections::HashSet;
use regex::Regex;

/// Form for creating a custom role
pub struct RoleCreator {
    pub name: String,
    pub description: String,
    pub category: RoleCategory,
    pub template: String,
    pub variables: Vec<TemplateVariable>,
    pub active_field: RoleCreatorField,
}

#[derive(Debug, Clone, PartialEq)]
pub enum RoleCreatorField {
    Name,
    Description,
    Category,
    Template,
}

#[derive(Debug, Clone)]
pub struct TemplateVariable {
    pub name: String,
    pub var_type: VariableType,
    pub label: String,
    pub required: bool,
}

#[derive(Debug, Clone, PartialEq)]
pub enum VariableType {
    Text,
    Files,
    Number,
    Choice(Vec<String>),
}

impl RoleCreator {
    pub fn new() -> Self {
        Self {
            name: String::new(),
            description: String::new(),
            category: RoleCategory::Analysis,
            template: String::new(),
            variables: Vec::new(),
            active_field: RoleCreatorField::Name,
        }
    }

    /// Extract variables from template text
    /// Looks for patterns like {name}, {scope}, {files}
    pub fn extract_variables(&mut self) {
        let re = Regex::new(r"\{([a-zA-Z_][a-zA-Z0-9_]*)\}").unwrap();
        let mut seen: HashSet<String> = HashSet::new();
        self.variables.clear();

        for cap in re.captures_iter(&self.template) {
            let var_name = cap[1].to_string();
            if seen.insert(var_name.clone()) {
                // Infer type from common names
                let var_type = match var_name.as_str() {
                    "files" | "file" | "target_files" | "code_files" | "doc_file" => VariableType::Files,
                    "count" | "number" | "max" | "min" | "limit" => VariableType::Number,
                    _ => VariableType::Text,
                };

                self.variables.push(TemplateVariable {
                    name: var_name.clone(),
                    var_type,
                    label: humanize_variable_name(&var_name),
                    required: true,
                });
            }
        }
    }

    /// Move to next field
    pub fn next_field(&mut self) {
        self.active_field = match self.active_field {
            RoleCreatorField::Name => RoleCreatorField::Description,
            RoleCreatorField::Description => RoleCreatorField::Category,
            RoleCreatorField::Category => RoleCreatorField::Template,
            RoleCreatorField::Template => RoleCreatorField::Name,
        };
    }

    /// Move to previous field
    pub fn prev_field(&mut self) {
        self.active_field = match self.active_field {
            RoleCreatorField::Name => RoleCreatorField::Template,
            RoleCreatorField::Description => RoleCreatorField::Name,
            RoleCreatorField::Category => RoleCreatorField::Description,
            RoleCreatorField::Template => RoleCreatorField::Category,
        };
    }

    /// Handle text input for current field
    pub fn handle_char(&mut self, c: char) {
        match self.active_field {
            RoleCreatorField::Name => self.name.push(c),
            RoleCreatorField::Description => self.description.push(c),
            RoleCreatorField::Category => {} // Category uses different input
            RoleCreatorField::Template => {
                self.template.push(c);
                self.extract_variables(); // Re-extract on change
            }
        }
    }

    /// Handle backspace for current field
    pub fn handle_backspace(&mut self) {
        match self.active_field {
            RoleCreatorField::Name => { self.name.pop(); }
            RoleCreatorField::Description => { self.description.pop(); }
            RoleCreatorField::Category => {}
            RoleCreatorField::Template => {
                self.template.pop();
                self.extract_variables();
            }
        }
    }

    /// Build the role from form data
    pub fn build_role(&self) -> Option<Role> {
        if self.name.is_empty() || self.template.is_empty() {
            return None;
        }

        Some(Role {
            id: RoleId::new_custom(&self.name),
            name: self.name.clone(),
            category: self.category.clone(),
            description: self.description.clone(),
            system_prompt: String::new(), // Derived from template at spawn time
            style: CommunicationStyle::Technical,
            required_reading: vec![],
            can_delegate_to: vec![],
            output_format: OutputFormat::Result,
            max_delegation_depth: 2,
            template: Some(RoleTemplate {
                prompt: self.template.clone(),
                variables: self.variables.clone(),
            }),
        })
    }
}

fn humanize_variable_name(name: &str) -> String {
    // Convert snake_case to Title Case
    name.split('_')
        .map(|word| {
            let mut chars = word.chars();
            match chars.next() {
                Some(first) => first.to_uppercase().chain(chars).collect(),
                None => String::new(),
            }
        })
        .collect::<Vec<_>>()
        .join(" ")
}

// Add to src/tui/views/roles.rs

/// Form for filling in template variables when spawning an agent
pub struct VariableInputForm {
    pub role: Role,
    pub values: HashMap<String, VariableValue>,
    pub active_variable: usize,
    pub file_picker_state: Option<FilePickerState>,
}

#[derive(Debug, Clone)]
pub enum VariableValue {
    Text(String),
    Files(Vec<PathBuf>),
    Number(i64),
    Choice(String),
}

pub struct FilePickerState {
    pub root: PathBuf,
    pub entries: Vec<FileEntry>,
    pub selected: usize,
    pub checked: HashSet<usize>,
}

pub struct FileEntry {
    pub path: PathBuf,
    pub is_dir: bool,
    pub depth: usize,
}

impl VariableInputForm {
    pub fn new(role: Role) -> Self {
        let mut values = HashMap::new();

        // Initialize default values for each variable
        if let Some(ref template) = role.template {
            for var in &template.variables {
                let default = match var.var_type {
                    VariableType::Text => VariableValue::Text(String::new()),
                    VariableType::Files => VariableValue::Files(vec![]),
                    VariableType::Number => VariableValue::Number(0),
                    VariableType::Choice(ref opts) => {
                        VariableValue::Choice(opts.first().cloned().unwrap_or_default())
                    }
                };
                values.insert(var.name.clone(), default);
            }
        }

        Self {
            role,
            values,
            active_variable: 0,
            file_picker_state: None,
        }
    }

    /// Get the current variable being edited
    pub fn current_variable(&self) -> Option<&TemplateVariable> {
        self.role.template.as_ref()?.variables.get(self.active_variable)
    }

    /// Navigate to next variable
    pub fn next_variable(&mut self) {
        if let Some(ref template) = self.role.template {
            if self.active_variable < template.variables.len().saturating_sub(1) {
                self.active_variable += 1;
            }
        }
    }

    /// Navigate to previous variable
    pub fn prev_variable(&mut self) {
        if self.active_variable > 0 {
            self.active_variable -= 1;
        }
    }

    /// Handle text input for current variable
    pub fn handle_char(&mut self, c: char) {
        if let Some(var) = self.current_variable() {
            if let Some(value) = self.values.get_mut(&var.name) {
                match value {
                    VariableValue::Text(ref mut s) => s.push(c),
                    VariableValue::Number(ref mut n) => {
                        if c.is_ascii_digit() {
                            *n = n.saturating_mul(10).saturating_add((c as i64) - ('0' as i64));
                        }
                    }
                    _ => {}
                }
            }
        }
    }

    /// Generate the filled prompt from template
    pub fn generate_prompt(&self) -> String {
        let Some(ref template) = self.role.template else {
            return self.role.system_prompt.clone();
        };

        let mut prompt = template.prompt.clone();
        for (name, value) in &self.values {
            let replacement = match value {
                VariableValue::Text(s) => s.clone(),
                VariableValue::Files(paths) => {
                    paths.iter()
                        .map(|p| format!("- {}", p.display()))
                        .collect::<Vec<_>>()
                        .join("\n")
                }
                VariableValue::Number(n) => n.to_string(),
                VariableValue::Choice(s) => s.clone(),
            };
            prompt = prompt.replace(&format!("{{{}}}", name), &replacement);
        }
        prompt
    }

    /// Check if all required variables are filled
    pub fn is_complete(&self) -> bool {
        let Some(ref template) = self.role.template else {
            return true;
        };

        for var in &template.variables {
            if !var.required {
                continue;
            }
            match self.values.get(&var.name) {
                Some(VariableValue::Text(s)) if s.is_empty() => return false,
                Some(VariableValue::Files(f)) if f.is_empty() => return false,
                None => return false,
                _ => {}
            }
        }
        true
    }
}

impl Widget for &VariableInputForm {
    fn render(self, area: Rect, buf: &mut Buffer) {
        let block = Block::default()
            .title(format!(" Spawn Agent: {} ", self.role.name))
            .borders(Borders::ALL);
        let inner = block.inner(area);
        block.render(area, buf);

        let Some(ref template) = self.role.template else {
            return;
        };

        let mut y = inner.y;

        // Header
        buf.set_string(inner.x + 1, y, "Fill in the template variables:", Style::default());
        y += 2;

        // Each variable
        for (idx, var) in template.variables.iter().enumerate() {
            let is_active = idx == self.active_variable;
            let label_style = if is_active {
                Style::default().fg(Color::Yellow).add_modifier(Modifier::BOLD)
            } else {
                Style::default().fg(Color::White)
            };

            // Label
            buf.set_string(inner.x + 1, y, &format!("{}:", var.label), label_style);
            y += 1;

            // Value display
            let value = self.values.get(&var.name);
            let value_str = match value {
                Some(VariableValue::Text(s)) => {
                    if is_active { format!("{}▋", s) } else { s.clone() }
                }
                Some(VariableValue::Files(paths)) => {
                    format!("[{} files selected]", paths.len())
                }
                Some(VariableValue::Number(n)) => n.to_string(),
                Some(VariableValue::Choice(s)) => s.clone(),
                None => String::new(),
            };

            let value_style = if is_active {
                Style::default().fg(Color::Cyan)
            } else {
                Style::default().fg(Color::Gray)
            };

            buf.set_string(inner.x + 3, y, &value_str, value_style);
            y += 2;
        }

        // Preview section
        y += 1;
        buf.set_string(inner.x + 1, y, "Preview:", Style::default().add_modifier(Modifier::BOLD));
        y += 1;

        let preview = self.generate_prompt();
        for line in preview.lines().take(5) {
            buf.set_string(inner.x + 3, y, line, Style::default().fg(Color::DarkGray));
            y += 1;
        }
        if preview.lines().count() > 5 {
            buf.set_string(inner.x + 3, y, "...", Style::default().fg(Color::DarkGray));
        }
    }
}

pub struct App {

    role_creator: Option<RoleCreator>,

    role_library: RoleLibrary,
}

pub enum CurrentView {
    // ... existing variants ...
    RoleSelection,
    RoleCreation,
    VariableInput,
}

impl App {
    /// Open role selection view
    pub fn open_role_selector(&mut self) {
        self.role_selector = Some(RoleSelector::new(self.role_library.clone()));
        self.current_view = CurrentView::RoleSelection;
    }

    /// Open role creation view
    pub fn open_role_creator(&mut self) {
        self.role_creator = Some(RoleCreator::new());
        self.current_view = CurrentView::RoleCreation;
    }

    /// Confirm role selection and proceed to variable input
    pub fn confirm_role_selection(&mut self) {
        if let Some(ref selector) = self.role_selector {
            if let Some(role) = selector.selected_role() {
                if role.template.is_some() {
                    // Role has variables - show input form
                    self.variable_input = Some(VariableInputForm::new(role.clone()));
                    self.current_view = CurrentView::VariableInput;
                } else {
                    // No variables - spawn directly
                    self.spawn_agent_with_role(role.clone(), HashMap::new());
                }
            }
        }
    }

    /// Confirm variable input and spawn agent
    pub fn confirm_variable_input(&mut self) {
        if let Some(ref form) = self.variable_input {
            if form.is_complete() {
                let prompt = form.generate_prompt();
                // Spawn agent with filled prompt
                self.spawn_agent_with_prompt(form.role.clone(), prompt);
                self.variable_input = None;
                self.current_view = CurrentView::Chat;
            }
        }
    }

    /// Save custom role to library
    pub fn save_custom_role(&mut self) {
        if let Some(ref creator) = self.role_creator {
            if let Some(role) = creator.build_role() {
                self.role_library.add_custom_role(role.clone());
                // Persist to .nexor/roles/
                self.persist_role(&role);
                self.role_creator = None;
                self.current_view = CurrentView::RoleSelection;
            }
        }
    }
}

    fn spawn_agent_with_prompt(&mut self, role: Role, prompt: String) {
        // Create the full task assignment with role context
        let mut role_with_prompt = role.clone();
        role_with_prompt.system_prompt = prompt;

        // Send spawn request to agent pool
        if let Some(ref tx) = self.agent_spawn_tx {
            let _ = tx.try_send(SpawnRequest {

            role_category: Some(role.category.display_name().to_string()),
