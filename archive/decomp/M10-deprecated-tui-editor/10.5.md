```rust
use crate::execution::files::{FileOps, FileError};

impl<'a> FileEditor<'a> {
    pub async fn save(&mut self, file_ops: &FileOps) -> Result<(), SaveError> {
        let content = self.content();

        file_ops.write_file(&self.path, &content)
            .await
            .map_err(SaveError::WriteError)?;

        self.original_content = content;
        self.modified = false;

        tracing::info!(path = %self.path.display(), "File saved");

        Ok(())
    }

    pub fn needs_save(&self) -> bool {
        self.modified
    }
}

#[derive(Debug, thiserror::Error)]
pub enum SaveError {
    #[error("failed to write file: {0}")]
    WriteError(#[from] FileError),
}
```

```rust
use ratatui::{
    layout::{Alignment, Constraint, Direction, Layout, Rect},
    style::{Color, Modifier, Style},
    text::{Line, Span},
    widgets::{Block, Borders, Clear, Paragraph},
    Frame,
};

pub struct ConfirmDialog {
    title: String,
    message: String,
    options: Vec<String>,
    selected: usize,
}

impl ConfirmDialog {
    pub fn new(title: impl Into<String>, message: impl Into<String>, options: Vec<String>) -> Self {
        Self {
            title: title.into(),
            message: message.into(),
            options,
            selected: 0,
        }
    }

    pub fn unsaved_changes() -> Self {
        Self::new(
            "Unsaved Changes",
            "You have unsaved changes. What would you like to do?",
            vec![
                "Save".to_string(),
                "Don't Save".to_string(),
                "Cancel".to_string(),
            ],
        )
    }

    pub fn select_next(&mut self) {
        self.selected = (self.selected + 1) % self.options.len();
    }

    pub fn select_prev(&mut self) {
        self.selected = if self.selected == 0 {
            self.options.len() - 1
        } else {
            self.selected - 1
        };
    }

    pub fn confirm(&self) -> usize {
        self.selected
    }

    pub fn render(&self, frame: &mut Frame, area: Rect) {
        // Center the dialog
        let dialog_width = 50;
        let dialog_height = 8;
        let x = (area.width.saturating_sub(dialog_width)) / 2;
        let y = (area.height.saturating_sub(dialog_height)) / 2;
        let dialog_area = Rect::new(x, y, dialog_width, dialog_height);

        // Clear the area behind the dialog
        frame.render_widget(Clear, dialog_area);

        // Render dialog box
        let block = Block::default()
            .title(self.title.as_str())
            .borders(Borders::ALL)
            .border_style(Style::default().fg(Color::Yellow));

        let inner = block.inner(dialog_area);
        frame.render_widget(block, dialog_area);

        // Render message
        let message = Paragraph::new(self.message.as_str())
            .alignment(Alignment::Center);
        let message_area = Rect::new(inner.x, inner.y, inner.width, 2);
        frame.render_widget(message, message_area);

        // Render options
        let options_area = Rect::new(inner.x, inner.y + 3, inner.width, 2);
        let option_spans: Vec<Span> = self.options
            .iter()
            .enumerate()
            .flat_map(|(i, opt)| {
                let style = if i == self.selected {
                    Style::default().fg(Color::Black).bg(Color::White).add_modifier(Modifier::BOLD)
                } else {
                    Style::default()
                };
                vec![
                    Span::raw(" "),
                    Span::styled(format!("[{}]", opt), style),
                    Span::raw(" "),
                ]
            })
            .collect();

        let options = Paragraph::new(Line::from(option_spans))
            .alignment(Alignment::Center);
        frame.render_widget(options, options_area);
    }
}
```

```rust
use ratatui::{
    layout::{Alignment, Constraint, Direction, Layout, Rect},
    style::{Color, Style},
    widgets::{Block, Borders, Clear, Paragraph},
    Frame,
};

pub struct CommitDialog {
    files: Vec<String>,
    message: String,
    cursor_pos: usize,
}

impl CommitDialog {
    pub fn new(files: Vec<String>) -> Self {
        Self {
            files,
            message: String::new(),
            cursor_pos: 0,
        }
    }

    pub fn message(&self) -> &str {
        &self.message
    }

    pub fn insert_char(&mut self, c: char) {
        self.message.insert(self.cursor_pos, c);
        self.cursor_pos += 1;
    }

    pub fn delete_char(&mut self) {
        if self.cursor_pos > 0 {
            self.cursor_pos -= 1;
            self.message.remove(self.cursor_pos);
        }
    }

    pub fn move_cursor_left(&mut self) {
        self.cursor_pos = self.cursor_pos.saturating_sub(1);
    }

    pub fn move_cursor_right(&mut self) {
        self.cursor_pos = (self.cursor_pos + 1).min(self.message.len());
    }

    pub fn render(&self, frame: &mut Frame, area: Rect) {
        let dialog_width = 60;
        let dialog_height = 12;
        let x = (area.width.saturating_sub(dialog_width)) / 2;
        let y = (area.height.saturating_sub(dialog_height)) / 2;
        let dialog_area = Rect::new(x, y, dialog_width, dialog_height);

        frame.render_widget(Clear, dialog_area);

        let block = Block::default()
            .title("Commit Changes")
            .borders(Borders::ALL)
            .border_style(Style::default().fg(Color::Green));

        let inner = block.inner(dialog_area);
        frame.render_widget(block, dialog_area);

        // Files section
        let files_text = format!("Files: {}", self.files.join(", "));
        let files_widget = Paragraph::new(files_text)
            .style(Style::default().fg(Color::DarkGray));
        let files_area = Rect::new(inner.x, inner.y, inner.width, 2);
        frame.render_widget(files_widget, files_area);

        // Message label
        let label = Paragraph::new("Commit message:");
        let label_area = Rect::new(inner.x, inner.y + 2, inner.width, 1);
        frame.render_widget(label, label_area);

        // Message input
        let input_style = Style::default().bg(Color::DarkGray);
        let display_message = if self.message.is_empty() {
            "Enter commit message..."
        } else {
            &self.message
        };
        let input = Paragraph::new(display_message)
            .style(if self.message.is_empty() {
                input_style.fg(Color::Gray)
            } else {
                input_style.fg(Color::White)
            });
        let input_area = Rect::new(inner.x, inner.y + 3, inner.width, 3);
        frame.render_widget(input, input_area);

        // Help text
        let help = Paragraph::new("Enter: Commit | Esc: Cancel")
            .style(Style::default().fg(Color::DarkGray))
            .alignment(Alignment::Center);
        let help_area = Rect::new(inner.x, inner.y + 7, inner.width, 1);
        frame.render_widget(help, help_area);
    }
}
```

### Slice 10.5.4: Create Commit on Current Branch via git2

**Do this**:
- Stage the saved file
- Create commit with user's message
- Use current branch
- Handle errors (not a git repo, etc.)

**Create/modify**:
- `src/execution/git.rs`
- `src/tui/dialogs/commit_dialog.rs`

**Code to add to git.rs**:
```rust
use git2::{Repository, Signature, IndexAddOption};

pub struct GitOps {
    repo: Repository,
}

impl GitOps {
    pub fn open(path: &Path) -> Result<Self, git2::Error> {
        let repo = Repository::open(path)?;
        Ok(Self { repo })
    }

    pub fn commit_file(&self, file_path: &Path, message: &str) -> Result<git2::Oid, GitError> {
        // Get relative path
        let workdir = self.repo.workdir()
            .ok_or_else(|| GitError::NotAWorkingDirectory)?;
        let relative_path = file_path.strip_prefix(workdir)
            .map_err(|_| GitError::PathOutsideRepo)?;

        // Stage the file
        let mut index = self.repo.index()?;
        index.add_path(relative_path)?;
        index.write()?;

        // Create tree from index
        let tree_oid = index.write_tree()?;
        let tree = self.repo.find_tree(tree_oid)?;

        // Get HEAD commit as parent
        let head = self.repo.head()?;
        let parent_commit = head.peel_to_commit()?;

        // Create signature
        let sig = self.repo.signature()?;

        // Create commit
        let commit_oid = self.repo.commit(
            Some("HEAD"),
            &sig,
            &sig,
            message,
            &tree,
            &[&parent_commit],
        )?;

        tracing::info!(
            commit = %commit_oid,
            file = %relative_path.display(),
            "Created commit"
        );

        Ok(commit_oid)
    }
}

#[derive(Debug, thiserror::Error)]
pub enum GitError {
    #[error("not a git repository")]
    NotARepo,
    #[error("not a working directory")]
    NotAWorkingDirectory,
    #[error("path is outside repository")]
    PathOutsideRepo,
    #[error("git error: {0}")]
    Git(#[from] git2::Error),
}
```

```rust
use ratatui::{
    layout::Rect,
    style::{Color, Style},
    widgets::{Block, Borders, Clear, Paragraph},
    Frame,
};
use std::time::{Duration, Instant};

pub struct Notification {
    message: String,
    kind: NotificationKind,
    created_at: Instant,
    duration: Duration,
}

#[derive(Debug, Clone, Copy)]
pub enum NotificationKind {
    Success,
    Error,
    Info,
}

impl Notification {
    pub fn success(message: impl Into<String>) -> Self {
        Self {
            message: message.into(),
            kind: NotificationKind::Success,
            created_at: Instant::now(),
            duration: Duration::from_secs(3),
        }
    }

    pub fn error(message: impl Into<String>) -> Self {
        Self {
            message: message.into(),
            kind: NotificationKind::Error,
            created_at: Instant::now(),
            duration: Duration::from_secs(5),
        }
    }

    pub fn info(message: impl Into<String>) -> Self {
        Self {
            message: message.into(),
            kind: NotificationKind::Info,
            created_at: Instant::now(),
            duration: Duration::from_secs(3),
        }
    }

    pub fn is_expired(&self) -> bool {
        self.created_at.elapsed() >= self.duration
    }

    pub fn render(&self, frame: &mut Frame, area: Rect) {
        let width = (self.message.len() + 4).min(50) as u16;
        let height = 3;
        let x = area.width.saturating_sub(width).saturating_sub(2);
        let y = area.height.saturating_sub(height).saturating_sub(2);
        let notif_area = Rect::new(x, y, width, height);

        frame.render_widget(Clear, notif_area);

        let (border_color, icon) = match self.kind {
            NotificationKind::Success => (Color::Green, "✓"),
            NotificationKind::Error => (Color::Red, "✗"),
            NotificationKind::Info => (Color::Blue, "ℹ"),
        };

        let content = format!("{} {}", icon, self.message);
        let widget = Paragraph::new(content)
            .block(Block::default()
                .borders(Borders::ALL)
                .border_style(Style::default().fg(border_color)));

        frame.render_widget(widget, notif_area);
    }
}

pub struct NotificationManager {
    notifications: Vec<Notification>,
}

impl NotificationManager {
    pub fn new() -> Self {
        Self { notifications: Vec::new() }
    }

    pub fn push(&mut self, notification: Notification) {
        self.notifications.push(notification);
    }

    pub fn tick(&mut self) {
        self.notifications.retain(|n| !n.is_expired());
    }

    pub fn render(&self, frame: &mut Frame, area: Rect) {
        // Render only the most recent notification
        if let Some(notif) = self.notifications.last() {
            notif.render(frame, area);
        }
    }
}
```
