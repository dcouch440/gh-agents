# Ticket 10.4: Diff Viewer

> Show changes made by agents

use ratatui::{
    layout::Rect,
    style::{Color, Style},
    text::{Line, Span},
    widgets::{Block, Borders, Paragraph, Scrollbar, ScrollbarOrientation, ScrollbarState},
    Frame,
};
use std::path::PathBuf;

#[derive(Debug, Clone)]
pub struct DiffHunk {
    /// Old file line range (start, count)
    pub old_range: (usize, usize),
    /// New file line range (start, count)
    pub new_range: (usize, usize),
    /// Lines in this hunk
    pub lines: Vec<DiffLine>,
}

#[derive(Debug, Clone)]
pub struct DiffLine {
    pub kind: DiffLineKind,
    pub content: String,
}

#[derive(Debug, Clone, Copy, PartialEq)]
pub enum DiffLineKind {
    Context,
    Addition,
    Deletion,
    Header,
}

pub struct DiffViewer {
    /// File path being diffed
    path: PathBuf,
    /// Parsed diff hunks
    hunks: Vec<DiffHunk>,
    /// All lines for rendering
    lines: Vec<DiffLine>,
    /// Current scroll position
    scroll_offset: usize,
    /// Current hunk index (for navigation)
    current_hunk: usize,
    /// View mode
    view_mode: DiffViewMode,
}

#[derive(Debug, Clone, Copy, PartialEq)]
pub enum DiffViewMode {
    Unified,
    SideBySide,
}

impl DiffViewer {
    pub fn new(path: PathBuf, diff_text: &str) -> Self {
        let (hunks, lines) = Self::parse_diff(diff_text);
        Self {
            path,
            hunks,
            lines,
            scroll_offset: 0,
            current_hunk: 0,
            view_mode: DiffViewMode::Unified,
        }
    }

    fn parse_diff(diff_text: &str) -> (Vec<DiffHunk>, Vec<DiffLine>) {
        let mut hunks = Vec::new();
        let mut all_lines = Vec::new();
        let mut current_hunk: Option<DiffHunk> = None;

        for line in diff_text.lines() {
            let diff_line = if line.starts_with("@@") {
                // Hunk header
                if let Some(hunk) = current_hunk.take() {
                    hunks.push(hunk);
                }
                // Parse hunk header: @@ -old_start,old_count +new_start,new_count @@
                current_hunk = Some(Self::parse_hunk_header(line));
                DiffLine {
                    kind: DiffLineKind::Header,
                    content: line.to_string(),
                }
            } else if line.starts_with('+') && !line.starts_with("+++") {
                DiffLine {
                    kind: DiffLineKind::Addition,
                    content: line[1..].to_string(),
                }
            } else if line.starts_with('-') && !line.starts_with("---") {
                DiffLine {
                    kind: DiffLineKind::Deletion,
                    content: line[1..].to_string(),
                }
            } else if line.starts_with(' ') {
                DiffLine {
                    kind: DiffLineKind::Context,
                    content: line[1..].to_string(),
                }
            } else {
                DiffLine {
                    kind: DiffLineKind::Context,
                    content: line.to_string(),
                }
            };

            if let Some(ref mut hunk) = current_hunk {
                hunk.lines.push(diff_line.clone());
            }
            all_lines.push(diff_line);
        }

        if let Some(hunk) = current_hunk {
            hunks.push(hunk);
        }

        (hunks, all_lines)
    }

    fn parse_hunk_header(line: &str) -> DiffHunk {
        // Basic parsing: @@ -1,5 +1,7 @@
        DiffHunk {
            old_range: (1, 5), // TODO: Parse properly
            new_range: (1, 7),
            lines: Vec::new(),
        }
    }

    pub fn render(&self, frame: &mut Frame, area: Rect) {
        let visible_height = area.height as usize;
        let visible_lines: Vec<Line> = self.lines
            .iter()
            .skip(self.scroll_offset)
            .take(visible_height)
            .map(|line| {
                let style = match line.kind {
                    DiffLineKind::Addition => Style::default().fg(Color::Green),
                    DiffLineKind::Deletion => Style::default().fg(Color::Red),
                    DiffLineKind::Header => Style::default().fg(Color::Cyan),
                    DiffLineKind::Context => Style::default(),
                };
                let prefix = match line.kind {
                    DiffLineKind::Addition => "+",
                    DiffLineKind::Deletion => "-",
                    DiffLineKind::Header => "",
                    DiffLineKind::Context => " ",
                };
                Line::from(Span::styled(format!("{}{}", prefix, line.content), style))
            })
            .collect();

        let title = format!("Diff: {}", self.path.display());
        let widget = Paragraph::new(visible_lines)
            .block(Block::default().borders(Borders::ALL).title(title));

        frame.render_widget(widget, area);
    }
}

use git2::{Repository, DiffOptions, DiffFormat};

impl DiffViewer {
    pub fn from_git_diff(repo_path: &Path, file_path: &Path) -> Result<Self, git2::Error> {
        let repo = Repository::open(repo_path)?;
        let relative_path = file_path.strip_prefix(repo_path).unwrap_or(file_path);

        let mut opts = DiffOptions::new();
        opts.pathspec(relative_path);

        // Get diff between HEAD and working directory
        let diff = repo.diff_index_to_workdir(None, Some(&mut opts))?;

        let mut diff_text = String::new();
        diff.print(DiffFormat::Patch, |_delta, _hunk, line| {
            let prefix = match line.origin() {
                '+' => "+",
                '-' => "-",
                ' ' => " ",
                '>' => ">",
                '<' => "<",
                'H' => "", // Header
                _ => "",
            };
            if let Ok(content) = std::str::from_utf8(line.content()) {
                diff_text.push_str(prefix);
                diff_text.push_str(content);
            }
            true
        })?;

        if diff_text.is_empty() {
            // Check if file is untracked
            let status = repo.status_file(relative_path)?;
            if status.is_wt_new() {
                // Untracked file - show all content as additions
                let content = std::fs::read_to_string(file_path)?;
                diff_text = content.lines()
                    .map(|l| format!("+{}\n", l))
                    .collect();
            } else {
                diff_text = "No changes".to_string();
            }
        }

        Ok(Self::new(file_path.to_path_buf(), &diff_text))
    }

impl DiffViewer {
    fn render_unified(&self, frame: &mut Frame, area: Rect) {
        let visible_height = area.height.saturating_sub(2) as usize; // Account for borders

        let visible_lines: Vec<Line> = self.lines
            .iter()
            .skip(self.scroll_offset)
            .take(visible_height)
            .map(|line| {
                let (prefix, style) = match line.kind {
                    DiffLineKind::Addition => (
                        "+",
                        Style::default().fg(Color::Black).bg(Color::Green),
                    ),
                    DiffLineKind::Deletion => (
                        "-",
                        Style::default().fg(Color::Black).bg(Color::Red),
                    ),
                    DiffLineKind::Header => (
                        "",
                        Style::default().fg(Color::Cyan).add_modifier(Modifier::BOLD),
                    ),
                    DiffLineKind::Context => (
                        " ",
                        Style::default().fg(Color::DarkGray),
                    ),
                };

                Line::from(Span::styled(
                    format!("{}{}", prefix, line.content),
                    style,
                ))
            })
            .collect();

        let stats = self.get_stats();
        let title = format!(
            "Diff: {} (+{} -{} lines)",
            self.path.display(),
            stats.additions,
            stats.deletions,
        );

        let widget = Paragraph::new(visible_lines)
            .block(Block::default().borders(Borders::ALL).title(title));

        frame.render_widget(widget, area);
    }

    fn get_stats(&self) -> DiffStats {
        let additions = self.lines.iter()
            .filter(|l| l.kind == DiffLineKind::Addition)
            .count();
        let deletions = self.lines.iter()
            .filter(|l| l.kind == DiffLineKind::Deletion)
            .count();
        DiffStats { additions, deletions }
    }
}

struct DiffStats {
    additions: usize,
    deletions: usize,
}

impl DiffViewer {
    pub fn next_hunk(&mut self) {
        if self.hunks.is_empty() {
            return;
        }

        self.current_hunk = (self.current_hunk + 1).min(self.hunks.len() - 1);
        self.scroll_to_hunk(self.current_hunk);
    }

    pub fn prev_hunk(&mut self) {
        if self.hunks.is_empty() {
            return;
        }

        self.current_hunk = self.current_hunk.saturating_sub(1);
        self.scroll_to_hunk(self.current_hunk);
    }

    fn scroll_to_hunk(&mut self, hunk_idx: usize) {
        // Find the line index where this hunk starts
        let mut line_idx = 0;
        for (i, line) in self.lines.iter().enumerate() {
            if line.kind == DiffLineKind::Header {
                if line_idx == hunk_idx {
                    self.scroll_offset = i;
                    return;
                }
                line_idx += 1;
            }
        }
    }

    pub fn scroll_up(&mut self, amount: usize) {
        self.scroll_offset = self.scroll_offset.saturating_sub(amount);
    }

    pub fn scroll_down(&mut self, amount: usize) {
        let max = self.lines.len().saturating_sub(1);
        self.scroll_offset = (self.scroll_offset + amount).min(max);
    }

    pub fn handle_input(&mut self, key: Key) -> DiffAction {
        match key {
            Key::Char('n') => {
                self.next_hunk();
                DiffAction::Continue
            }
            Key::Char('p') => {
                self.prev_hunk();
                DiffAction::Continue
            }
            Key::Up => {
                self.scroll_up(1);
                DiffAction::Continue
            }
            Key::Down => {
                self.scroll_down(1);
                DiffAction::Continue
            }
            Key::PageUp => {
                self.scroll_up(20);
                DiffAction::Continue
            }
            Key::PageDown => {
                self.scroll_down(20);
                DiffAction::Continue
            }
            Key::Esc | Key::Char('q') => DiffAction::Exit,
            _ => DiffAction::Continue,
        }
    }

    fn render_status_bar(&self, frame: &mut Frame, area: Rect) {
        let hunk_info = if self.hunks.is_empty() {
            "No changes".to_string()
        } else {
            format!("Hunk {}/{}", self.current_hunk + 1, self.hunks.len())
        };

        let help = "n Next | p Prev | ↑↓ Scroll | q Close";
        let status = format!("{} | {}", hunk_info, help);

        let widget = Paragraph::new(status)
            .style(Style::default().bg(Color::DarkGray).fg(Color::White));
        frame.render_widget(widget, area);
    }
}

pub enum DiffAction {
    Continue,
    Exit,
}
