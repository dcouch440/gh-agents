# Ticket 10.7: Agent Integration

> Connect editor to agent workflow

## Goal

Integrate the file viewing/editing features with the agent workflow so users can easily view files agents are working on, see their changes, and intervene when needed.

**Checkpoint**: Can see files an agent is modifying, view changes in real-time, and handle conflicts when both user and agent edit the same file.

```rust
    pub fn handle_action(&self, key: char) -> Option<FeedAction> {
        if let Some(selected_item) = self.get_selected_item() {
            match (&selected_item.content, key) {
                (FeedContent::FileModified { path, .. }, 'v') => {
                    Some(FeedAction::ViewFile(path.clone()))
                }
                (FeedContent::FileModified { path, .. }, 'd') => {
                    Some(FeedAction::DiffFile(path.clone()))
                }
                (FeedContent::FileCreated { path, .. }, 'v') |
                (FeedContent::FileCreated { path, .. }, 'e') => {
                    Some(FeedAction::EditFile(path.clone()))
                }
                _ => None,
            }
        } else {
            None
        }
    }
}
```

```rust
pub struct FileBrowser {
    // ... existing fields ...

    /// Files modified by agents (tracked from agent events)
    agent_modified_files: HashSet<PathBuf>,
}

impl FileBrowser {
    pub fn mark_agent_modified(&mut self, path: PathBuf) {
        self.agent_modified_files.insert(path);
    }

    pub fn clear_agent_modified(&mut self, path: &Path) {
        self.agent_modified_files.remove(path);
    }

    fn render_node(&self, node: &FileNode, idx: usize) -> ListItem {
        let indent = "  ".repeat(node.depth);

        let is_agent_modified = self.agent_modified_files.contains(&node.path);

        let icon = if node.is_dir {
            if node.expanded { "📂" } else { "📁" }
        } else if is_agent_modified {
            "🤖" // Robot icon for agent-modified files
        } else {
            "📄"
        };

        let name_style = if is_agent_modified {
            Style::default().fg(Color::Yellow).add_modifier(Modifier::BOLD)
        } else if idx == self.selected {
            Style::default().bg(Color::Blue).fg(Color::White)
        } else {
            Style::default()
        };

        let status = if is_agent_modified {
            " [agent modified]"
        } else {
            ""
        };

        ListItem::new(Line::from(vec![
            Span::raw(indent),
            Span::raw(format!("{} ", icon)),
            Span::styled(node.name.clone(), name_style),
            Span::styled(status, Style::default().fg(Color::Yellow)),
        ]))
    }

    pub fn filter_agent_modified(&mut self) {
        // Toggle filter to show only agent-modified files
        self.filter_mode = Some(FilterMode::AgentModified);
        self.refresh();
    }
}

// In app.rs, connect agent events to file browser

impl App {
    fn handle_agent_event(&mut self, event: AgentEvent) {
        match event {
            AgentEvent::FileModified { path, .. } => {
                if let Some(browser) = &mut self.file_browser {

                }
                // Also emit to feed
                self.feed.push(FeedItem::file_modified(path, agent_id));
            }
            AgentEvent::FileCreated { path, .. } => {
                if let Some(browser) = &mut self.file_browser {
                    browser.mark_agent_modified(path);
                }
            }
            // ...
        }
    }
}
```

```rust
        self.lines = new_content.lines().map(String::from).collect();

    }

    pub fn can_auto_refresh(&self) -> bool {
        // Can refresh if no unsaved changes
        !self.modified
    }

    pub fn refresh(&mut self, new_content: String) {
        let lines: Vec<String> = new_content.lines().map(String::from).collect();
        self.textarea = TextArea::new(lines);
        self.original_content = new_content;
        self.modified = false;
    }
}

// In app.rs

impl App {
    fn handle_file_modified_event(&mut self, path: &Path) {
        // Check if this file is currently open
        match &mut self.current_view {
            AppView::FileViewer(viewer) if viewer.path() == path => {
                // Always refresh viewer
                if let Ok(content) = std::fs::read_to_string(path) {
                    viewer.refresh(content);
                    self.notifications.push(Notification::info(
                        "File updated by agent"
                    ));
                }
            }
            AppView::FileEditor(editor) if editor.path() == path => {
                if editor.can_auto_refresh() {
                    // No unsaved changes, auto-refresh
                    if let Ok(content) = std::fs::read_to_string(path) {
                        editor.refresh(content);
                        self.notifications.push(Notification::info(
                            "File updated by agent (auto-refreshed)"
                        ));
                    }
                } else {
                    // User has unsaved changes - conflict!
                    self.show_conflict_dialog(path);
                }
            }
            _ => {}
        }
    }
}
```

```rust
// src/tui/dialogs/conflict_dialog.rs

pub struct ConflictDialog {
    path: PathBuf,
    user_content: String,
    agent_content: String,
    selected_option: usize,
}

impl ConflictDialog {
    pub fn new(path: PathBuf, user_content: String, agent_content: String) -> Self {
        Self {
            path,
            user_content,
            agent_content,
            selected_option: 0,
        }
    }

    pub fn options() -> Vec<&'static str> {
        vec![
            "Keep my changes",
            "Take agent's changes",
            "View diff",
            "Cancel",
        ]
    }

    pub fn select_next(&mut self) {
        self.selected_option = (self.selected_option + 1) % Self::options().len();
    }

    pub fn select_prev(&mut self) {
        self.selected_option = if self.selected_option == 0 {
            Self::options().len() - 1
        } else {
            self.selected_option - 1
        };
    }

    pub fn confirm(&self) -> ConflictResolution {
        match self.selected_option {
            0 => ConflictResolution::KeepMine,
            1 => ConflictResolution::TakeTheirs,
            2 => ConflictResolution::ViewDiff,
            _ => ConflictResolution::Cancel,
        }
    }

    pub fn render(&self, frame: &mut Frame, area: Rect) {
        let dialog_width = 60;
        let dialog_height = 12;
        let x = (area.width.saturating_sub(dialog_width)) / 2;
        let y = (area.height.saturating_sub(dialog_height)) / 2;
        let dialog_area = Rect::new(x, y, dialog_width, dialog_height);

        frame.render_widget(Clear, dialog_area);

        let block = Block::default()
            .title("⚠️  Conflict Detected")
            .borders(Borders::ALL)
            .border_style(Style::default().fg(Color::Red));

        let inner = block.inner(dialog_area);
        frame.render_widget(block, dialog_area);

        // Message
        let message = format!(
            "An agent modified '{}' while you were editing it.\n\
             Your unsaved changes conflict with the agent's changes.",
            self.path.file_name().unwrap_or_default().to_string_lossy()
        );
        let msg_widget = Paragraph::new(message)
            .wrap(ratatui::widgets::Wrap { trim: true });
        let msg_area = Rect::new(inner.x + 1, inner.y + 1, inner.width - 2, 3);
        frame.render_widget(msg_widget, msg_area);

        // Options
        let options: Vec<Span> = Self::options()
            .iter()
            .enumerate()
            .flat_map(|(i, opt)| {
                let style = if i == self.selected_option {
                    Style::default().fg(Color::Black).bg(Color::White)
                } else {
                    Style::default()
                };
                vec![
                    Span::raw(" "),
                    Span::styled(format!("[{}]", opt), style),
                    Span::raw(" "),
                ]
            })
            .collect();

        let options_widget = Paragraph::new(Line::from(options));
        let options_area = Rect::new(inner.x + 1, inner.y + 5, inner.width - 2, 2);
        frame.render_widget(options_widget, options_area);
    }
}

#[derive(Debug, Clone, Copy)]
pub enum ConflictResolution {
    KeepMine,
    TakeTheirs,
    ViewDiff,
    Cancel,
}

// In app.rs

impl App {
    fn show_conflict_dialog(&mut self, path: &Path) {
        if let AppView::FileEditor(editor) = &self.current_view {
            let user_content = editor.content();
            if let Ok(agent_content) = std::fs::read_to_string(path) {
                self.dialog = Some(Dialog::Conflict(ConflictDialog::new(
                    path.to_path_buf(),
                    user_content,
                    agent_content,
                )));
            }
        }
    }

    fn handle_conflict_resolution(&mut self, resolution: ConflictResolution) {
        match resolution {
            ConflictResolution::KeepMine => {
                // User's changes are already in the editor, do nothing
                self.notifications.push(Notification::info("Keeping your changes"));
            }
            ConflictResolution::TakeTheirs => {
                // Reload file from disk
                if let AppView::FileEditor(editor) = &mut self.current_view {
                    if let Ok(content) = std::fs::read_to_string(editor.path()) {
                        editor.refresh(content);
                        self.notifications.push(Notification::info("Loaded agent's changes"));
                    }
                }
            }
            ConflictResolution::ViewDiff => {
                // Switch to diff view
                // (Implementation would create a side-by-side diff of user vs agent)
                self.notifications.push(Notification::info("Diff view not yet implemented"));
            }
            ConflictResolution::Cancel => {
                // Close dialog, keep current state
            }
        }
        self.dialog = None;
    }
}
```

- [ ] "Keep mine" preserves user's edits
- [ ] "Take theirs" loads agent's version
- [ ] Dialog shows file name and clear explanation
