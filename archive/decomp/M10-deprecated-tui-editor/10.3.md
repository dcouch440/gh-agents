**Code skeleton**:
```rust
use ratatui::{
    layout::Rect,
    style::{Color, Modifier, Style},
    text::{Line, Span},
    widgets::{Block, Borders, List, ListItem, ListState},
    Frame,
};
use std::path::{Path, PathBuf};

#[derive(Debug, Clone)]
pub struct FileNode {
    pub path: PathBuf,
    pub name: String,
    pub is_dir: bool,
    pub depth: usize,
    pub expanded: bool,
    pub children_loaded: bool,
}

pub struct FileBrowser {
    /// Root directory
    root: PathBuf,
    /// Flattened list of visible nodes
    visible_nodes: Vec<FileNode>,
    /// Currently selected index
    selected: usize,
    /// List state for scrolling
    list_state: ListState,
    /// File type filter (e.g., "rs", "ts")
    filter: Option<String>,
}

impl FileBrowser {
    pub fn new(root: PathBuf) -> std::io::Result<Self> {
        let mut browser = Self {
            root: root.clone(),
            visible_nodes: Vec::new(),
            selected: 0,
            list_state: ListState::default(),
            filter: None,
        };
        browser.load_directory(&root, 0)?;
        browser.list_state.select(Some(0));
        Ok(browser)
    }

    fn load_directory(&mut self, path: &Path, depth: usize) -> std::io::Result<()> {
        let mut entries: Vec<_> = std::fs::read_dir(path)?
            .filter_map(|e| e.ok())
            .collect();

        // Sort: directories first, then alphabetically
        entries.sort_by(|a, b| {
            let a_is_dir = a.file_type().map(|t| t.is_dir()).unwrap_or(false);
            let b_is_dir = b.file_type().map(|t| t.is_dir()).unwrap_or(false);
            match (a_is_dir, b_is_dir) {
                (true, false) => std::cmp::Ordering::Less,
                (false, true) => std::cmp::Ordering::Greater,
                _ => a.file_name().cmp(&b.file_name()),
            }
        });

        for entry in entries {
            let path = entry.path();
            let name = entry.file_name().to_string_lossy().to_string();

            // Skip hidden files
            if name.starts_with('.') {
                continue;
            }

            let is_dir = entry.file_type().map(|t| t.is_dir()).unwrap_or(false);

            // Apply filter
            if let Some(ref ext) = self.filter {
                if !is_dir && !name.ends_with(&format!(".{}", ext)) {
                    continue;
                }
            }

            self.visible_nodes.push(FileNode {
                path,
                name,
                is_dir,
                depth,
                expanded: false,
                children_loaded: false,
            });
        }

        Ok(())
    }

    pub fn move_up(&mut self) {
        if self.selected > 0 {
            self.selected -= 1;
            self.list_state.select(Some(self.selected));
        }
    }

    pub fn move_down(&mut self) {
        if self.selected < self.visible_nodes.len().saturating_sub(1) {
            self.selected += 1;
            self.list_state.select(Some(self.selected));
        }
    }

    pub fn selected_path(&self) -> Option<&PathBuf> {
        self.visible_nodes.get(self.selected).map(|n| &n.path)
    }

    pub fn render(&mut self, frame: &mut Frame, area: Rect) {
        let items: Vec<ListItem> = self.visible_nodes
            .iter()
            .enumerate()
            .map(|(idx, node)| {
                let indent = "  ".repeat(node.depth);
                let icon = if node.is_dir {
                    if node.expanded { "📂" } else { "📁" }
                } else {
                    "📄"
                };

                let style = if idx == self.selected {
                    Style::default().bg(Color::Blue).fg(Color::White)
                } else {
                    Style::default()
                };

                ListItem::new(Line::from(vec![
                    Span::raw(indent),
                    Span::raw(format!("{} {}", icon, node.name)),
                ])).style(style)
            })
            .collect();

        let list = List::new(items)
            .block(Block::default().borders(Borders::ALL).title("Files"));

        frame.render_stateful_widget(list, area, &mut self.list_state);
    }
}
```

### Slice 10.3.2: Add Expand/Collapse for Directories

**Create/modify**:

**Code to add**:
```rust
impl FileBrowser {
    pub fn toggle_selected(&mut self) -> Option<PathBuf> {
        let selected = self.selected;
        if let Some(node) = self.visible_nodes.get(selected) {
            if node.is_dir {
                self.toggle_directory(selected);
                None
            } else {
                // Return file path for opening
                Some(node.path.clone())
            }
        } else {
            None
        }
    }

    fn toggle_directory(&mut self, idx: usize) {
        let node = &self.visible_nodes[idx];
        let path = node.path.clone();
        let depth = node.depth;
        let was_expanded = node.expanded;

        if was_expanded {
            // Collapse: remove all children
            self.collapse_at(idx);
        } else {
            // Expand: load and insert children
            self.expand_at(idx);
        }
    }

    fn expand_at(&mut self, idx: usize) {
        let node = &mut self.visible_nodes[idx];
        node.expanded = true;

        let path = node.path.clone();
        let child_depth = node.depth + 1;

        // Load children
        let mut children = Vec::new();
        if let Ok(entries) = std::fs::read_dir(&path) {
            let mut entries: Vec<_> = entries.filter_map(|e| e.ok()).collect();

            entries.sort_by(|a, b| {
                let a_is_dir = a.file_type().map(|t| t.is_dir()).unwrap_or(false);
                let b_is_dir = b.file_type().map(|t| t.is_dir()).unwrap_or(false);
                match (a_is_dir, b_is_dir) {
                    (true, false) => std::cmp::Ordering::Less,
                    (false, true) => std::cmp::Ordering::Greater,
                    _ => a.file_name().cmp(&b.file_name()),
                }
            });

            for entry in entries {
                let child_path = entry.path();
                let name = entry.file_name().to_string_lossy().to_string();

                if name.starts_with('.') {
                    continue;
                }

                let is_dir = entry.file_type().map(|t| t.is_dir()).unwrap_or(false);

                if let Some(ref ext) = self.filter {
                    if !is_dir && !name.ends_with(&format!(".{}", ext)) {
                        continue;
                    }
                }

                children.push(FileNode {
                    path: child_path,
                    name,
                    is_dir,
                    depth: child_depth,
                    expanded: false,
                    children_loaded: false,
                });
            }
        }

        // Insert children after current node
        let insert_pos = idx + 1;
        for (i, child) in children.into_iter().enumerate() {
            self.visible_nodes.insert(insert_pos + i, child);
        }
    }

    fn collapse_at(&mut self, idx: usize) {
        let node = &mut self.visible_nodes[idx];
        node.expanded = false;
        let parent_depth = node.depth;

        // Remove all nodes with greater depth until we hit same or lesser depth
        let mut remove_count = 0;
        for i in (idx + 1)..self.visible_nodes.len() {
            if self.visible_nodes[i].depth > parent_depth {
                remove_count += 1;
            } else {
                break;
            }
        }

        self.visible_nodes.drain((idx + 1)..(idx + 1 + remove_count));
    }
}
```

```rust
impl FileBrowser {
    pub fn set_filter(&mut self, extension: Option<String>) {
        self.filter = extension;
        self.refresh();
    }

    pub fn refresh(&mut self) {
        let root = self.root.clone();
        self.visible_nodes.clear();
        self.selected = 0;
        let _ = self.load_directory(&root, 0);
        self.list_state.select(Some(0));
    }

    fn render_status_bar(&self, frame: &mut Frame, area: Rect) {
        let filter_text = match &self.filter {
            Some(ext) => format!("Filter: *.{}", ext),
            None => "Filter: none (press 'f' to filter)".to_string(),
        };

        let selected_info = self.selected_path()
            .map(|p| p.display().to_string())
            .unwrap_or_default();

        let status = format!("{} | {}", selected_info, filter_text);
        let widget = Paragraph::new(status)
            .style(Style::default().bg(Color::DarkGray).fg(Color::White));
        frame.render_widget(widget, area);
    }
}

pub struct FileBrowserWithFilter {
    browser: FileBrowser,
    filter_mode: bool,
    filter_input: String,
}

impl FileBrowserWithFilter {
    pub fn handle_input(&mut self, input: Input) -> BrowserAction {
        if self.filter_mode {
            match input.key {
                Key::Enter => {
                    let ext = if self.filter_input.is_empty() {
                        None
                    } else {
                        Some(self.filter_input.clone())
                    };
                    self.browser.set_filter(ext);
                    self.filter_mode = false;
                    BrowserAction::Continue
                }
                Key::Esc => {
                    self.filter_mode = false;
                    BrowserAction::Continue
                }
                Key::Char(c) => {
                    self.filter_input.push(c);
                    BrowserAction::Continue
                }
                Key::Backspace => {
                    self.filter_input.pop();
                    BrowserAction::Continue
                }
                _ => BrowserAction::Continue,
            }
        } else {
            match input.key {
                Key::Char('f') => {
                    self.filter_mode = true;
                    self.filter_input.clear();
                    BrowserAction::Continue
                }
                Key::Up => {
                    self.browser.move_up();
                    BrowserAction::Continue
                }
                Key::Down => {
                    self.browser.move_down();
                    BrowserAction::Continue
                }
                Key::Enter => {
                    if let Some(path) = self.browser.toggle_selected() {
                        BrowserAction::OpenFile(path)
                    } else {
                        BrowserAction::Continue
                    }
                }
                Key::Esc => BrowserAction::Exit,
                _ => BrowserAction::Continue,
            }
        }
    }
}

pub enum BrowserAction {
    Continue,
    OpenFile(PathBuf),
    Exit,
}
```

```rust
    ViewFile(PathBuf),
    EditFile(PathBuf),
    Exit,
}

impl FileBrowserWithFilter {
    pub fn handle_input(&mut self, input: Input) -> BrowserAction {
        if self.filter_mode {
            // ... filter handling ...
        } else {
            match input.key {
                Key::Enter => {
                    // View file
                    if let Some(path) = self.browser.toggle_selected() {
                        BrowserAction::ViewFile(path)
                    } else {
                        BrowserAction::Continue
                    }
                }
                Key::Char('e') => {
                    // Edit file
                    if let Some(node) = self.browser.visible_nodes.get(self.browser.selected) {

                    }
                    BrowserAction::Continue
                }
                // ... other keys ...
            }
        }
    }

    fn render_help(&self, frame: &mut Frame, area: Rect) {
        let help = "↑↓ Navigate | Enter View | e Edit | f Filter | Esc Close";

            .style(Style::default().fg(Color::DarkGray));
        frame.render_widget(widget, area);
    }
}
```
